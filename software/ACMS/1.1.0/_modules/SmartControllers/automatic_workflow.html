    def __call__(self):
        log.debug("called Recipe Manager auto at {}".format(self))
        for key in CI.registered_types:
            CI.registered_types[key].set_project_properties(self._settings)
            for instance in CI.registered_types[key].get_registered_instances_keys():
                CI.registered_instances[instance].set_auto_mode_mapping(self._auto_mode_mapping_data)
                CI.registered_instances[instance].start()
        self.work()

    def work(self):
        while self.argument == 0:
            self.wait_please()
            for key in CI.registered_types:
                for instance in CI.registered_types[key].get_registered_instances_keys():
                    CI.registered_instances[instance].monitor_actions()
