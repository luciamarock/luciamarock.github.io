class PLC_settings:
    def __init__(self,parameters):
        if "valid_recipe_path" not in parameters.keys():
            self._recipes_base_dir = None
            logger.error("valid_recipe_path not specified for at least one instance of PLC")
        else:
            if parameters["valid_recipe_path"].endswith(os.path.sep):
                self._recipes_base_dir = parameters["valid_recipe_path"]
            else:
                self._recipes_base_dir = parameters["valid_recipe_path"] + os.path.sep
        if "verify_each_step" in parameters.keys():
            if parameters["verify_each_step"] == "yes":
                self._verify_each_step = True
            else:
                self._verify_each_step = False
        else: 
            self._verify_each_step = False
        if "optimized" in parameters.keys():
            if parameters["optimized"] == "yes":
                self._optimized = True
            else:
                self._optimized = False
        else:
            self._optimized = False
        
        self._known_PVs = {}

    def loadRecipe(self, recipename):
        result = "success"
        start_time = time.time()
        recipe_file = self._recipes_base_dir + recipename + ".json"
        if not os.path.isfile(recipe_file):
            logger.error("{} is not a valid recipe file".format(recipe_file))
            error("{} is not a valid recipe file".format(recipe_file))
            return 
        with open(recipe_file, 'r') as recipefile:
            data = json.load(recipefile)
        for key in data.keys(): #TODO this will not work as expexcted in python2 see TPG_Interface
            current_step = data[key]  # also handle the case in which there are no steps, see TPG_Interface 
            for pv in current_step.keys():
                if self._optimized:
                    if pv in self._known_PVs.keys():
                        currentPV = self._known_PVs[pv]["instance"] #already conneced 
                        if key != self._known_PVs[pv]["step"]:
                            self._known_PVs[pv]["step"] = key
                    else:
                        currentPV = epics.PV(pv)
                        if not currentPV.wait_for_connection():
                            logger.error("{} failed to connect".format(currentPV))
                            error("{} failed to connect".format(currentPV))
                            return
                        self._known_PVs[pv] = {}
                        self._known_PVs[pv]["instance"] = currentPV
                        self._known_PVs[pv]["step"] = key
                else:
                    currentPV = epics.PV(pv)
                    if not currentPV.wait_for_connection():
                        logger.error("{} failed to connect".format(currentPV))
                        error("{} failed to connect".format(currentPV))
                        return
                currentPV.put(current_step[pv])
            if self._verify_each_step:
                time.sleep(0.01)
                result = self._verify_step(current_step)
                if result != "success":
                    return 
        if result == "success":
            elapsed_time = time.time() - start_time
            logger.info("Elapsed time in putting settings is {}".format(elapsed_time))
            info("Elapsed time in putting settings is {}".format(elapsed_time))
        return result

    def verifyRecipe(self,recipename):
        result = "success"
        recipe_file = self._recipes_base_dir + recipename + ".json"
        if not os.path.isfile(recipe_file):
            logger.error("{} is not a valid recipe file".format(recipe_file))
            error("{} is not a valid recipe file".format(recipe_file))
            return 
        with open(recipe_file, 'r') as recipefile:
            data = json.load(recipefile)
        for key in data.keys(): #TODO no steps recipes 
            current_step = data[key]
            result = self._verify_step(current_step)
            if result != "success":
                return 
        return result


    
    def _verify_step(self,step_dictionary):
        result = "success"
        for pv in step_dictionary.keys():
            if pv in self._known_PVs.keys():
                currentPV = self._known_PVs[pv]["instance"]
            else:
                currentPV = epics.PV(pv)
                if not currentPV.wait_for_connection():
                    logger.error("{} failed to connect".format(currentPV))
                    error("{} failed to connect".format(currentPV))
                    return 
            current_value = currentPV.get()
            if current_value != step_dictionary[pv]:
                logger.error("current value = {} but recipe value = {} for PV = {}".format(current_value,step_dictionary[pv],pv))
                error("current value = {} but recipe value = {} for PV = {}".format(current_value,step_dictionary[pv],pv))
                result = "failure"
        return result
