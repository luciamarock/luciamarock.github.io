def init():
    """Check existence of recipe files and directories 
    
    In case not existing, this function generates the default recipes directories. 
    It also checks if, all the controller's type objects belong to the same substistem and if all the declared instances 
    belong to one and only one controller type.     
    """
    global subsystem
    log.debug("workflows interface started")
    complete = True # completeness flag to enable the start() from the main_program
    recipes_dir = UF.settings["workspace_dir"] + "recipes" + os.path.sep
    if not os.path.exists(recipes_dir):
        log.info("creating recipe directory: {}".format(recipes_dir))
        os.makedirs(recipes_dir)
    for ctype in CI.registered_types.keys():
        if subsystem is None:
            subsystem = CI.registered_types[ctype].subsystem
        elif CI.registered_types[ctype].subsystem != subsystem:
            log.warning("controller type from another subsystem found, this could lead to future errors")
        for cinst in CI.registered_types[ctype]._registered_instances:
            if cinst not in CI.registered_instances.keys():
                log.error("instance {} is declared in type {} but not found among the registered instances".format(cinst,ctype))
            else:
                inst_dir = recipes_dir + cinst
                if not os.path.exists(inst_dir):
                    log.info("creating instance sub-folder {}".format(inst_dir))
                    os.makedirs(inst_dir)
    return complete

def start():
    """Starts the program workflow 
    
    It loads the available recipes in the corresponding Controller's instance and start the User Interface
    """
    errors = CI.create_validators()
    for message in errors:
        log.error(message)
    import SmartControllers.automatic_workflow as AWF
    AWF.set_handler(console_handler)
    recipe_manager = AWF.create_instance()
    auto_mode_mapping = UF.settings["resources_directory"] + "auto_mode_mapping.json"
    with open(auto_mode_mapping, 'r') as json_file:
        data = json.load(json_file)
    recipe_manager.set_auto_mode_mapping_data(data)
    recipe_manager.general_settings(UF.settings)
    if float(sys.version[0]) < 3:
        t = Thread(target=recipe_manager) # , args=(1,)
        t.daemon = True
        t.start()
    else:
        t = Thread(target=recipe_manager, daemon=True) # , args=(1,), daemon=True
        t.start()
    if UF.settings["UI"] == "GUI":
        log.info("Starting PyQt5 Graphical User Interface")
        from PyQt5.QtWidgets import QApplication
        from GUI.gui_main import MainWindow
        app = QApplication(sys.argv)
        main_win = MainWindow(recipe_manager)
        main_win.show()
        # Exec loop blocks until main window is closed
        sys.exit(app.exec_())
    elif UF.settings["UI"] == "CLI":
        log.info("using the Command Line Interface")
        import utilities.command_line_interface as cli
        try:
            Uinp = "init"
            while Uinp != "exit":
                Uinp = cli.start().strip()
                log.info("Received command: {}".format(Uinp))
                #TODO commands lookup 
                if Uinp == "talk":
                    recipe_manager.talk()
            recipe_manager.stop()
        except KeyboardInterrupt:
            log.info("Recipe Management System interrupted by the User")
            recipe_manager.stop()
            sys.exit()
