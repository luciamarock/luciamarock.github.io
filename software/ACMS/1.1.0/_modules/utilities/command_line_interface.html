def start():
    prompt = "CLI interface started, type 'help' for a list of possible commands, 'exit' for closing:\n"
    if UF.settings["pyver"][0] < 3:
        user_input = raw_input(prompt)
    else:    
        user_input = input(prompt)

    return user_input
