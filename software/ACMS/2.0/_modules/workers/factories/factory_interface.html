    def create_worker(self, conf_file):
        """
        Creates a worker instance from an XML configuration file.

        Args:
            config_file (str): The path to the XML configuration file.

        Returns:
            object: An instance of the worker class, or None if an error occurs.
        """
        created_workers = {}
        ACTinterfacesPV = {}
        macros = {}
        if isinstance(conf_file, list):
            workers_path = self._settings["resources_directory"] + "workers" + os.path.sep
            for worker_config in conf_file:
                current_conf_file = workers_path + worker_config
                log.info("generating worker instance using {}".format(current_conf_file))
                worker_name = os.path.splitext(worker_config)[0]
                worker_instance, interfacePV, macroString = self._create_worker_from_file(current_conf_file, worker_name)
                if worker_instance is not None:
                    created_workers[worker_name] = worker_instance
                    if macroString in macros.keys():
                        log.error(f"cannot assign {macroString} to {worker_name}, macro already in use")
                    macros[worker_name] = macroString
                    if interfacePV in ACTinterfacesPV.keys() or not interfacePV:
                        log.error(f"cannot assign {interfacePV} to {worker_name}, unexistent or already in use")
                        return None, None
                    ACTinterfacesPV[interfacePV] = worker_name
        else:
            log.info("Operator request from HMI ...")
            #TODO do stuff, this means that the request comes from the Phoebus HMI
        return created_workers, ACTinterfacesPV, macros

    def _create_worker_from_file(self, filename, instance_name):
        try:
            tree = ET.parse(filename)
            root = tree.getroot()
            parameters = root.find("conditioning_parameters")
            parameters_dict = UF.element_to_dict(parameters) # pass this to worker instance 
            observables = root.find("observable_pvs")
            observable_variables = UF.element_to_list(observables) # also this list of observables
            interlocks_element = root.find("interlocks")
            interlocks = UF.element_with_attributes_to_dict(interlocks_element) # also this
            workflow_element = root.find("workflow_pvs")
            workflow_pvs = UF.element_to_dict(workflow_element)
            operation = root.find("epics")
            operation_variables = UF.element_to_dict(operation) # also this dictionary
            sequence = root.find("sequence").text
            auto_rearm_file = root.find("auto_rearm_file").text
            interfacePV = root.find("ACTinterface").text
            macroString = root.find("macroed_string").text
            if not os.path.splitext(sequence)[1]:
                sequence += '.csv'
            if not os.path.splitext(auto_rearm_file)[1]:
                auto_rearm_file += '.xml'
            sequences_dir = self._settings["resources_directory"] + "sequences" + os.path.sep
            autorearm_dir = self._settings["resources_directory"] + "autorearm" + os.path.sep
            if not os.path.exists(sequences_dir):
                log.error(f"Sequence directory '{sequences_dir}' does not exist. Creating it. Put your '{sequence}' there.")
                os.makedirs(sequences_dir)
                return None
            if not os.path.exists(autorearm_dir):
                log.error(f"Autorearm directory '{autorearm_dir}' does not exist. Creating it. Put your '{auto_rearm_file}' there.")
                os.makedirs(autorearm_dir)
                return None
            sequence_file = sequences_dir + sequence 
            log.info("loading {}".format(sequence_file))
            sequence_data = UF.parse_csv(sequence_file) # also this list of dictionaries 
            if not sequence_data:
                return None, None, None 
            autorearm_file_path = autorearm_dir + auto_rearm_file
            if os.path.exists(autorearm_file_path):
                log.info("loading {}".format(autorearm_file_path))
                rearm_conf_tree = ET.parse(autorearm_file_path)
                epics_pvs = rearm_conf_tree.getroot()
                auto_rearm_data = UF.parse_epics_xml_by_tag(epics_pvs)
            else:
                log.error("specified auto-rearm configuration file {} does not exists".format(autorearm_file_path))
                auto_rearm_data = None
            try:
                HMI_count_dict = {}
                HMI_count_dict["total_steps_pv_name"] = root.find("total_steps").text
                HMI_count_dict["current_step_pv_name"] = root.find("current_step").text
            except:
                HMI_count_dict = None
            sequence_all = {}
            try:
                sequence_all["sequence_combo"] = root.find("sequence_combo").text
                sequence_all["sequence_load"] = root.find("sequence_load").text
            except:
                sequence_all["sequence_combo"] = None
                sequence_all["sequence_load"] = None
            worker_validation = validators.worker_validator(parameters_dict, interlocks, instance_name)
            phase_one_results = worker_validation.check_data()
            if phase_one_results["status"] == "error":
                for message in phase_one_results["messages"]:
                    log.error(message)
                return None, None, None
            #TODO Verification of the sequence_data based on the rules, if successfull continue, else return None
            sequence_all["sequence_data"] = sequence_data
            sequence_all["sequence_name"] = sequence
            sequence_all["HMI_count_dict"] = HMI_count_dict
            return WF.Worker(parameters_dict, observable_variables, interlocks, workflow_pvs, operation_variables,  
            instance_name, auto_rearm_data, sequence_all), interfacePV, macroString

        except FileNotFoundError:
            log.error(f"Error: Configuration file '{filename}' not found.")
            return None
        except ET.ParseError:
            log.error(f"Error: Invalid XML format in '{filename}'.")
            return None
        except AttributeError:
            log.error(f"Error: Missing Attributes in '{filename}'.")
            return None
        except Exception as e:
            log.error(f"An unexpected error occurred: {e}")
            return None
