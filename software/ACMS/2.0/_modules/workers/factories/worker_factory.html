    def __init__(self, parameters_dict, observable_variables, all_interlocks, workflow_pvs, operation_variables,  
    instance_name, auto_rearm_data, sequence_all):
        """
        Worker instance representing the Generic Workflow.
        
        :param parameters_dict: conditioning_parameters node of the xml file (converted in dictionary)
        :param observable_variables: observable_pvs node of the xml file (converted in list of PV names)
        :param rearmable_interlocks: rearmable_interlocks node of the xml file (converted in list of PV names)
        :param operation_variables: epics node of the xml file (converted in dictionary of format alias_name:PVname)
        :param sequence_data: ordered list of steps where each element is a dictionary (column_name:value) 
        """
        self._settings = tool_settings.settings
        self.name = instance_name
        self.sequence_name = sequence_all["sequence_name"]
        self.sequence = sequence_all["sequence_data"]
        self.parameters = parameters_dict #TODO contains auto_rearm PV (and validate xml tags)
        self.configuration = operation_variables
        self.observables = observable_variables
        self.interlocks = all_interlocks #TODO validate xml attributes format
        self.RF_system_starting = True
        self.workflow_pvs = workflow_pvs
        self.workerPVs = {} # a place where to store this worker PV instances {"typology":{PV_name: PV_instance}}
        self.initialState = {} # {PV_name : value}
        self._initConditionsChanged = True
        self._status = "not running"
        self.sens_digit = None
        self.auto_rearm_manager = None  # Will be initialized after connection
        self.auto_rearm_data = auto_rearm_data
        self._warning_dict = {}
        self._warning_dict["status"] = "free"
        self._error_dict = {}
        self._error_dict["status"] = "free"
        self._generate_steps_update_dict(sequence_all["HMI_count_dict"])        
        self._HMI_pv_names = {}
        self._HMI_pv_names["sequence_combo"] = sequence_all["sequence_combo"]
        self._HMI_pv_names["sequence_load"] = sequence_all["sequence_load"]

    def _generate_steps_update_dict(self, HMI_count_dict):
        if HMI_count_dict is None:
            self._steps_update_dict = None
        else:
            self._steps_update_dict = {}
            self._steps_update_dict["total_steps_pv_name"] = epics.PV(HMI_count_dict["total_steps_pv_name"])
            if not self._steps_update_dict["total_steps_pv_name"].wait_for_connection(timeout=1):
                self._steps_update_dict = None
                return
            self._steps_update_dict["current_step_pv_name"] = epics.PV(HMI_count_dict["current_step_pv_name"])
            if not self._steps_update_dict["current_step_pv_name"].wait_for_connection(timeout=1):
                self._steps_update_dict = None
                return

    def connect(self):
        connection_error = False 
        #connecting to the observables
        self.workerPVs["observables"] = {}
        for variable in self.observables:
            self.workerPVs["observables"][variable] = epics.PV(variable)
            if not self.workerPVs["observables"][variable].wait_for_connection(timeout=1):
                log.error(f"{self.name} failed to connect to {variable}, stopping now")
                connection_error = True 
                break 
            self.initialState[variable] = self.workerPVs["observables"][variable].get()
        #connecting to operation variables
        if not connection_error:
            log.info(f"worker {self.name} fully connected to its observable")
            self.workerPVs["operation"] = {}
            for key, variable  in self.configuration.items(): #TODO make sure to validate the keys
                self.workerPVs["operation"][variable] = epics.PV(variable)
                if not self.workerPVs["operation"][variable].wait_for_connection(timeout=1):
                    log.error(f"{self.name} failed to connect to {variable}, stopping now")
                    connection_error = True 
                    break 
                self.initialState[variable] = self.workerPVs["operation"][variable].get()
        #connecting to workflow PVs 
        if not connection_error:
            log.info(f"worker {self.name} fully connected to its operation PVs")
            self.workerPVs["workflow"] = {}
            for key, variable  in self.workflow_pvs.items(): #TODO make sure to validate the keys
                self.workerPVs["workflow"][variable] = epics.PV(variable)
                if not self.workerPVs["workflow"][variable].wait_for_connection(timeout=1):
                    log.error(f"{self.name} failed to connect to {variable}, stopping now")
                    connection_error = True 
                    break
                self.initialState[variable] = self.workerPVs["workflow"][variable].get()
        #connecting to auto-rearm PV 
        if not connection_error:
            log.info(f"worker {self.name} fully connected to its workflow PVs")
            self.workerPVs["rearming"] = {}
            auto_rearm_PV_name = self.parameters["manual_restart"] #TODO ensure the exitence of this during validation (use mapping)
            self.workerPVs["rearming"][auto_rearm_PV_name] = epics.PV(auto_rearm_PV_name)
            if not self.workerPVs["rearming"][auto_rearm_PV_name].wait_for_connection(timeout=1):
                log.error(f"{self.name} failed to connect to {auto_rearm_PV_name}, stopping now")
                connection_error = True
            self.initialState[auto_rearm_PV_name] = self.workerPVs["rearming"][auto_rearm_PV_name].get()
            ramp_speed_pv_name = self.parameters["cavity_voltage_ramp_speed"]
            ramp_speed_pv_instance = epics.PV(ramp_speed_pv_name)
            if not ramp_speed_pv_instance.wait_for_connection(timeout=1):
                log.error(f"{self.name} failed to connect to {ramp_speed_pv_name}, stopping now")
                connection_error = True
            self.parameters["cavity_voltage_ramp_speed"] = ramp_speed_pv_instance
            cavity_phase_pv_name = self.parameters["cavity_phase"]
            cavity_phase_pv_instance = epics.PV(cavity_phase_pv_name)
            if not cavity_phase_pv_instance.wait_for_connection(timeout=1):
                log.error(f"{self.name} failed to connect to {cavity_phase_pv_name}, stopping now")
                connection_error = True
            self.parameters["cavity_phase"] = cavity_phase_pv_instance
        #connecting the interlocks 
        if not connection_error:
            log.info(f"{self.name} connected to its auto-rearm PV, connecting the Interlocks")
            self.workerPVs["interlocks"] = {}
            for key, variables_dict  in self.interlocks.items():
                self.workerPVs["interlocks"][key] = variables_dict
                self.workerPVs["interlocks"][key]["PV_instance"] = epics.PV(key)
                if not self.workerPVs["interlocks"][key]["PV_instance"].wait_for_connection(timeout=1):
                    log.error(f"{self.name} failed to connect to Interlock {key}, stopping now")
                    connection_error = True
                    break
                self.initialState[key] = self.workerPVs["interlocks"][key]["PV_instance"].get()
        if not connection_error:
            log.info(f"{self.name} fully connected to its interlocks, connecting HMI PVs")
            if self._HMI_pv_names["sequence_combo"] is not None and self._HMI_pv_names["sequence_load"] is not None:
                self._sequence_combo_pv = epics.PV(self._HMI_pv_names["sequence_combo"])
                if not self._sequence_combo_pv.wait_for_connection(timeout=1):
                    log.error(f"{self.name} failed to connect to {self._HMI_pv_names['sequence_combo']}, stopping now")
                    connection_error = True
                self._sequence_load_pv = epics.PV(self._HMI_pv_names["sequence_load"])
                if self._sequence_load_pv.wait_for_connection(timeout=1): 
                    self._sequence_load_pv.add_callback(self._selected_sequence_callback)
                else:
                    log.error(f"{self.name} failed to connect to {self._HMI_pv_names['sequence_load']}, stopping now")
                    connection_error = True        
                if not connection_error:
                    self._update_sequence_list()
        
        if not connection_error:
            log.info(f"{self.name} fully connected to the EPICS layer")
            # Initialize the auto-rearm manager
            self.auto_rearm_manager = ARM.AutoRearmManager(
                self.workerPVs, 
                self.parameters, 
                self.configuration, 
                self.workflow_pvs, 
                self.interlocks,
                self.name
            )
            log.info("Connecting its auto-rearm Manager ... ")
            connection_error = self.auto_rearm_manager.connect(self.auto_rearm_data)
        
        return not connection_error

    def load_compare_save(self,workspace_directory):
        workers_startup_state_path = workspace_directory + "workers_startup_state"
        workers_startup_state_file = workspace_directory + "workers_startup_state" + os.path.sep + self.name + ".json"
        if not os.path.exists(workers_startup_state_path):
            os.makedirs(workers_startup_state_path)
        if not os.path.exists(workers_startup_state_file):
            with open(workers_startup_state_file, 'w') as f:
                json.dump(self.initialState, f, indent=4)
        else:
            with open(workers_startup_state_file, 'r') as f:
                temp_dict = json.load(f)
            if self.initialState == temp_dict:
                self._initConditionsChanged = False
                log.warning(f"unchanged initial conditions for worker {self.name}")
            else:
                with open(workers_startup_state_file, 'w') as f:
                    json.dump(self.initialState, f, indent=4)
                log.info(f"initial conditions for worker {self.name} are changed, updating the .json file")
        return

    def run(self):
        log.info(f"{self.name} started the execution of sequence {self.sequence_name}")
        number_of_steps = len(self.sequence)
        self.current_step = 0
        self._status = "running"
        pending_warnings = []
        pending_errors = []
        pending_interlocks = []
        init_conf = self.get_current_system_state()
        # removing this condition because it's checked inside the auto_rearm_manager:
        #  and self.auto_rearm_manager.get_rearm_counter() - 1 < int(self.parameters["retry_attempts"]) 
        while self.current_step < number_of_steps and self._status == "running":
            self._update_steps_count(number_of_steps, self.current_step)
            loop_start_time = time.time()
            while time.time() - loop_start_time < float(self.parameters["stability_check_interval"]):
                #call all the checks 
                report_dict = self.check_observables()
                if report_dict:
                    error_list = report_dict["errors"]
                    warning_list = report_dict["warnings"]
                    for pv_name in error_list:
                        if pv_name not in pending_errors:
                            pending_errors.append(pv_name)
                    for pv_name in warning_list:
                        if pv_name not in pending_warnings:
                            pending_warnings.append(pv_name)
                else:
                    pending_warnings = []
                    pending_errors = []
                list_of_interlocks = self.auto_rearm_manager.detect_interlocks()
                if list_of_interlocks:
                    for pv_name in list_of_interlocks:
                        if pv_name not in pending_interlocks:
                            pending_interlocks.append(pv_name)
                else:
                    pending_interlocks = []
                time.sleep(float(self.parameters["retry_delay"])) #TODO verifications: retry_delay must be a fraction of stability_check_interval
            if len(pending_errors) > 0:
                for pv_name in pending_errors:
                    log.error(f"{self.name} has detected error in observable {pv_name}")
                current_configuration = self.get_current_system_state()
                #if current_configuration == init_conf: #TODO remove these two variables 
                if self.current_step < 2:
                    log.error("encountered error at the lowest step, aborting.")
                    self.abort_due_to_error()
                    break
                if self._error_dict["status"] == "free":
                    self._error_dict["status"] = "busy"
                    if self._error_dict["previous_derivative"] < 0:
                        self.current_step = self._error_dict["min_rel_index"] + 1
                    rollback_step_result = self.rollback_step()
                    if rollback_step_result:
                        self.current_step += 1
                        log.info(f"{self.name} rollback step successfully applyied, updating step")
                continue  
            else:
                if self._error_dict["status"] == "busy":
                    self._error_dict["status"] = "free"
            if len(pending_warnings) > 0:
                for pv_name in pending_warnings:
                    log.warning(f"{self.name} has detected warning in observable {pv_name}")
                if self._warning_dict["status"] == "free":
                    self.case_warning()
                continue
            else:
                if self._warning_dict["status"] == "busy":
                    previous_ramp_speed = self._warning_dict["ramp_speed"]
                    log.info(f"worker {self.name} is setting ramp_speed to previous value {previous_ramp_speed}")
                    self.parameters["cavity_voltage_ramp_speed"].put(previous_ramp_speed)
                    self._warning_dict["status"] = "free"
            if len(pending_interlocks) > 0:
                for interlock_pv in pending_interlocks:
                    log.error(f"{self.name}: detected interlock {interlock_pv}")
                    rearming_result = self.auto_rearm_manager.execute_rearming_workflow(interlock_pv)
                    if not rearming_result:
                        log.error(f"{self.name}: Failed to rearm interlock {interlock_pv}. Aborting now ...")
                        self.abort_due_to_error()
                        break
                if self._error_dict["status"] == "free":
                    self._error_dict["status"] = "busy"
                    if self._error_dict["previous_derivative"] < 0:
                        self.current_step = self._error_dict["min_rel_index"]
                continue
            else:
                if self._error_dict["status"] == "busy":
                    self._error_dict["status"] = "free"
            result = self.apply_step()
            if not result:
                continue
            self.current_step += 1
        if self._status == "running":
            self._status = "completed"
            log.info(f"worker {self.name} finished its workflow")

    def case_warning(self):
        """ Handle a warning state. """
        self._warning_dict["status"] = "busy"
        voltage_readback_pv = self.workerPVs["operation"][self.configuration["power_get"]]
        voltage_setpoint_pv = self.workerPVs["operation"][self.configuration["power_set"]]
        current_ramp_speed = self.parameters["cavity_voltage_ramp_speed"].get()
        self._warning_dict["ramp_speed"] = current_ramp_speed
        new_ramp_speed = self.parameters["user_defined_ramp_speed"]
        log.info(f"worker {self.name} is setting ramp_speed to UserDefined")
        self.parameters["cavity_voltage_ramp_speed"].put(new_ramp_speed)
        voltage_setpoint_pv_name = self.configuration["power_set"]
        current_voltage = voltage_readback_pv.get()
        new_setpoint = current_voltage * float(self.parameters["Vwarning_multiplier"])
        log.info(f"worker {self.name} is setting {new_setpoint} to {voltage_setpoint_pv_name}")
        voltage_setpoint_pv.put(new_setpoint) # wait=True ?
        default_min_stable_time = UF.calculate_min_stable_time(0,0,0)
        self.parameters["min_stable_time"] = default_min_stable_time

    def get_current_system_state(self):
        """ Retrieve current system state as a dictionary. """
        if not self.sens_digit:
            self.sens_digit = round(math.log10(1/float(self.parameters["readback_tolerance"])))
        state_dict = {}
        operation_dict = self.workerPVs["operation"]
        for key, value in self.configuration.items():
            instance = operation_dict[value]
            pv_value = round(instance.get(),self.sens_digit)
            state_dict[key] = pv_value
        return state_dict

    def check_observables(self):
        """ Check if any observable PVs are in an error or warning state. """
        # EPICS severity codes: 0=NO_ALARM, 1=MINOR, 2=MAJOR, 3=INVALID
        #instance.get() gets the value, instance.severity gets the severity
        observables_dict = self.workerPVs["observables"]
        list_of_warning = []
        list_of_errors = []
        for key, instance in observables_dict.items():
            if float(instance.severity) == 2:
                list_of_errors.append(key)
            elif float(instance.severity) == 1:
                list_of_warning.append(key)
        if len(list_of_warning) > 0 or len(list_of_errors) > 0:
            return_dict = {}
            return_dict["warnings"] = list_of_warning
            return_dict["errors"] = list_of_errors
            return return_dict
        return None

    def rollback_step(self):
        """ Rollback to the previous step in case of an error. """
        result = None 
        if self.current_step > 0:
            self.current_step -= 1
            log.info(f"{self.name}: Rolling back to step {self.current_step}.")
            result = self.apply_step()
        else:
            log.error(f"{self.name}: Cannot roll back further. Aborting.")
            self.abort_due_to_error()
        return result

    def _track_derivatives(self, current_sequence_step, current_worker_state):
        if not "previous_derivative" in self._error_dict.keys():
            self._error_dict["previous_derivative"] = 0
            self._error_dict["min_rel_index"] = 0
        current_derivative = float(current_sequence_step["power_set"]) - current_worker_state["power_set"]
        if current_derivative >= 0 and self._error_dict["previous_derivative"] < 0:
            self._error_dict["min_rel_index"] = self.current_step - 1
        self._error_dict["previous_derivative"] = current_derivative

    def apply_step(self):
        """ Apply the configuration settings for the current step. """
        log.info(f"{self.name}: applying step {self.current_step} of the sequence {self.sequence_name}")
        operation_dict = self.workerPVs["operation"]
        set_dict = {}
        get_dict = {}
        tasks = self.sequence[self.current_step] #TODO validate keys 
        current_state = self.get_current_system_state()
        self._track_derivatives(tasks,current_state)
        max_delta = 0
        for key in tasks:
            if "set" in key:
                set_dict[self.configuration[key]] = tasks[key]
                get_key = key.replace("set","get")
                current_delta = abs(current_state[get_key] - float(tasks[key]))
                if current_delta > max_delta:
                    max_delta = current_delta
                get_dict[self.configuration[get_key]] = tasks[key]
        #apply set_dict 
        for pv_name, value in set_dict.items():
            pv_instance = operation_dict[pv_name]
            status = pv_instance.put(float(value), wait=True)
            if not bool(status):
                log.error(f"{self.name}: Failed to set {pv_name} to {value}")
                return None
        #TODO verify readbacks from an additional method (check auto_rearm_manager), also continuously check observables
        readback_start_time = time.time()
        readback_completed = False
        ramp_rate = self.parameters["cavity_voltage_ramp_speed"].get()
        cavity_phase = self.parameters["cavity_phase"].get()
        new_min_stable_time = UF.calculate_min_stable_time(max_delta, ramp_rate, cavity_phase)
        self.parameters["min_stable_time"] = new_min_stable_time
        while time.time() - readback_start_time < float(self.parameters["min_stable_time"]) and not readback_completed:
            readback_completed = True
            for pv_name, value in get_dict.items():
                pv_instance = operation_dict[pv_name]
                readback_value = pv_instance.get()
                if round(float(value),self.sens_digit) != round(readback_value,self.sens_digit):
                    readback_completed = False
            time.sleep(0.1) #TODO hardcoded 100 ms (check xml)
        elapsed_time = time.time() - readback_start_time
        if not readback_completed:
            log.error(f"{self.name}: Could not verify readback after {elapsed_time}. Trying again ...")
            return None
        log.info(f"{self.name}: Readback Values verified after {elapsed_time}")
        action = self.sequence[self.current_step]["Action"]
        method_name = action.lower()
        method = getattr(self, method_name, None)
        if method is not None:
            method()
        log.info(f"{self.name}: Step {self.current_step} applied successfully.")
        return True
