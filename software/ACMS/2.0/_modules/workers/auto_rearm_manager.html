class AutoRearmManager:
    """
    Manages the auto-rearming procedure for interlocks in the conditioning workflow.
    This class is responsible for detecting, validating, and executing interlock rearming.
    """
    
    def __init__(self, worker_pvs, parameters, configuration, workflow_pvs, interlocks,worker_name):
        """
        Initialize the AutoRearmManager.
        
        :param worker_pvs: Dictionary containing all PV instances organized by type
        :param parameters: Worker parameters dictionary containing rearming settings
        :param configuration: Operation variables configuration
        :param workflow_pvs: Workflow PVs configuration
        :param interlocks: Interlocks configuration
        """
        self.worker_name = worker_name
        self.worker_pvs = worker_pvs
        self.parameters = parameters
        self.configuration = configuration
        self.workflow_pvs = workflow_pvs
        self.interlocks = interlocks
        self.rearm_registry = {}
        self.arc_counter = 0
        self.sens_digit = None
        self.auto_rearm_data = {}

    def connect(self,data):
        """
        Connect to the auto-rearm data.
        Return False if no connection errors occured 
        """
        if data:
            log.info("Starting connection to the auto-rearm PVs")
            for key in data.keys():
                self.auto_rearm_data[key] = {}
                data_dict = data[key]
                for subkey, value in data_dict.items():
                    self.auto_rearm_data[key][subkey] = value
                pv_instance = epics.PV(data_dict["pv_name"]) 
                if not pv_instance.wait_for_connection(timeout=1):
                    log.error("failed to connect to {}, stopping now".format(data_dict["pv_name"]))
                    return True 
                self.auto_rearm_data[key]["PV_instance"] = pv_instance
            log.info("Auto Rearm Manager for this worker is fully connected to the EPICS layer")
        else:
            log.warning("Auto Rearm Manager has nothing to do")
        
        return False

    def detect_interlocks(self):
        """ 
        Detect and return currently active interlocks.
        
        :return: List of active interlock PV names, or None if no interlocks
        """
        interlocks_dict = self.worker_pvs["interlocks"]
        active_interlocks = []
        
        for key in interlocks_dict.keys():
            interlock_severity = interlocks_dict[key]["PV_instance"].severity
            if interlock_severity > 0:
                active_interlocks.append(key)
                
        return active_interlocks if active_interlocks else None

    def can_auto_rearm(self, interlock_pv_name):
        """
        Check if an interlock can be auto-rearmed.
        
        :param interlock_pv_name: Name of the interlock PV
        :return: True if auto-rearming is possible, False otherwise
        """
        # Check if manual-restart is enabled
        manual_restart_value = self.worker_pvs["rearming"][self.parameters["manual_restart"]].get()
        if manual_restart_value < 1:
            log.info(f"Manual Restart is OFF, checking other conditions.")
        else:
            return True
            
        # Check if quick rearm mode is enabled
        if self.worker_pvs["workflow"][self.workflow_pvs["quick_rearm_mode"]].get() < 1:
            log.error(f"Quick Rearm Mode DISABLED, cannot rearm interlock")
            return False
            
        # Check if interlock is rearmable
        interlock_dict = self.worker_pvs["interlocks"][interlock_pv_name]
        if interlock_dict["rearmable"].lower() == "false":
            log.error(f"Interlock {interlock_pv_name} is not rearmable.")
            return False
        else:
            interlock_instance = interlock_dict["PV_instance"]
            interlock_codes = interlock_dict["codes"]
            codes_list = [int(code.strip()) for code in interlock_codes.split(',') if code.strip()]
            if codes_list and interlock_instance.get() not in codes_list:
                log.error(f"Value of Interlock {interlock_pv_name} is not in the list of allowed codes.")
                return False
            
        # Check rearm attempt limits and number of arcs limit 
        if interlock_pv_name not in self.rearm_registry:
            self.rearm_registry[interlock_pv_name] = 0
        elif self.rearm_registry[interlock_pv_name] >= int(self.parameters["retry_attempts"]):
            self.arc_counter += 1
            log.error(f"Interlock {interlock_pv_name} has been rearmed too many times.")
            return False
        if self.arc_counter >= int(self.parameters["arc_limit"]):
            log.error(f"Number of arcs has exceeded the limit.")
            return False
        
        return True

    def execute_rearming_workflow(self, interlock_pv_name):
        """
        Execute the complete auto-rearming workflow for an interlock.
        
        :param interlock_pv_name: Name of the interlock PV to rearm
        :return: True if rearming was successful, False otherwise
        """
        if not self.can_auto_rearm(interlock_pv_name):
            return False
            
        # Execute the rearming procedure
        success = self._perform_rearming_procedure(interlock_pv_name)
        
        if success:
            # Update rearm registry
            self.rearm_registry[interlock_pv_name] += 1
            self.arc_counter = 0
        else:
            self.arc_counter += 1
            
        return success

    def _perform_rearming_procedure(self, interlock_pv_name):
        """
        Perform the actual rearming procedure.
        
        :param interlock_pv_name: Name of the interlock PV to rearm
        :return: True if rearming was successful, False otherwise
        """
        log.info(f"Attempting to rearm interlock {interlock_pv_name}.")
        
        # Get current system state
        current_state = self._get_current_system_state()
        self._sync_frequency_mode()
        self._update_rearm_registry()
        
        # Step 1: Shutting down the system
        if self.arc_counter > 0:
            log.info("Previous rearming attempt failed")
            log.info("Auto Rearm Manager of worker: {}, is going to shutdown the power now".format(self.worker_name))
            self.parameters["cavity_voltage_ramp_speed"].put(7) # Ramp to Now 
            self._sleep()
            target_value = 0
            self.worker_pvs["operation"][self.configuration["power_set"]].put(target_value)
            pv_instance = self.worker_pvs["operation"][self.configuration["power_get"]]
            pv_description = "Cavity Voltage"
            self._verify_readback(target_value,pv_instance,pv_description)
        else:
            log.info("No previous failed rearming attempts were found")
            self.rearm_registry["prev_ampref"] = self.auto_rearm_data["min_reference_voltage_setpoint"]["PV_instance"].get()
            log.info("Auto Rearm Manager of worker: {}, is going to shutdown the power now".format(self.worker_name))
            self.parameters["cavity_voltage_ramp_speed"].put(7) # Ramp to Now
            self._sleep()
            target_value = 0
            self.worker_pvs["operation"][self.configuration["power_set"]].put(target_value)
            self.auto_rearm_data["amplitude_phase_control"]["PV_instance"].put(0)
            self.auto_rearm_data["beam_control_feedforward"]["PV_instance"].put(0)
            pv_instance = self.worker_pvs["operation"][self.configuration["power_get"]]
            pv_description = "Cavity Voltage"
            self._verify_readback(target_value,pv_instance,pv_description)
            self.rearm_registry["auto_freq"] = self.auto_rearm_data["auto_frequency_control_status"]["PV_instance"].get()
            self.rearm_registry["man_freq"] = self.auto_rearm_data["manual_frequency_control_status"]["PV_instance"].get()
        self.rearm_registry["script_status"] = 2
        
        # Step 2: turn off all frequency modes and loops
        log.info("Turning OFF all frequency modes and loops")
        self.auto_rearm_data["manual_frequency_control_command"]["PV_instance"].put(0)
        self.auto_rearm_data["auto_frequency_control_command"]["PV_instance"].put(0)
        target_value = 0
        self._looped_operations(self.parameters["loops_PV_suffix"],target_value)
        self.rearm_registry["script_status"] = 3
        
        # Step 3: Rearm the interlock
        interlock_instance = self.interlocks[interlock_pv_name]["PV_instance"]
        log.info(f"Rearming interlock {interlock_pv_name}")
        if self.interlocks[interlock_pv_name]["codes"]:
            self._looped_operations(self.parameters["reset_PV_suffix"],1)
            self._sleep()
            self._looped_operations(self.parameters["reset_PV_suffix"],0)
        else:
            interlock_instance.put(0)    
        self.rearm_registry["script_status"] = 4
        
        # Step 4: synchronizing the conditioning process with the system's operating mode
        if self.rearm_registry["continuous_wave_operation"] == 0:
            self.auto_rearm_data["conditioning_mode_control"]["PV_instance"].put(1)
        if self.rearm_registry["continuous_wave_operation"] == 1:
            self.auto_rearm_data["conditioning_mode_control"]["PV_instance"].put(0)
        self.rearm_registry["script_status"] = 5
        
        # Step 5: Set cavity voltage to percentage of previous value
        # prev_cavvol = current_state["power_set"]
        percentage = float(self.parameters["Vrestore_multiplier"])
        new_cavity_voltage_value = current_state["power_set"] * percentage
        log.info(f"Setting cavity voltage to {new_cavity_voltage_value}")
        if self.rearm_registry["is_pulsed_on_cw_mode"] == 0:
            self.worker_pvs["operation"][self.configuration["power_set"]].put(new_cavity_voltage_value)
            pv_instance = self.auto_rearm_data["reference_cavity_voltage"]["PV_instance"]
            pv_description = "Reference Cavity Voltage"
            self._verify_readback(new_cavity_voltage_value,pv_instance,pv_description)
        else: # == 1
            previous_amp_ref = self.rearm_registry["prev_ampref"]
            self.auto_rearm_data["min_reference_voltage_setpoint"]["PV_instance"].put(previous_amp_ref)
            self.worker_pvs["operation"][self.configuration["power_set"]].put(new_cavity_voltage_value)
            pv_instance = self.worker_pvs["operation"][self.configuration["power_get"]]
            pv_description = "Cavity Voltage"
            self._verify_readback(new_cavity_voltage_value,pv_instance,pv_description)
        self.rearm_registry["script_status"] = 6
        
        # Step 6: Turn the Loops ON
        log.info("Verifying Forward Power Readback")
        enabling_PV_instance = self.auto_rearm_data["forward_power_readback"]["PV_instance"]
        pv_value = enabling_PV_instance.get()
        min_threashod = float(self.parameters["min_FW_power"])
        stable_time = float(self.parameters["min_stable_time"])
        start_time = time.time()
        elapsed_time = 0 
        while pv_value < min_threashod and elapsed_time < stable_time:
            pv_value = enabling_PV_instance.get()
            elapsed_time = time.time() - start_time
            self._sleep()
        if pv_value < min_threashod:
            log.error("Verification of Forward Power Readback value has timed-out, can't proceed")
            return False #TODO increase arc count and do something else (maybe)
        log.info("Turning the Loops ON")
        self._looped_operations(self.parameters["loop_on_suffix"],1)
        self._looped_verifications(self.parameters["loop_rdbk_suffix"],1)
        self.rearm_registry["script_status"] = 16

        # Step 7: Handle frequency mode restoration
        if self.rearm_registry["auto_freq"] == 1 and self.rearm_registry["man_freq"] == 0:
            self.auto_rearm_data["auto_frequency_control_command"]["PV_instance"].put(1)
        if self.rearm_registry["auto_freq"] == 0 and self.rearm_registry["man_freq"] == 1:
            self.auto_rearm_data["manual_frequency_control_command"]["PV_instance"].put(1)
        self.rearm_registry["script_status"] = 17
            
        # Step 8: Restore cavity voltage to previous value
        log.info("Setting Cavity Voltage ramp to user_defined")
        if self.rearm_registry["is_pulsed_on_cw_mode"] == 1:
            ramp_speed = self.auto_rearm_data["final_voltage_ramp_speed"]["PV_instance"].get()
            self.parameters["cavity_voltage_ramp_speed"].put(ramp_speed) # wait=True ?
        else:
            second_restore_multiplier = float(self.parameters["V2restore_multiplier"])
            intermediate_cavity_voltage = second_restore_multiplier * current_state["power_set"]
            ramp_speed = self.auto_rearm_data["first_stage_voltage_ramp"]["PV_instance"].get()
            self.parameters["cavity_voltage_ramp_speed"].put(ramp_speed) # wait=True ?
            log.info(f"Setting Cavity Voltage value to intermediate value {intermediate_cavity_voltage}")
            pv_instance = self.auto_rearm_data["reference_cavity_voltage"]["PV_instance"]
            pv_description = "Reference Cavity Voltage"
            self.worker_pvs["operation"][self.configuration["power_set"]].put(intermediate_cavity_voltage)
            self._verify_readback(intermediate_cavity_voltage,pv_instance,pv_description,True)
        ramp_speed = self.auto_rearm_data["second_stage_voltage_ramp"]["PV_instance"].get()
        self.parameters["cavity_voltage_ramp_speed"].put(ramp_speed) # wait=True ?
        prev_cav_value = current_state["power_set"]
        log.info(f"Setting Cavity Voltage value to {prev_cav_value}")
        pv_instance = self.worker_pvs["operation"][self.configuration["power_get"]]
        pv_description = "Cavity Voltage"
        self.worker_pvs["operation"][self.configuration["power_set"]].put(prev_cav_value)
        self._verify_readback(prev_cav_value,pv_instance,pv_description)
        self.rearm_registry["script_status"] = 22
        
        return True

    def _get_current_system_state(self):
        """
        Get current system state for operation variables.
        
        :return: Dictionary of current operation variable values
        """
        if not self.sens_digit:
            self.sens_digit = round(math.log10(1/float(self.parameters["readback_tolerance"])))
            
        state_dict = {}
        operation_dict = self.worker_pvs["operation"]
        
        for key, value in self.configuration.items():
            instance = operation_dict[value]
            pv_value = round(instance.get(), self.sens_digit)
            state_dict[key] = pv_value
        
        for key, value in self.workflow_pvs.items():
            pv_value = self.worker_pvs["workflow"][value].get()
            state_dict[key] = pv_value
            
        return state_dict

    def _verify_readback(self, expected_value,pv_instance,pv_description,monitoring=False):
        """
        Verify that cavity voltage readback matches expected value.
        
        :param expected_value: Expected cavity voltage value
        :return: True if verification successful, False otherwise
        """
        readback_value = pv_instance.get()
        elapsed_time = 0
        start_time = time.time()
        ramp_rate = self.parameters["cavity_voltage_ramp_speed"].get()
        cavity_phase = self.parameters["cavity_phase"].get()
        new_min_stable_time = UF.calculate_min_stable_time(abs(readback_value - expected_value), ramp_rate, cavity_phase)
        self.parameters["min_stable_time"] = new_min_stable_time
        while (round(readback_value, self.sens_digit) != round(expected_value, self.sens_digit) and 
               elapsed_time < float(self.parameters["min_stable_time"])):
            readback_value = pv_instance.get()
            elapsed_time = time.time() - start_time
            if monitoring:
                self._continuous_monitoring() #TODO use the return variable 
            self._sleep()
            
        if round(readback_value, self.sens_digit) == round(expected_value, self.sens_digit):
            log.info("{} readback value has been verified".format(pv_description))
            return True
            
        log.error("Verification of {} readback value has timed-out".format(pv_description))
        return False

    def _update_rearm_registry(self):
        """Get the current rearm registry."""
        self.rearm_registry["script_status"] = 1
        conditioning_mode_control = self.auto_rearm_data["conditioning_mode_control"]["PV_instance"].get()
        if conditioning_mode_control == 0:
            self.rearm_registry["continuous_wave_operation"] = 1
        else:
            self.rearm_registry["continuous_wave_operation"] = 0
        min_reference_voltage_setpoint = self.auto_rearm_data["min_reference_voltage_setpoint"]["PV_instance"].get()
        if conditioning_mode_control == 1 and min_reference_voltage_setpoint > 1:
            self.rearm_registry["is_pulsed_on_cw_mode"] = 1
        else:
            self.rearm_registry["is_pulsed_on_cw_mode"] = 0

    def _looped_operations(self,suffix,target_value):
        for chain_num in range(1, int(self.parameters["chains_amount"])+1):
            for i in range(int(self.parameters["cavity_amount"])):
                cavity_char = chr(ord('A') + i)
                pv_name = (
                f"{self.parameters['main_prefix']}"
                f"{self.parameters['chain_prefix']}"
                f"{chain_num}"
                f"{self.parameters['cavity_prefix']}"
                f"{cavity_char}"
                f"{suffix}"
                )
                pv_instance = epics.PV(pv_name)
                try:
                    if pv_instance.wait_for_connection(timeout=0.25):
                        pv_instance.put(target_value, wait=True) # Synchronous --> wait for a confirmation from the EPICS server that the new value has been successfully written
                    else:
                        log.error(f"Failed to connect to PV {pv_name}")
                except Exception as e:
                    log.error(e)
                finally:
                    if pv_instance.connected:
                        pv_instance.disconnect()
                        log.info(f"Successfully set {pv_name} to {target_value}")
