# -*- coding: utf-8 -*-
"""
Phase Alignment Worker Module
=============================

Provides the :class:`PhaseAlignmentWorker` execution engine to perform real-time
multi-chain RF phase alignment for RFQ cavities. 

The worker orchestrates EPICS Channel Access PV connections, chain hardware isolation,
oscilloscope sampling via :mod:`utilities.scopelib`, Levenberg-Marquardt field fitting
via :mod:`utilities.rfq_calib_utils`, and automated safety validation checks.
"""

import utilities.rfq_calib_utils as RFQCU
import utilities.scopelib as scopelib
import utilities.util_functions as UF
import threading
import epics
import time
import math
import logging 
log = logging.getLogger(__name__)

class PhaseAlignmentWorker:
    """Orchestrates multi-chain RF phase alignment, hardware isolation, and safety checks.

    Communicates with EPICS Process Variables (PVs) to manage master and slave amplifier
    states, monitor runtime user commands (RUN, PAUSE, ABORT), and perform least-squares
    optimization of cavity phase setpoints.

    :param data: Configuration dictionary containing hardware topologies, worker parameters,
                 EPICS PV configurations, and safety threshold limits.
    :type data: dict
    """
    def __init__(self, data):
        """Initialize the PhaseAlignmentWorker with topology data and establish PV connections."""
        log.info("PhaseAlignmentWorker initialized")
        self.data = data
        mode = data["worker_parameters"]["mode"]
        log.info(f"mode is {mode}")
        if mode == "open":
            pv_suffix = data["worker_parameters"]["open_mode_suffix"]
        else:
            pv_suffix = data["worker_parameters"]["closed_mode_suffix"]
        chain_prefix = data["worker_parameters"]["chain_prefix"]
        cavity_prefix = data["worker_parameters"]["cavity_prefix"]
        amplifier_prefix = data["worker_parameters"]["amplifier_prefix"]
        amplifier_suffix = data["worker_parameters"]["amplifier_suffix"]
        self._build_pv_dict(pv_suffix, chain_prefix, cavity_prefix, amplifier_prefix, amplifier_suffix)
        self._user_control_pv = None
        self._control_state = {}
        self._control_state["status"] = "unknown"
        self._control_state["command"] = "unknown"
        connection_error = self._connect()
        if not connection_error:
            log.info("PhaseAlignmentWorker entirely connected")
        self._user_intentions = UF.getUserIntention(self.data["workflow_metadata"]["name"])

    def monitor_actions(self):
        """Monitor state transitions and dispatch asynchronous execution threads.

        Evaluates current worker status against operator control commands (RUN, PAUSE, ABORT)
        received from EPICS or user intentions. Launches :meth:`execute_full_alignment` in a
        daemon thread when moving to the ``running`` state.
        """
        if self._control_state["status"] == "initialized" and self._control_state["command"] == "RUN":
            self._control_state["status"] = "running"
            threading.Thread(target=self.execute_full_alignment, daemon=True).start()
        elif self._control_state["status"] == "running" and self._control_state["command"] == "PAUSE":
            self._control_state["status"] = "paused"
            self._user_intentions.setCommand("PAUSE")
        elif self._control_state["status"] == "paused" and self._control_state["command"] == "RUN":
            self._control_state["status"] = "running"
            self._user_intentions.setCommand("IDLE")
        elif self._control_state["status"] == "error" and self._control_state["command"] == "PAUSE":
            self._control_state["status"] = "initialized"
            self._user_intentions.setCommand("IDLE")
        elif self._control_state["status"] == "completed" and self._control_state["command"] == "PAUSE":
            self._control_state["status"] = "initialized"
        elif self._control_state["status"] == "running" and self._control_state["command"] == "ABORT":
            log.error("User sent an Abort signal")
            self._control_state["status"] = "error"
            self._user_intentions.setCommand("ABORT")

    def _on_user_control_change(self, pvname=None, value=None, **kwargs):
        """Asynchronous EPICS Channel Access callback triggered on PV value update.

        Maps raw integer PV values to human-readable control commands:
            * ``1`` -> ``RUN``
            * ``0`` -> ``PAUSE``
            * ``2`` -> ``ABORT``

        :param pvname: Name of the updated EPICS Process Variable, defaults to None.
        :type pvname: str, optional
        :param value: Updated value received from Channel Access, defaults to None.
        :type value: int or float, optional
        :param kwargs: Additional keyword arguments passed by ``epics.PV`` callbacks.
        """
        if self._control_state["status"] == "unknown":
            self._control_state["status"] = "initialized"
            return

        val_int = int(value)
        cmd_map = {1: "RUN", 0: "PAUSE", 2: "ABORT"}
        command = cmd_map.get(val_int, "UNKNOWN")

        self._control_state["raw_val"] = val_int
        self._control_state["command"] = command

        log.info(f"PV '{pvname}' changed to {val_int} -> Command: {command}")

    def _connect(self):
        """Establish EPICS Channel Access PV instances and assign update callbacks.

        Connects user control PVs, master chain amplifier/target PVs, and all slave chain PVs.

        :return: True if any PV connection timed out or failed; False if all connected.
        :rtype: bool
        """
        connection_error = False
        #Connect user control PV
        user_control_pv_name = self.data["user_control"]["pv_name"]
        user_control_pv_instance = epics.PV(user_control_pv_name)
        if user_control_pv_instance.wait_for_connection(timeout=2):
            self._user_control_pv = user_control_pv_instance
            self._user_control_pv.add_callback(self._on_user_control_change)
        else:
            log.error(f"Couldn't connect to {user_control_pv_name}")
            connection_error = True
        #Connect Master
        amplifier_pv_name = self._epics_data["master_chain"]["amplifier_pv"]["name"]
        amplifier_pv_instance = epics.PV(amplifier_pv_name)
        if amplifier_pv_instance.wait_for_connection(timeout=2):
            self._epics_data["master_chain"]["amplifier_pv"]["instance"] = amplifier_pv_instance
        else:
            log.error(f"Couldn't connect {amplifier_pv_name}")
            connection_error = True
        target_pv_name = self._epics_data["master_chain"]["target_pv"]["name"]
        target_pv_instance = epics.PV(target_pv_name)
        if target_pv_instance.wait_for_connection(timeout=2):
            self._epics_data["master_chain"]["target_pv"]["instance"] = target_pv_instance
        else:
            log.error(f"Couldn't connect to {target_pv_name}")
            connection_error = True
        #Connect sleeve
        for amp_dict in self._epics_data["slave_chains"]["amplifier_pvs"]:
            pv_name = amp_dict["name"]
            pv_instance = epics.PV(pv_name)
            if pv_instance.wait_for_connection(timeout=2):
                amp_dict["instance"] = pv_instance
            else:
                log.error(f"Couldn't connect to Slave Amplifier PV: {pv_name}")
                connection_error = True
        for target_dict in self._epics_data["slave_chains"]["target_pvs"]:
            pv_name = target_dict["name"]
            pv_instance = epics.PV(pv_name)
            if pv_instance.wait_for_connection(timeout=2):
                target_dict["instance"] = pv_instance
            else:
                log.error(f"Couldn't connect to Slave Target PV: {pv_name}")
                connection_error = True
        return connection_error

    def _build_pv_dict(self,pv_suffix, chain_prefix, cavity_prefix, amplifier_prefix, amplifier_suffix):
        """Construct EPICS PV name mapping structure based on hardware topology settings.

        :param pv_suffix: PV suffix corresponding to open or closed loop operation.
        :type pv_suffix: str
        :param chain_prefix: Prefix string identifying the RF chain address.
        :type chain_prefix: str
        :param cavity_prefix: Prefix string identifying cavity addresses.
        :type cavity_prefix: str
        :param amplifier_prefix: Prefix string for amplifier control PVs.
        :type amplifier_prefix: str
        :param amplifier_suffix: Suffix string for amplifier control PVs.
        :type amplifier_suffix: str
        """
        self._epics_data = {}
        master_chain_data = self.data["hardware_topology"]["master_chain"]
        self._epics_data["master_chain"] = {}
        amplifier_pv_name = amplifier_prefix + str(master_chain_data["id"]) + amplifier_suffix
        pv_target_name = chain_prefix + master_chain_data["chain_addr"][0] + cavity_prefix + master_chain_data["chain_addr"][1] + pv_suffix
        self._epics_data["master_chain"]["amplifier_pv"] = {}
        self._epics_data["master_chain"]["amplifier_pv"]["name"] = amplifier_pv_name
        self._epics_data["master_chain"]["amplifier_pv"]["instance"] = None
        self._epics_data["master_chain"]["target_pv"] = {}
        self._epics_data["master_chain"]["target_pv"]["name"] = pv_target_name
        self._epics_data["master_chain"]["target_pv"]["instance"] = None
        # sleeve chains
        self._epics_data["slave_chains"] = {}
        self._epics_data["slave_chains"]["amplifier_pvs"] = []
        self._epics_data["slave_chains"]["target_pvs"] = []
        slave_chains_data = self.data["hardware_topology"]["slave_chains"]
        for slave_dict in slave_chains_data:
            amplifier_dict = {}
            target_dict = {}
            amplifier_pv_name = amplifier_prefix + str(slave_dict["id"]) + amplifier_suffix
            pv_target_name = chain_prefix + slave_dict["chain_addr"][0] + cavity_prefix + slave_dict["chain_addr"][1] + pv_suffix
            amplifier_dict["name"] = amplifier_pv_name
            amplifier_dict["instance"] = None
            target_dict["name"] = pv_target_name
            target_dict["instance"] = None
            self._epics_data["slave_chains"]["amplifier_pvs"].append(amplifier_dict)
            self._epics_data["slave_chains"]["target_pvs"].append(target_dict)

    def _set_amplifier_state(self, amp_pv_dict, enable=True):
        """Toggle an amplifier state using its connected ``epics.PV`` instance.

        :param amp_pv_dict: Dictionary holding the amplifier PV name and PV object instance.
        :type amp_pv_dict: dict
        :param enable: True to enable (set to 1), False to disable (set to 0), defaults to True.
        :type enable: bool, optional
        """
        pv_name = amp_pv_dict["name"]
        pv_inst = amp_pv_dict["instance"]
        val = 1 if enable else 0
        log.info(f"setting {pv_name} to {val}")
        pv_inst.put(val, wait=True)
        time.sleep(0.5)

    def isolate_slave_chain(self, target_slave_idx):
        """Isolate RF fields for a specific slave chain.

        Enforces Master AMP1 = ON and Target Slave AMP = ON, while turning off all
        other slave amplifiers. Pauses for 5 seconds post-switching to allow fields to settle.

        :param target_slave_idx: Index of the slave chain to isolate and keep active.
        :type target_slave_idx: int
        """
        log.info(f"Isolating system state for Slave Chain (AMP{target_slave_idx + 2})")

        # 1. Master AMP1 must remain ON
        self._set_amplifier_state(self._epics_data["master_chain"]["amplifier_pv"], enable=True)

        # 2. Toggle slave amplifiers
        for idx, slave_amp in enumerate(self._epics_data["slave_chains"]["amplifier_pvs"]):
            should_enable = (idx == target_slave_idx)
            self._set_amplifier_state(slave_amp, enable=should_enable)

        log.info("Chain isolation complete. Settling RF fields for 5 seconds...")
        time.sleep(5.0)

    def _restore_all_amplifiers(self):
        """Re-enable all master and slave amplifier chains.

        Called upon sequence completion or procedure abort to restore full operating state.
        """
        log.info("Restoring all amplifier chains to ON state...")
        self._set_amplifier_state(self._epics_data["master_chain"]["amplifier_pv"], enable=True)
        for amp in self._epics_data["slave_chains"]["amplifier_pvs"]:
            self._set_amplifier_state(amp, enable=True)

    def execute_full_alignment(self):           
        """Execute complete multi-chain RF phase alignment scan sequence.

        Iterates over all configured slave chains, isolates each target chain, acquires
        phase scan voltage samples using the oscilloscope interface, and performs
        Levenberg-Marquardt least-squares curve fitting.

        Evaluates safety constraints prior to setting new phase angles:
            * **Forward Wave Unbalance Limit:** Must not exceed ``max_allowed_unbalance_percent``.
            * **Minimum Cavity Voltage:** Peak voltage must equal or exceed ``min_expected_cavity_voltage``.

        If an error or safety violation occurs, the current slave chain is restored to its
        original initial phase setpoint and all amplifiers are re-enabled.

        :raises ValueError: If forward wave unbalance exceeds safety limit or cavity peak voltage drops below minimum.
        """
        log.info("Executing full alignment")
        n_points = self.data["worker_parameters"]["n_point"]
        max_unbalance = self.data["safety_limits"]["max_allowed_unbalance_percent"]
        min_voltage = self.data["safety_limits"]["min_expected_cavity_voltage"]
        try:
            scope_ip = self.data["worker_parameters"]["scope_ip"]
            scope = scopelib.CreateScope(scope_ip)
            RFQCU.prepareScopeChannel(scope)
        except Exception as scope_err:
            log.error(f"Failed to initialize oscilloscope at {scope_ip}: {scope_err}")
            self._control_state["status"] = "error"
            return
        slave_targets = self._epics_data["slave_chains"]["target_pvs"]
        for idx, slave_target in enumerate(slave_targets):
            if self._control_state["status"] == "error":
                log.error("Phase Alignment Procedure is in ERROR state, terminating now")
                return 
            checkVal = self._user_intentions.checkCommand()
            if checkVal == 1:
                return
            slave_id = idx + 2
            self._control_state["active_slave_index"] = slave_id
            target_pv_inst = slave_target["instance"]
            pv_name = slave_target["name"]
            log.info(f"Beginning Alignment Sequence for Slave Chain {slave_id} ({pv_name})")
            self.isolate_slave_chain(idx)
            original_phase_deg = target_pv_inst.get()
            log.info(f"Stored initial fallback phase for Slave {slave_id}: {original_phase_deg:.2f}°")
            try:
                scan_points = RFQCU.doScan(n_points, pv_name, target_pv_inst, scope, self._user_intentions, quiet=False)
                if self._control_state["status"] == "error":
                    return
                angles_rad = scan_points[:, 0]
                voltages = scan_points[:, 1]
                log.info("Levenberg-Marquardt Least-Squares Optimization")
                gpar = RFQCU.firstGuess(angles_rad, voltages)
                fpar = RFQCU.doFit(gpar, angles_rad, voltages)
                fpar[2] %= (2 * math.pi)
                log.info("Telemetry figures of merit")
                forward_unbalance = abs((fpar[0] - fpar[1]) / (fpar[0] + fpar[1])) * 100.0
                opt_phase_deg = fpar[2] * 180.0 / math.pi
                peak_voltage = RFQCU.F(fpar[2], *fpar)
                log.info("Safety Validation Checks")
                if forward_unbalance > max_unbalance and not scope.mock:
                    log.error(f"Forward wave unbalance too high: {forward_unbalance:.2f}% (Limit: {max_unbalance}%)")
                    self._control_state["status"] = "error"
                    raise ValueError(f"Forward wave unbalance too high: {forward_unbalance:.2f}% (Limit: {max_unbalance}%)")
                if peak_voltage < min_voltage and not scope.mock:
                    log.error(f"Cavity field too weak: {peak_voltage:.2e} V (Limit: {min_voltage} V)")
                    self._control_state["status"] = "error"
                    raise ValueError(f"Cavity field too weak: {peak_voltage:.2e} V (Limit: {min_voltage} V)")
                # Lock optimized setpoint
                RFQCU.setPhase(fpar[2], target_pv_inst, pv_name)
                log.info(f"Lock applied for Slave {slave_id} at {opt_phase_deg:.2f}°")

            except Exception as run_error:
                log.error(f"Error during alignment on Slave {slave_id}: {run_error}")
                target_pv_inst.put(original_phase_deg, wait=True)
                self._control_state["status"] = "error"
                break
        
        self._restore_all_amplifiers()
        if self._control_state["status"] == "running":
            log.info("Phase Alignment Procedure Completed")
            self._control_state["status"] = "completed"
