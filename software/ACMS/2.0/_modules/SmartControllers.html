def discover_and_run_modules():
    """
    Scans the current directory for .py files, imports them as modules,
    and calls a predefined target function if it exists.
    """
    # Get the directory of the current __init__.py file
    current_dir = pathlib.Path(__file__).parent.resolve()
    current_dir_string = str(current_dir)
    current_dir = current_dir_string.replace("SmartControllers", "workflows")
    
    # Add the current directory to the system path to allow imports
    sys.path.insert(0, str(current_dir))

    # Iterate through all files in the current directory
    for file_name in os.listdir(current_dir):
        # We only care about Python files and exclude __init__.py itself
        if file_name.endswith('.py') and file_name != '__init__.py':
            module_name = file_name[:-3] # Remove the .py extension
            
            try:
                # Dynamically import the module
                module = importlib.import_module(module_name)

                # Check if the target function exists in the module
                if hasattr(module, TARGET_FUNCTION_NAME) and callable(getattr(module, TARGET_FUNCTION_NAME)):
                    # Call the function
                    getattr(module, TARGET_FUNCTION_NAME)()
                    
                else:
                    print(f"  -> Warning: Target function '{TARGET_FUNCTION_NAME}' not found or is not callable.")
                    
            except Exception as e:
                print(f"  -> Error importing or running module '{module_name}': {e}")
                
    # Clean up the system path
    sys.path.pop(0)



# This is a common pattern to ensure the discovery logic runs when the package is imported
discover_and_run_modules()
