"""projectModel module 

Created on Wed Apr  5 15:04:38 2023 \n
This module (more in general this package) allows to externalize data modelling with respect to the real Physical Component's 
representation. \n
At the time of writing this code (and documentation) there are no clear requirements about the data modelling and how the data must be 
treated and used in the future; thus the programmer took advantage for foreseeing a whole "spectra" of possible models, in case to 
be written in this package. \n
The only clear requirements we have (for now) are: \n
1- It must be possible to "group" different instances of controllers (regardless of their type) based on the Plant System functionalities. \n
2- It must be possible to "arrange" the software objects in a hierarchycal structure. \n
Given these two constraints the Projectmodel class have been implemented as follows.

@author: luciamarock
"""

class NodeObject(NodeMixin):  # Add Node feature
    """Inherits from NodeMixin (anytree) python library
    
    This external library allows to create node objects to be connected in hierachies \n
    See anytree documentation `here <https://anytree.readthedocs.io/en/latest/>`_
    
    """
    def __init__(self, name, parent=None, children=None):
        super(NodeObject, self).__init__() # children, descendants, leaves
        self.name = name
        self.parent = parent
        if children:
            self.children = children
        self.processed = False
        self.group = None

    def __repr__(self):
        return "<NodeObject %s at 0x%x>" % (repr(self.name), id(self))

class projectmodel:
    """The project model is the way the information is organized
    
    Currently the Smart Controller's can be grouped by "Plant System Functionlity" and can be (if needed) 
    organized in a hierachycal structure, based on parent-child relationship. \n
    One important pre-requisite is that the Controller Instances's names are unique. \n 
    In the current implementation of the ProjecModel the dictionary called _registered_groups contains the 
    information about how the Controllers must be divided into Functional Groups while the dictionary _registered_nodes
    contains all the "nodes" objects with connection capabilities in order to be arranged in a hierarchycal structure. \n 
    By design the node objects are different from the Instances objects but they duplicates the latters; this choice of
    keeping separated the two software entities is mainly due to the lack of visibility about how this software will be used 
    in the near future, having the software entities separated allow a separate development and assures (at the same time) to always 
    recall a Controller Instance by its name based on the Model needs. 
    
    """
    def __init__(self,settings):
        self._settings = settings
        self._name = settings["project_name"]
        log.info("created project with name: {}".format(self._name))
        self._model = NodeObject("model")
        self._registered_nodes = {}
        self._registered_groups = {}

    def _add_to_parent(self,current_instance,parent_node):
        """adds and instance node to its parent
        
        If the node corresponding to the current_instance does not exist, the method creates it. \n
        In any case the method attaches the Instance's node to the specified parent. 
        
        :param current_instance: The controller's Instance (software object)
        :param parent_node: the name of the parent node (equal to the name of the parent controller)
        
        """
        if current_instance.instance in self._registered_nodes.keys():
            self._registered_nodes[current_instance.instance].parent = self._registered_nodes[parent_node]
            self._registered_nodes[current_instance.instance].processed = True
            self._registered_nodes[current_instance.instance].group = current_instance.group
        else:
            new_node = NodeObject(current_instance.instance,parent = self._registered_nodes[parent_node])
            new_node.processed = True
            new_node.group = current_instance.group
            self._registered_nodes[new_node.name] = new_node

    def add_node(self,current_instance):
        """This method is responsible for assembling the attributes dictionaries _registered_nodes and _registered_groups
        
        it first registers the group to which each controller instance belongs to, below each group it adds the instance name 
        (just the name, it is unique); then it tries to retrieve the "parent" controller object, if the parent is not found or it is
        unspecified, the method registers the node and attaches it to the root node (_model).
        If the current instance does not exist as registered node the method creates a new node object with the same name
        of the current controller's instance. \n
        If the node already exist (for example because it was previously created in quality of parent of another node) the method simply 
        assignes to it the proper parent (if different from _model) and marks it as processed (attribute processed = True). \n 
        The processed attribute will be used for checking possible anomalies. \n 
        The so implemented logic is able to build a complex hierarchycal tree regardless of the order of declaration of the nodes. 
        
        :param current_instance: the current Controller's Instance (as returned by instances_factories)
        
        """
        if current_instance.group:
            if current_instance.group not in self._registered_groups.keys():
                self._registered_groups[current_instance.group] = []
                self._registered_groups[current_instance.group].append(current_instance.instance)
            else:
                self._registered_groups[current_instance.group].append(current_instance.instance)

        try:
            parent_node = current_instance.parent
        except:
            parent_node = ""
        if not parent_node:
            if current_instance.instance in self._registered_nodes.keys():
                self._registered_nodes[current_instance.instance].parent = self._model
                self._registered_nodes[current_instance.instance].processed = True
                self._registered_nodes[current_instance.instance].group = current_instance.group
            else:
                new_node = NodeObject(current_instance.instance,parent = self._model)
                new_node.processed = True
                new_node.group = current_instance.group
                self._registered_nodes[new_node.name] = new_node 
        else:
            if parent_node not in self._registered_nodes.keys():
                parent_temp = NodeObject(parent_node,parent = self._model)
                self._registered_nodes[parent_temp.name] = parent_temp 
            self._add_to_parent(current_instance,parent_node)

    def export(self):
        """Export method
        
        This method exports the tree model both in the form of textual file and image. 
        
        """
        filename = self._settings["workspace_dir"] + "projectmodel"
        
        with open(filename + ".txt","w") as modelfile:
            for pre, fill, node in RenderTree(self._model):
                try:
                    modelfile.write("%s%s\n" % (pre,node.name))
                except:
                    modelfile.write("%s%s\n" % (pre.encode('utf-8'),node.name))
            modelfile.close()
