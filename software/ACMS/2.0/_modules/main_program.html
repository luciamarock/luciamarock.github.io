# -*- coding: utf-8 -*-
"""Main Program Entry Point.

Created on Wed Nov  2 15:14:01 2022

Serves as the primary entry point for the Recipe Management System (RMS).
Initializes system configurations, sets up logging handlers, populates global
runtime settings, and boots system controller interfaces and workflows.

Usage:
    $ python main_program.py path/to/settings.ini

.. note::
   This document adheres to Sphinx/reStructuredText docstring format standards.
   For user-facing guides, refer to the PDF manual in the documentation folder.

:author: luciamarock
"""

import sys 
import os
import configparser
import logging
from notify import info, warning, error
config = configparser.ConfigParser()

def _init():
    """Initialize system-wide configurations and setup application environment.

    Reads and validates the ``settings.ini`` file passed as a command-line argument.
    Initializes project directories, configures workspace paths, establishes file/EPICS
    logging streams, and populates the global runtime dictionary 
    :data:`utilities.util_functions.settings`.

    :raises SystemExit:
        If command-line arguments are missing, the target ini file does not exist,
        or required configuration sections (``project``, ``workspace``, ``logging``,
        ``resources``, ``validation``) are malformed or missing.

    .. seealso::
       * :mod:`controllersInterface` - Invoked following initialization to parse XML configurations.
       * :mod:`utilities.util_functions` - Holds the global ``settings`` dictionary updated here.
    """
    if len(sys.argv) < 2:
        error("no given arguments, expected path to settings.ini")
        sys.exit()
    
    main_position = os.path.abspath(__file__)
    console_position = os.getcwd()
    settings_position = sys.argv[1]
    if os.path.exists(settings_position):
        config.read(settings_position)
        if "project" in config.sections():
            try:
                project_name = config["project"]["name"] # section object can be navigated as a dictionary
            except:
                error("wrong format for project.name")
                sys.exit()
            try: 
                UI = config["project"]["UI"]
            except:
                error("wrong format for project.UI")
                sys.exit()
        else:
            error("no given project name")
            sys.exit()
        if "workspace" in config.sections():
            try:
                ws_dir = config["workspace"]["ws_dir"]
                if ws_dir.endswith(os.path.sep):
                    pass
                else:
                    ws_dir = ws_dir + os.path.sep
                if not os.path.exists(ws_dir):
                    os.makedirs(ws_dir)
            except:
                error("wrong format for workspace settings")
                sys.exit()
        else:
            error("no workspace defined")
            sys.exit()
        if "logging" in config.sections():
            try:
                logfile = ws_dir + "logging" + os.path.sep + project_name + ".log" #config["logging"]["filename"]
                if not os.path.exists(os.path.dirname(logfile)): # os.path.dirname extracts the directory only 
                    os.makedirs(os.path.dirname(logfile))
                loglevel = eval("logging." + config["logging"]["log_level"])
                logging.basicConfig(format='%(asctime)s %(levelname)s %(name)s: %(message)s',
                                    filename=logfile,filemode='w',level=loglevel,datefmt='%Y-%m-%d %H:%M:%S')

                
                for name, value in config["logging"].items():
                    if name == "terminal_handler" and value == "True":
                        root_logger = logging.getLogger() 
                        console_handler = logging.StreamHandler(sys.stderr)
                        formatter = logging.Formatter('%(asctime)s %(name)s %(levelname)s: %(message)s', datefmt='%Y-%m-%d %H:%M:%S')
                        console_handler.setFormatter(formatter)
                        console_handler.setLevel(loglevel)
                        root_logger.addHandler(console_handler)
                        print("Added terminal log handler")
                    elif name == "terminal_handler" and value == "False":
                        root_logger = logging.getLogger()
                        import utilities.tool_internals as internal_functions
                        epics_handler = internal_functions.get_epics_log_handler()
                        epics_handler.setLevel(loglevel)
                        root_logger.addHandler(epics_handler)
                        print("Added EPICS log handlers")
            except:
                error("wrong format for logging configuration")
                sys.exit()
        else:
            error("no logging configurations found")
            sys.exit()
        if "resources" in config.sections():
            try:
                resources_dir = config["resources"]["resources_dir"]                
            except:
                error("wrong format for resources directory")
                sys.exit()
            if not os.path.exists(resources_dir):
                error("resource directory specified in the init file does not exist")
                sys.exit()
            if resources_dir.endswith(os.path.sep):
                pass
            else:
                resources_dir = resources_dir + os.path.sep
            cConfFiles = resources_dir + "types"
            cInstFiles = resources_dir + "instances"
        else:
            error("no resources folder location provided")
            sys.exit()
        if "validation" in config.sections():
            try:
                validators = config["validation"]["validators"].split(",")
                for i in range(len(validators)):
                    validators[i] = validators[i].strip()
            except Exception as e:
                error("key {} not found in the validation section".format(e))
                sys.exit()
        else:
            error("vlidators not specified in the settigs.ini file")
            sys.exit()
    else:
        error("settings not found")
        sys.exit()
    import utilities.util_functions as UF
    UF.settings["main_position"] = main_position
    UF.settings["console_position"] = console_position
    UF.settings["settings_position"] = settings_position
    UF.settings["workspace_dir"] = ws_dir
    UF.settings["logfile"] = logfile
    UF.settings["loglevel"] = config["logging"]["log_level"]
    UF.settings["controller_types_location"] = cConfFiles
    UF.settings["controller_instances_location"] = cInstFiles
    UF.settings["resources_directory"] = resources_dir
    UF.settings["project_name"] = project_name
    UF.settings["UI"] = UI
    UF.settings["PID"] = os.getpid()
    UF.settings["pyver"] = sys.version_info
    UF.settings["validators"] = validators
    try:
        current_user = os.getlogin()
    except OSError:
        current_user = None
    UF.settings['current_user'] = current_user



    
if __name__ == "__main__":
    _init()
    import controllersInterface as CI
    CI.parse_xml_types()
    CI.parse_xml_instances()
    import workflows_interface
    if workflows_interface.init():
        workflows_interface.start()
