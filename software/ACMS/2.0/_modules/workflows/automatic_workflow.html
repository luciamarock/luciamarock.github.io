# -*- coding: utf-8 -*-
"""Automatic Recipe Execution Workflow Module (RMS Auto).

This module implements the foundational automatic recipe execution workflow (``RMSauto``) 
for the Recipe Management System (RMS) within the ACMS suite.

It acts as a top-level orchestrator that:
    1. Ingests global project settings and auto-mode mapping payloads.
    2. Broadcasts project properties and mapping schedules to all registered smart controller 
       types and instances via :mod:`controllersInterface`.
    3. Spawns and starts underlying controller threads.
    4. Executes a continuous polling and monitoring loop (``work()``) to evaluate controller 
       actions and operational states.
    5. Exposes diagnostic health logging and graceful shutdown routines.

Module Dependencies:
    - :mod:`workflows_interface`: Interface repository for workflow registration.
    - :mod:`controllersInterface`: Central registry for smart controller types and instances.

Created on Fri Feb  2 09:10:28 2024

@author: luciamarock
"""

import workflows_interface as WFI
import controllersInterface as CI
import time
import logging 
log = logging.getLogger(__name__)

class RMSauto:
    """Recipe Management System Automatic Execution Workflow Engine.

    This class encapsulates the continuous execution loop responsible for orchestrating 
    smart controllers during automatic recipe execution.

    Args:
        argument (int): Execution control flag. A value of ``0`` keeps the execution 
            loop active, while setting it to non-zero (e.g., ``1`` via :meth:`stop`) 
            triggers a graceful exit.

    Attributes:
        argument (int): Internal loop control state flag.
        _auto_mode_mapping_data (dict or list): Configuration data payload mapping device 
            instances to auto-mode execution schedules.
        _settings (dict): Global project configuration parameters and hardware settings.
    """
    def __init__(self, argument):
        """Initialize the RMSauto workflow instance with an initial control state flag."""
        self.argument = argument

    def general_settings(self,settings):
        """Ingest global project configuration parameters.

        Args:
            settings (dict): Project-wide properties, EPICS PV prefixes, and runtime settings.
        """
        self._settings = settings 
        log.info("Automatic Workflow received the general settings")



    def __call__(self):
        """Callable entry point to initialize controller instances and launch the execution loop.

        Iterates over all registered controller types and instances in :mod:`controllersInterface`:
            1. Propagates project properties to controller types.
            2. Assigns auto-mode mappings to specific device instances.
            3. Spawns/starts controller threads.
            4. Enters the continuous execution loop via :meth:`work`.
        """
        log.debug("called Recipe Manager auto at {}".format(self))
        for key in CI.registered_types:
            CI.registered_types[key].set_project_properties(self._settings)
            for instance in CI.registered_types[key].get_registered_instances_keys():
                CI.registered_instances[instance].set_auto_mode_mapping(self._auto_mode_mapping_data)
                CI.registered_instances[instance].start()
        self.work()

    def wait_please(self):
        """Provide execution pacing between monitoring cycles within the main loop.

        Blocks thread execution for ~0.99 seconds to prevent CPU starvation during continuous 
        polling operations.
        """
        #print("waiting ...") #TODO maybe here write PLC PVs for informing about the instances status 
        time.sleep(0.99)

    def work(self):
        """Main continuous execution loop for active workflow monitoring.

        Loops continuously while :attr:`argument` remains ``0``. On each cycle:
            1. Invokes :meth:`wait_please` for loop pacing.
            2. Triggers :meth:`monitor_actions` across all registered smart controller instances 
               to evaluate state transitions and entry/exit criteria.
        """
        while self.argument == 0:
            self.wait_please()
            for key in CI.registered_types:
                for instance in CI.registered_types[key].get_registered_instances_keys():
                    CI.registered_instances[instance].monitor_actions()

    def stop(self):
        """Gracefully terminate the workflow execution loop.

        Sets :attr:`argument` to ``1``, causing the :meth:`work` while-loop to break 
        on its next iteration.
        """
        self.argument = 1
