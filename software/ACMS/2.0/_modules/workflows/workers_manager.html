"""Auto-Conditioning Worker Manager Workflow Module.

This module implements the ``WorkerManager`` workflow (registered as 
``"AutoConditioningWorkflow"`` in :mod:`workflows_interface`), which manages 
the lifecycle, EPICS PV integration, HMI communications, and execution threads 
for automated conditioning workers.

Key Responsibilities:
    1. Worker Parsing & Factory Instantiation: Parses XML configuration files from 
       the resource directory via :class:`workers.factories.factory_interface.WorkerFactory`.
    2. EPICS Connection & State Monitoring: Binds EPICS PV control interfaces to individual 
       worker instances, handling enable/abort/reset/stop actions dynamically via callbacks.
    3. Multithreaded Execution: Spawns dedicated daemon threads for parallel worker execution.
    4. HMI Integration: Controls macro propagation and exports worker availability and status 
       waveforms to EPICS for user display.
    5. Configuration Persistency: Synchronizes workspace state on connection setup.

Module Dependencies:
    - :mod:`workflows_interface`: Central registration interface for system workflows.
    - :mod:`utilities.util_functions`: System-wide settings and utility functions.
    - :mod:`workers.factories.factory_interface`: Factory interface creating worker instances from XML.
"""

import threading
import time
import epics
import os
import workflows_interface as WFI
import utilities.util_functions as UF
import workers.factories.factory_interface as FI
import logging
log = logging.getLogger(__name__)

class WorkerManager:
    """Orchestrator for Auto-Conditioning worker threads and EPICS control interfaces.

    This workflow manages worker lifecycle events (start, stop, abort, reset), handles 
    EPICS PV callbacks, and communicates status updates to HMI displays.

    Attributes:
        workers (dict[str, object]): Map of active worker instances keyed by worker name.
        epics_interfaces (dict[str, str]): Map linking EPICS interface PV names to worker names.
        managerPVs (dict[str, epics.PV]): Map of connected EPICS PV instances managing worker state.
        _jobs (dict[str, str]): Queue for pending HMI macro updates or state jobs.
        _settings (dict): Project settings retrieved from :mod:`utilities.util_functions`.
        _factoryInterface (FI.WorkerFactory): Worker factory instance for XML parsing.
        _workers_dir (str): Absolute file system path to the XML workers configuration folder.
        ACT_presence (epics.PV): PV signaling the active presence of the Auto Conditioning Tool.
        hmi_macro (epics.PV): PV broadcasting selected macro parameters to HMI.
        table_index (epics.PV): PV monitoring table index selections from HMI.
        workers_list_pv (epics.PV): EPICS Waveform PV broadcasting active worker names and states.
        running (bool): Main loop execution state flag.
    """
    def __init__(self):
        """Initialize the WorkerManager instance, configuration paths, and factory interfaces."""
        log.info("WorkerManager initialized")
        self.workers = {}  # Dictionary of active workers {worker_name: worker_instance}
        self.epics_interfaces = {} # Dictionary of PV interfaces for the workers {PV_name: worker_name}
        self.managerPVs = {} # a place where to store this worker manager PVs {worker_name: PV_instance}
        self._jobs = {}
        self._settings = UF.settings
        self._factoryInterface = FI.WorkerFactory(self._settings)
        self._workers_dir = self._settings["resources_directory"] + "workers"

    def _parse_xml_workers(self):
        """Parse worker XML definitions from the resources directory and setup core EPICS PVs.

        Scans ``self._workers_dir`` for `.xml` files and invokes the factory to instantiate 
        workers, EPICS interfaces, and macros. Initializes system presence and HMI selection PVs.

        Returns:
            bool: ``True`` if an error occurred or no XML files were found; ``False`` on success.
        """
        log.info("Retrieving Workers from Default \"resources\" directory")
        if not os.path.exists(self._workers_dir):
            log.error(f"Workers directory '{self._workers_dir}' does not exist. It must contain the workers configuration files")
            try:
                os.makedirs(self._workers_dir)
            except Exception as e:
                log.error(f"Failed to create workers directory '{self._workers_dir}': {e}")
        xml_files = [f for f in os.listdir(self._workers_dir) if f.endswith('.xml')]
        if len(xml_files) == 0:
            log.error("no xml configuration files found for workers, exiting now")
            return True
        self.workers, self.epics_interfaces, self.macros = self._factoryInterface.create_worker(xml_files) 
        if not self.workers or not self.epics_interfaces:
            return True
        self.ACT_presence = epics.PV("LLRF9:TransmCPLs")
        self.ACT_presence.wait_for_connection(timeout=1)
        self.ACT_presence.put(1)
        self.hmi_macro = epics.PV("HMI:worker_name")
        self.hmi_macro.wait_for_connection(timeout=1)
        self.table_index = epics.PV("TABLE:SELECTION")
        self.table_index.wait_for_connection(timeout=1)
        self.table_index.add_callback(self._update_macro)
        active_workers_names = list(self.workers.keys())
        self.workers_list_pv = epics.PV("HMI:ComboBox")  # EPICS waveform PV
        self.workers_list_pv.wait_for_connection(timeout=1)
        self.workers_list_pv.put(active_workers_names)
        return False

    def _update_macro(self,pvname=None, value=None, char_value=None, **kw):
        """EPICS callback triggered when the HMI table selection index changes.

        Args:
            pvname (str, optional): Name of the EPICS PV.
            value (int or float, optional): New index value selected in the HMI table.
            char_value (str, optional): String representation of the value.
            **kw: Additional keyword arguments from PyEpics callback engine.
        """
        try:
            items_list = list(self.macros.items())
            item = items_list[int(value)]
            self._jobs["hmi_macro"] = item[1]
            log.info(f"HMI: selected {item[0]} setting macro to {item[1]}")
        except Exception as e:
            log.error(e)


    
    def __call__(self):
        """Main execution loop processing queued manager jobs (e.g., HMI macro updates)."""
        self.running = True 
        while self.running:
            if self._jobs:
                for key, value in self._jobs.items():
                    if key == "hmi_macro":
                        self.hmi_macro.put(value)
            self._jobs = {}
            time.sleep(0.5)

    def connect(self):
        """Connect the manager and instantiated workers to the EPICS layer and synchronize workspace state.

        Performs:
            1. XML configuration parsing (:meth:`_parse_xml_workers`).
            2. EPICS PV interface creation and callback setup for individual worker state management.
            3. Worker EPICS connectivity verification.
            4. Workspace initial state synchronization (load/compare/save).

        Returns:
            bool: ``True`` if all EPICS connections and worker synchronizations succeeded; ``False`` otherwise.
        """
        error = self._parse_xml_workers()
        if error:
            return not error
        for interface_pv, worker_name in self.epics_interfaces.items():
            self.managerPVs[worker_name] = epics.PV(interface_pv)
            if self.managerPVs[worker_name].wait_for_connection(timeout=1):
                self.managerPVs[worker_name].add_callback(self._enable_callback)
            else:
                log.error(f"can't connect to {interface_pv} for worker {worker_name}")
                error = True 
                break
        if not error:
            log.info("Workers manager fully connected to the EPICS layer, connecting the workers ...")
            for worker_name, worker_instance in self.workers.items():
                if not worker_instance.connect():
                    error = True 
                    break
        if not error:
            log.info("All the Workers are fully connected to the EPICS layer, saving/loading their Initial Configuration to/from the workspace...")
            #save/load current configuration of each worker 
            for worker_name, worker_instance in self.workers.items():
                workspace_directory = self._settings["workspace_dir"]
                worker_instance.load_compare_save(workspace_directory)
        return not error

    def _enable_callback(self,pvname=None, value=None, char_value=None, **kw):
        """EPICS callback handling state transition requests for specific workers.

        Evaluates incoming control PV values against worker execution states (``running``, 
        ``completed``, ``error``, ``not running``, ``aborted``) to trigger ``abort()``, 
        ``stop()``, ``reset()``, or :meth:`start_worker`.

        Args:
            pvname (str, optional): Triggering EPICS PV name.
            value (int, optional): Control command value (``0`` to stop/abort/reset, ``1`` to start).
            char_value (str, optional): String representation of the PV value.
            **kw: Additional callback keyword arguments.
        """
        # here stop specific worker (worker_id = worker_name)
        worker_instance = self.workers[self.epics_interfaces[pvname]]
        itsStatus = worker_instance.get_status()
        if value == 0 and itsStatus == "running":
            worker_instance.abort()
        elif value == 0 and itsStatus == "completed":
            worker_instance.stop()
        elif value == 0 and itsStatus == "error":
            worker_instance.reset()
        elif value == 1 and itsStatus == "not running":
            self.start_worker(worker_instance)
        elif value == 1 and itsStatus == "aborted":
            worker_instance.reset()
        else:
            log.info(f"{self.epics_interfaces[pvname]} successfully resetted")

    def start_worker(self, worker):
        """Start a target worker in a separate background daemon thread.

        Args:
            worker (object): Worker instance possessing a ``start()`` method.
        """
        worker_thread = threading.Thread(target=worker.start, daemon=True)
        worker_thread.start()

    def stop_worker(self, worker_id):
        """Stop an active worker instance and unregister it from the manager.

        Args:
            worker_id (str): Key identifying the target worker in :attr:`workers`.
        """
        if worker_id not in self.workers:
            log.info(f"Worker {worker_id} is not running.")
            return
        
        self.workers[worker_id].stop()
        del self.workers[worker_id]
        log.info(f"Stopped worker {worker_id}.")
        self.update_active_workers()

    def update_active_workers(self):
        """Format active worker step execution progress and update EPICS waveform PV."""
        status_list = [
            f"{wid}: {w.current_step}/{w.total_steps} | Last Action: {w.last_action}"
            for wid, w in self.workers.items()
        ]

        max_elements = 100
        truncated_list = status_list[:max_elements]

        if not self.workers_list_pv.wait_for_connection():
            log.error("EPICS PV connection failed.")
            return

        self.workers_list_pv.put(truncated_list)
        log.info("Updated EPICS waveform with active worker statuses.")
