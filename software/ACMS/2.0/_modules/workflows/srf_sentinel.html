# -*- coding: utf-8 -*-
"""Superconducting Radio Frequency (SRF) Interlock Sentinel Workflow Module.

This module implements the ``SRFsentinel`` continuous interlock monitoring daemon and the 
associated ``SentinelRule`` rule engine.

Key Functionality:
    1. CSV-Driven Interlock Rules: Loads sentinel rules from ``srf_sentinel.csv`` to monitor 
       EPICS trigger Process Variables (PVs) against condition statements (including logical 
       operators like ``AND`` / ``OR`` and ``NaN`` readback guards).
    2. Interlock Action Dispatching: Automatically writes specified target values to driven 
       PVs whenever interlock trigger conditions evaluate to ``True``.
    3. Waveform Logging & HMI Metrics: Logs interlock activity messages and timestamps to EPICS 
       waveform arrays (capped at 500 entries) and updates the total active interlock count.
    4. Template Editor Subprocess Execution: Monitors requests via EPICS to launch external 
       GUI template editor processes via :mod:`gui_process_launcher` and :mod:`editor_interface`.

Module Dependencies:
    - :mod:`workflows_interface`: Workflow registration repository.
    - :mod:`editor_interface`: Phoebus display (.bob) sanitization and preparation utilities.
    - :mod:`gui_process_launcher`: Subprocess launcher for external GUI editors.

Created on Fri Feb  2 09:10:28 2024

@author: luciamarock
"""

class SentinelRule:
    """Represents a single CSV-defined interlock evaluation rule.

    Args:
        trigger_pv (str): Name of the EPICS PV to monitor as the rule trigger.
        trigger_condition (str): Condition expression string (e.g., ``"<= 45 or >= 100"``, ``"== 1"``).
        driven_pv (str): Name of the target EPICS PV to actuate upon interlock trigger.
        driven_val (float or str): Numerical value to put to ``driven_pv`` when triggered.
        message (str): Diagnostic message describing the interlock event.

    Attributes:
        trigger_pv (str): Monitored trigger PV name.
        trigger_condition (str): Expression condition string evaluated against trigger values.
        driven_pv (str): Target actuation PV name.
        driven_val (float): Floating-point value set on interlock trigger.
        message (str): Human-readable interlock alert message.
        execute (bool): Flag indicating that an interlock actuation needs to be dispatched.
        isInterlocked (bool): Current active interlock status flag for this rule.
    """
    def __init__(self, trigger_pv, trigger_condition, driven_pv, driven_val, message):
        """Initialize a SentinelRule instance with trigger, condition, driven parameters, and message."""
        self.trigger_pv = trigger_pv
        self.trigger_condition = trigger_condition
        self.driven_pv = driven_pv
        self.driven_val = float(driven_val)
        self.message = message
        self.execute = False
        self.isInterlocked = False

    def evaluate(self, trigger_value):
        """Evaluate incoming trigger PV value against rule condition expression.

        Handles string casting, ``NaN`` detection (forcing interlock on invalid numbers), 
        and compound logical conditions (e.g., ``"and"`` / ``"or"`` splits).

        Args:
            trigger_value (float or str): Fresh readback value received from the trigger PV callback.
        """
        try:
            # Readback/trigger may be NaN; normal comparisons will then all be False.
            # In that case we force the interlock action as requested in readme.txt.
            try:
                trigger_val = float(trigger_value.strip()) if isinstance(trigger_value, str) else float(trigger_value)
                if math.isnan(trigger_val):
                    self.execute = True
                    self.isInterlocked = True
                    return
            except Exception:
                # If conversion fails, continue with normal rule evaluation.
                pass

            # Determine which operator is being used to split the condition
            operator = None
            if "and" in self.trigger_condition.lower():
                operator = "and"
            elif "or" in self.trigger_condition.lower():
                operator = "or"

            if operator:
                # Split by the detected operator (case-insensitive)
                # We use a case-insensitive split to be robust against "OR", "Or", "and", etc.
                parts = re.split(f" {operator} ", self.trigger_condition, flags=re.IGNORECASE)
                
                # Inject the trigger_value into each side of the expression
                # Example: "trigger_value <= 45 or trigger_value >= 100"
                eval_string = f" {operator} ".join([f"{trigger_value} {p.strip()}" for p in parts])
            else:
                # Standard simple condition: "trigger_value == 60"
                eval_string = f"{trigger_value} {self.trigger_condition.strip()}"

            # Evaluate the constructed Python expression
            if eval(eval_string):
                #print(f"Condition Match: {eval_string} is True")
                #print(f"Action Message: {self.message}")
                self.execute = True
                self.isInterlocked = True
            else:
                self.execute = False
                self.isInterlocked = False

        except Exception as e:
            log.error(f"Error evaluating condition '{self.trigger_condition}' with value '{trigger_value}': {e}")
            self.execute = False

class SRFsentinel:
    """Superconducting RF Sentinel interlock monitoring and execution engine.

    Args:
        argument (int): Control state flag. ``0`` keeps the sentinel loop active; non-zero 
            triggers graceful loop exit.

    Attributes:
        argument (int): Execution loop control state.
        trigger_pvs (dict[str, epics.PV]): Map of connected trigger EPICS PV instances.
        driven_pvs (dict[str, epics.PV]): Map of connected target/driven EPICS PV instances.
        _template_editor_request (str): Current pending template editor request payload key.
        _rules (list[SentinelRule]): List of parsed sentinel interlock rules.
        _hmi_table (epics.PV): Waveform PV storing recent interlock alert messages.
        _hmi_timestamps (epics.PV): Waveform PV storing interlock alert timestamps.
        _hmiItlkCount (epics.PV): PV outputting the total number of active interlocks.
        _hmiBobfile (epics.PV): PV receiving template editor launch commands.
    """
    def __init__(self, argument):
        """Initialize the SRFsentinel instance with an initial control state flag."""
        log.info("initialized SRF sentinel")
        self.argument = argument
        self.trigger_pvs = {}
        self.driven_pvs = {}
        self._template_editor_request = "init"

    def general_settings(self,settings):
        """Ingest global project configuration and initialize rule parsing and PV connections.

    def log_to_waveform(self, new_message,new_timestamp):
        """Prepend a new interlock event message and timestamp to HMI EPICS waveforms.

        Maintains synchronized, non-empty waveform lists capped at a maximum of 500 entries (NELM).

        Args:
            new_message (str): Interlock alert message string (truncated to 39 chars max).
            new_timestamp (str): Formatted timestamp string (``%Y-%m-%d %H:%M:%S``).
        """
        safe_message = new_message[:39]
        # 1. Get the current list of messages
        # Use as_string=True if you need to ensure they come back as a list of strings
        current_messages = self._hmi_table.get(as_numpy=False)
        current_timestamps = self._hmi_timestamps.get(as_numpy=False)        
        if current_messages is None:
            current_messages = []
        else:
            # Convert to list if it comes back as a numpy array
            current_messages = list(current_messages)  
        if current_timestamps is None:
            current_timestamps = []
        else:
            # Convert to list if it comes back as a numpy array
            current_timestamps = list(current_timestamps)

        # 2. Prepend the new message
        # We strip empty strings to avoid the log getting filled with "nulls" 
        # from the initial 500 empty slots
        clean_messages = [m for m in current_messages if m and m.strip() != ""]
        clean_timestamps = [t for t in current_timestamps if t and t.strip() != ""]
        
        updated_messages = [safe_message] + clean_messages
        updated_timestamps = [new_timestamp] + clean_timestamps

        final_length = min(len(updated_messages), len(updated_timestamps), 500)
        synchronized_messages = updated_messages[:final_length]
        synchronized_timestamps = updated_timestamps[:final_length]

        # 3. Cap at 500 (your NELM) and write back
        self._hmi_table.put(synchronized_messages)
        self._hmi_timestamps.put(synchronized_timestamps)

    def _general_callback(self, pvname=None, value=None, **kwargs):
        """EPICS callback triggered when a monitored trigger PV value changes.

        Args:
            pvname (str, optional): Name of the updating EPICS PV.
            value (any, optional): New value received from EPICS.
            **kwargs: Additional callback parameters.
        """
        for rule in self._rules:
            if pvname == rule.trigger_pv: 
                rule.evaluate(value)

            value (str, optional): Target .bob filename requested for template editing.
            **kwargs: Additional callback parameters.

        if pvname != "init":
            self._template_editor_request = value

    def _load_sentinel_rules(self,filepath):
        """Parse interlock rules from the specified CSV file path.

        Args:
            filepath (str): Path to the ``srf_sentinel.csv`` rules file.
        """
        self._rules = []
        try:
            with open(filepath, mode='r', encoding='utf-8') as f:
                reader = csv.DictReader(f)
                for row in reader:
                    self._rules.append(SentinelRule(
                        row['Trigger PV'],
                        row['Trigger condition'],
                        row['driven PV'],
                        row['driven value'],
                        row['message']
                    ))
        except Exception as e:
            log.error(f"Error loading sentinel rules: {e}")

    def _connect_pvs(self):
        """Establish EPICS PV connections for HMI status logging, triggers, and driven targets."""
        self._hmi_table = epics.PV("SRF:Messages")
        if not self._hmi_table.wait_for_connection(timeout=1):
            log.error(f"HMI table PV failed to connect")
        self._hmi_timestamps = epics.PV("SRF:Timestamps")
        if not self._hmi_timestamps.wait_for_connection(timeout=1):
            log.error(f"HMI Timestamps PV failed to connect")
        self._hmiItlkCount = epics.PV("SRF:ItlkCount")
        if not self._hmiItlkCount.wait_for_connection(timeout=1):
            log.error(f"HMI ItlkCout PV failed to connect")
        self._hmiBobfile = epics.PV("SRF:bobfile")
        if not self._hmiBobfile.wait_for_connection(timeout=1):
            log.error(f"HMI Bobfile PV failed to connect")
        else:
            self._hmiBobfile.add_callback(self._bobfile_callback)
        for rule in self._rules:
            trigger_name = rule.trigger_pv
            if trigger_name not in self.trigger_pvs.keys():
                trigger_pv = epics.PV(trigger_name)
                if trigger_pv.wait_for_connection(timeout=1):
                    trigger_pv.add_callback(self._general_callback)
                    self.trigger_pvs[trigger_name] = trigger_pv
            driven_name = rule.driven_pv
            if driven_name not in self.driven_pvs.keys():
                driven_pv = epics.PV(driven_name)
                if driven_pv.wait_for_connection(timeout=1):
                    self.driven_pvs[driven_name] = driven_pv



    def __call__(self):
        """Callable entry point to start the sentinel execution loop."""
        log.debug("called SRF sentinel at {}".format(self))
        self.work()

    def work(self):
        """Continuous sentinel execution loop.

        Performs:
            1. Template editor subprocess launch handling.
            2. Interlock actuation dispatching to driven PVs.
            3. Logging interlock events and timestamps to EPICS waveform PVs.
            4. Updating active interlock count metric (``SRF:ItlkCount``).
        """
        while self.argument == 0:
            self.wait_please()
            if self._template_editor_request != "init":
                resources_dir = self._settings["console_position"] + os.path.sep + self._settings["resources_directory"]
                payload = EI.san_bob_file(resources_dir, self._template_editor_request)
                try:
                    GPL.launch_template_editor(payload)
                except Exception as e:
                    log.error("Failed to execute external GUI process: {}".format(e))
                self._template_editor_request = "init"
                self._hmiBobfile.put("init")
            itlk_count = 0 
            for rule in self._rules:
                if rule.execute:
                    try:
                        driven_pv = self.driven_pvs[rule.driven_pv]
                        driven_pv.put(rule.driven_val)
                        now = datetime.now()
                        timestamp = now.strftime("%Y-%m-%d %H:%M:%S")
                        log.info(f"{rule.message}")
                        self.log_to_waveform(rule.message,timestamp)
                        rule.execute = False 
                    except Exception as e:
                        log.error(e)
                if rule.isInterlocked:
                    itlk_count += 1
            self._hmiItlkCount.put(itlk_count)

    def talk(self):
        """Print active interlock statuses and pending puts to stdout."""
        print("\n\t******* printing SRF sentinel status:")
        has_interlocks = False
        for rule in self._rules:
            if rule.isInterlocked:
                print(f"interlocked {rule.driven_pv}")
                has_interlocks = True
            if rule.execute:
                print(f"pending put {rule.driven_val} into {rule.driven_pv}")
        if not has_interlocks:
            print("no interlocks")
