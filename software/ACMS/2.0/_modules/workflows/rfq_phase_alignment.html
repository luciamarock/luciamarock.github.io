"""Radio Frequency Quadrupole (RFQ) Phase Alignment Workflow Module.

This module implements the ``RFQPhaseAlignment`` workflow execution class, which manages 
the configuration, lifetime loop, and delegate monitoring for RFQ phase alignment tasks.

Key Functionality:
    1. Configuration Ingestion: Reads JSON runtime settings from 
       ``rfq_phase_alignment_config.json`` using paths defined in general settings.
    2. Worker Delegation: Instantiates and continuously executes monitoring loops 
       via :class:`workers.phase_alignment_worker.PhaseAlignmentWorker`.
    3. Workflow Lifecycle Management: Implements standard workflow controls including 
       start (``__call__``), graceful exit (``stop``), status reporting (``talk``), 
       and registration with :mod:`workflows_interface`.

Module Dependencies:
    - :mod:`workflows_interface`: Central workflow registry repository.
    - :mod:`workers.phase_alignment_worker`: Delegate worker class executing phase alignment routines.

Created on Mon Jul 20 09:10:28 2024

@author: luciamarock
"""

import time

class RFQPhaseAlignment:
    """RF Quadrupole Phase Alignment workflow management class.

    Attributes:
        argument (int): Execution flag state. ``0`` keeps the continuous workflow loop 
            running; non-zero value stops execution loop.
        _settings (dict): Configuration dictionary supplied by the application runner.
        data (dict): Parsed JSON configuration loaded from ``rfq_phase_alignment_config.json``.
        phase_alignment_worker (workers.phase_alignment_worker.PhaseAlignmentWorker):
            Underlying worker instance handling alignment monitoring and actions.
    """

    def __init__(self):
        """Initialize the RFQPhaseAlignment instance and set default loop state flag."""
        self.argument = 0
        log.info("RFQPhaseAlignment workflow initialized")

        self._settings = settings


    
    def __call__(self):
        """Executable entry point starting the continuous RFQ phase alignment monitoring loop.

        Loads configuration settings from ``rfq_phase_alignment_config.json``, instantiates 
        the underlying :class:`~workers.phase_alignment_worker.PhaseAlignmentWorker`, 
        and enters a pacing loop calling ``monitor_actions()`` every ~0.99 seconds until stopped.
        """
        log.info("RFQPhaseAlignment workflow started")
        config_file = self._settings["resources_directory"] + "rfq_phase_alignment_config.json"
        with open(config_file, 'r') as json_file:
            self.data = json.load(json_file)
        self.phase_alignment_worker = PAMW.PhaseAlignmentWorker(self.data)
        while self.argument == 0:
            self.phase_alignment_worker.monitor_actions()
            time.sleep(0.99)
