# -*- coding: utf-8 -*-
"""Interface module for the Workflows.

This module constitutes the operational starting point of the Recipe Management System
and ACMS execution framework. It is invoked directly by ``main_program.py`` after the 
hardware configuration and discovery phase ("knowledge acquisition") has finished loading 
the physical Smart Controller definitions from system resource XML files.

The interface is responsible for:
    1. Validating recipe folder structures and subsystem integrity.
    2. Registering and starting background background workflow processing threads.
    3. Dispatching execution to the requested User Interface / Operation Mode:
       
       * **API Mode:** Boots the ACMS background API client and execution runner to interact 
         with the LIPAc Campaign Manager (LCM) via polling (`GET /api/v0.5/automation/jobs/next`), 
         event batching (`POST /api/v0.5/automation/jobs/{job_id}/events`), and 2-second heartbeats.
       * **GUI Mode:** Launches the PyQt5 graphical interface for local operator control.
       * **CLI Mode:** Provides an interactive command-line loop for debugging, manual workflow 
         querying, and safe process spawning of tools like the Template Editor.

Module Dependencies:
    - :mod:`utilities.util_functions`: Global workspace configuration and settings state.
    - :mod:`controllersInterface`: Type and instance registration for EPICS controllers.
    - :mod:`editor_interface`: Archive scanning and template editor operations.
    - :mod:`gui_process_launcher`: Isolated process launcher for external PySide/PyQt tools.
    - :mod:`API.runner`: Entry point for launching the ACMS-LCM communication loop.

@author: luciamarock
"""

import os
import sys
import json
from threading import Thread
import logging 
log = logging.getLogger(__name__)

subsystem = None
registered_workflows = {}

import utilities.util_functions as UF
import controllersInterface as CI
import editor_interface as EI
import gui_process_launcher as GPL

def init():
    """Initialize workspace recipe directories and validate subsystem controller integrity.
    
    This function performs structural pre-flight checks before workflows can be safely executed:
        - Verifies the existence of the base ``recipes/`` directory in the workspace, creating it if missing.
        - Ensures all registered controller types belong to a single, unified hardware subsystem.
        - Checks that every controller instance declared inside a controller type actually exists within
          the registered instances dictionary, creating instance sub-folders as needed.
        - Establishes network/EPICS connectivity if an ``AutoConditioningWorkflow`` instance is registered.

    Returns:
        bool: ``True`` if initialization and connectivity checks succeeded, enabling 
        ``main_program`` to invoke :func:`start()`; ``False`` otherwise.
    """
    global subsystem
    log.debug("workflows interface started")
    complete = True # completeness flag to enable the start() from the main_program
    recipes_dir = UF.settings["workspace_dir"] + "recipes" + os.path.sep
    if not os.path.exists(recipes_dir):
        log.info("creating recipe directory: {}".format(recipes_dir))
        os.makedirs(recipes_dir)
    for ctype in CI.registered_types.keys():
        if subsystem is None:
            subsystem = CI.registered_types[ctype].subsystem
        elif CI.registered_types[ctype].subsystem != subsystem:
            log.warning("controller type from another subsystem found, this could lead to future errors")
        for cinst in CI.registered_types[ctype]._registered_instances:
            if cinst not in CI.registered_instances.keys():
                log.error("instance {} is declared in type {} but not found among the registered instances".format(cinst,ctype))
            else:
                inst_dir = recipes_dir + cinst
                if not os.path.exists(inst_dir):
                    log.info("creating instance sub-folder {}".format(inst_dir))
                    os.makedirs(inst_dir)
    if "AutoConditioningWorkflow" in registered_workflows.keys():
        act_instance = registered_workflows["AutoConditioningWorkflow"]
        complete = act_instance.connect()
    return complete

def start():
    """Start workflow worker threads and launch the selected operating mode.
    
    The execution flow progresses as follows:
        1. Invokes :func:`CI.create_validators()` to build runtime validators across all controllers.
        2. Configures general settings and spawns dedicated background daemon threads for every 
           registered workflow (e.g., automatic workflow, sentinel workers, phase alignment).
        3. Evaluates ``UF.settings["UI"]`` to branch into the designated interface loop:
        
           - **API Mode:** Imports and launches the ACMS API Runner service (:func:`API.runner.start_api_server`),
             connecting ACMS to the LIPAc Campaign Manager (LCM) backend endpoints over HTTP. Blocks on a 1-second 
             keep-alive loop until interrupted.
           - **GUI Mode:** Initializes a :class:`PyQt5.QtWidgets.QApplication` instance and displays the 
             main desktop application window.
           - **CLI Mode:** Enters a terminal loop processing interactive commands (``status``, ``editor``, 
             ``help``, ``exit``) and gracefully stops registered workflows upon termination.

    Raises:
        KeyboardInterrupt: Trapped during API or CLI modes to safely halt registered workflow threads 
            and release hardware/EPICS resources prior to process shutdown.
    """
    errors = CI.create_validators()
    for message in errors:
        log.error(message)
    for workflow_name, workflow_instance in registered_workflows.items():
        if workflow_name == "automatic_workflow":
            auto_mode_mapping = UF.settings["resources_directory"] + "auto_mode_mapping.json"
            with open(auto_mode_mapping, 'r') as json_file:
                data = json.load(json_file)
            workflow_instance.set_auto_mode_mapping_data(data)
        workflow_instance.general_settings(UF.settings)
        if float(sys.version[0]) < 3:
            t = Thread(target=workflow_instance) # , args=(1,)
            t.daemon = True
            t.start()
        else:
            t = Thread(target=workflow_instance, daemon=True) # , args=(1,), daemon=True
            t.start()
    if UF.settings["UI"] == "API":
        log.info("Starting ACMS API Service for LCM communication...")
        from API.runner import start_api_server
        
        host = UF.settings.get("api_host", "127.0.0.1")
        port = int(UF.settings.get("api_port", 9000))
        start_api_server(host=host, port=port)
        #TODO Operate the workflows
        # Keep main thread alive if running strictly in API mode
        try:
            import time
            while True:
                time.sleep(1)
        except KeyboardInterrupt:
            log.info("Stopping ACMS API service...")
            for key in registered_workflows.keys():
                registered_workflows[key].stop()
    elif UF.settings["UI"] == "GUI":
        #TODO Operate the workflows instead
        recipe_manager = UF.MockRecipeManager()
        log.info("Starting PyQt5 Graphical User Interface")
        from PyQt5.QtWidgets import QApplication
        from GUI.gui_main import MainWindow
        app = QApplication(sys.argv)
        main_win = MainWindow(recipe_manager)
        main_win.show()
        # Exec loop blocks until main window is closed
        sys.exit(app.exec_())
    elif UF.settings["UI"] == "CLI":
        log.info("using the CLI")
        import utilities.command_line_interface as cli
        try:
            command_help = {
                "help": "print this message",
                "status": "shows the status of the workers",
                "exit": "exit the program",
                "editor": "opens the template editor"
            }
            Uinp = "init"
            while Uinp != "exit":
                Uinp = cli.start().strip()
                log.info("Received command: {}".format(Uinp))
                if Uinp == "status":
                    for key in registered_workflows.keys():
                        registered_workflows[key].talk()
                elif Uinp == "exit":
                    log.info("Stopping Registered Workflows")
                elif Uinp == "editor":
                    log.info("Scanning Archive Network")
                    if "home" not in UF.settings["resources_directory"]:
                        resources_dir = UF.settings["console_position"] + os.path.sep +UF.settings["resources_directory"]
                    else:
                        resources_dir = UF.settings["resources_directory"]
                    payload = EI.scan_archive_network(resources_dir)
                    if not payload:
                        log.error("Network sweep returned an empty payload. Aborting GUI launch.")
                        continue
                    log.info("Spawning Template Editor GUI process window safely...")
                    try:
                        GPL.launch_template_editor(payload)
                    except Exception as e:
                        log.error("Failed to execute external GUI process: {}".format(e))
                elif Uinp == "help":
                    for key in command_help.keys():
                        print("{}: {}".format(key,command_help[key]))
                else:
                    log.warning("command {} not recognised".format(Uinp))
            for key in registered_workflows.keys():
                    registered_workflows[key].stop()
        except KeyboardInterrupt:
            log.info("Recipe Management System interrupted by the User")
            for key in registered_workflows.keys():
                registered_workflows[key].stop()
            sys.exit()
