def parse_xml_types():
    """It gets called immediately from the main_program
    
    This function is responsible for collecting the .xml files describing the Controller's types. \n
    It goes into the directory specified in UF.settings["controller_types_location"] and collects all the .xml files in the root folder, 
    please notice it does not walk the sub-folders ! \n
    For each type .xml file it retrieves the root node and with that, it invokes the :obj:`SmartControllers.factories.types_factories.create_type`
    function in order to create a class object representing the current type of Smart Controller. \n
    Once the software object is created it register it in the module level registered_types dictionary. \n
    
    At the beginning this function also initializes the log (writing all the settings) and instantiates the :obj:`projectmodel.projectModel.projectmodel` 
    software object (Project). 
    
    """
    global Project
    _start_log()
    Project = PM.projectmodel(UF.settings)
    if _check_dir(UF.settings["controller_types_location"]):
        if UF.settings["controller_types_location"].endswith(os.path.sep):
            path_to_files = UF.settings["controller_types_location"]
        else:
            path_to_files = UF.settings["controller_types_location"] + os.path.sep
        for root, dirs, files in os.walk(path_to_files):
            if root == path_to_files: # ignoring sub-folders
                for name in files:
                    if name.endswith("xml"):
                        log.info("loading {}".format(path_to_files + name))
                        tree = etree.parse(path_to_files + name) 
                        rootnode = tree.getroot()
                        key, value = type_factory.create_type(rootnode,name)
                        if key not in registered_types:
                            registered_types[key] = value
                        else:
                            log.error("cannot register {} name already exist".format(path_to_files + name))
                    else: # ignoring non xml files 
                        log.info("skipping {}".format(path_to_files + name))
            else:
                log.info("ignoring sub-directory {}".format(root))
        return
    else:
        log.error("couldn't find specified directory for controller types")
        return

def parse_xml_instances():
    """Called right after parse_xml_types function by the main_program. 
    
    This function mirrors parse_xml_types but parses the .xml configuration files for the Controller's instances. 
    Unlikely the parse_xml_types, this function "walks" also the sub-directories of the UF.settings["controller_instances_location"]. 
    This allows the users to create their own organization structure in the file-system. \n
    When parsing the xml files, if the instance name is not found in the module-level registered_instances dictionary, this function 
    proceed to the call of :obj:`SmartControllers.factories.instances_factories.create_instance` creation function for the Instances. 
    \n Later on, if the returned instance is not None, it register the object in the registered_instance dictionary and 
    it calls the project model class to add the new created object among the  ProjectModel nodes. 
    
    """
    global Project
    if _check_dir(UF.settings["controller_instances_location"]):
        for root, dirs, files in os.walk(UF.settings["controller_instances_location"]):
            for name in files:
                if name.endswith("xml"):
                    if root.endswith(os.path.sep):
                        path_to_files = root
                    else:
                        path_to_files = root + os.path.sep
                    log.info("loading {}".format(path_to_files + name))
                    tree = etree.parse(path_to_files + name) 
                    rootnode = tree.getroot()
                    controller_instance_name = name.split(".")[0]
                    if controller_instance_name not in registered_instances.keys(): # instance names must be unique 
                        current_instance = instance_factory.create_instance(rootnode,controller_instance_name,registered_types)
                        if current_instance is not None:
                            registered_instances[controller_instance_name] = current_instance
                            Project.add_node(current_instance) #TODO check this 
                        else:
                            log.error("cannot register instance {}, couldn't create its class".format(controller_instance_name))
                    else:
                        log.error("cannot register instance {}, an instance with the same name already exist".format(controller_instance_name))
                else:
                    log.info("skipping file {} in the instances folder".format(name))
    else:
        log.error("couldn't find specified directory for controller instances")
        return 
    Project.export()

def create_validators():
    """Recipes (from file .json) loaders
    
    :returns: **errors** a list of error messages  
    
    At the beginning this function tries to create the validator objects (:obj:`utilities.validation.validation_engine.create`), 
    if the creation is successfull it register the object in the 
    module level validators dictionary (in case to be reused by other modules).
    The validators are declared in the settings.ini file.\n
    call the validate method like this:\n
    errors = validators[validator_name].validate(registered_instances[instance],PVs,os.path.splitext(filename)[0])
    
    """
    errors = []
    if not validators:
        for validator in UF.settings["validators"]:
            current_validator = validation.create(validator,UF.settings["resources_directory"])
            if current_validator is not None:
                validators[validator] = current_validator
            else:
                message = "current validator is None, unable to create"
                errors.append(message)
    
            
    return errors
