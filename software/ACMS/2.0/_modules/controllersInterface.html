"""Controllers Interface Module.

Created on Thu Jan  5 11:57:03 2023

Serves as the central registry and parser interface for Smart Controller types and 
instances. Discovers XML definition files, instantiates controller hardware/software 
objects via factories, populates the central :class:`projectmodel.projectModel.projectmodel` 
tree, and manages validation engines.

.. note::
   This module is imported by :mod:`main_program` immediately following 
   the execution of the ``_init()`` setup function.

:author: luciamarock
"""

import sys
import logging 
log = logging.getLogger(__name__)
from lxml import etree
import SmartControllers.factories.types_factories as type_factory 
import SmartControllers.factories.instances_factories as instance_factory

import json

registered_instances = {}

#: dict: Registry containing instantiated validation engines specified in runtime settings.

#: projectmodel.projectModel.projectmodel or None: Global reference to the active project model instance.

def _check_dir(directory):
    """Verify filesystem directory existence.

    :param directory: Absolute or relative directory path to inspect.
    :type directory: str
    :return: ``True`` if the path exists, ``False`` otherwise.
    :rtype: bool
    """
    if os.path.exists(directory):
        return True
    return False

def parse_xml_types():
    """Parse and register Smart Controller type definitions from XML.

    Scans the directory specified by ``UF.settings["controller_types_location"]`` 
    for XML configuration files (ignoring subdirectories). For each file:
    
    1. Parses the XML tree using :mod:`lxml.etree`.
    2. Invokes :func:`SmartControllers.factories.types_factories.create_type` to instantiate 
       the controller type class object.
    3. Registers the returned object into :data:`registered_types`.

    Additionally, this function calls :func:`_start_log` to record startup runtime 
    settings and instantiates the global :class:`~projectmodel.projectModel.projectmodel` 
    object (:data:`Project`).

    .. seealso::
       * :func:`parse_xml_instances` - Invoked after type parsing completes.
       * :func:`SmartControllers.factories.types_factories.create_type`
    """
    global Project
    _start_log()
    Project = PM.projectmodel(UF.settings)
    if _check_dir(UF.settings["controller_types_location"]):
        if UF.settings["controller_types_location"].endswith(os.path.sep):
            path_to_files = UF.settings["controller_types_location"]
        else:
            path_to_files = UF.settings["controller_types_location"] + os.path.sep
        for root, dirs, files in os.walk(path_to_files):
            if root == path_to_files: # ignoring sub-folders
                for name in files:
                    if name.endswith("xml"):
                        log.info("loading {}".format(path_to_files + name))
                        tree = etree.parse(path_to_files + name) 
                        rootnode = tree.getroot()
                        key, value = type_factory.create_type(rootnode,name)
                        if key not in registered_types:
                            registered_types[key] = value
                        else:
                            log.error("cannot register {} name already exist".format(path_to_files + name))
                    else: # ignoring non xml files 
                        log.info("skipping {}".format(path_to_files + name))
            else:
                log.info("ignoring sub-directory {}".format(root))
        return
    else:
        log.error("couldn't find specified directory for controller types")
        return

def parse_xml_instances():
    """Recursively parse and register Smart Controller instance definitions.

    Walks the root directory and all subdirectories within 
    ``UF.settings["controller_instances_location"]`` to load instance XML files. 
    
    For each unique instance name encountered:
    
    1. Invokes :func:`SmartControllers.factories.instances_factories.create_instance` 
       passing the root node and :data:`registered_types`.
    2. Registers successfully created instance objects in :data:`registered_instances`.
    3. Appends the instance node to the global :data:`Project` tree.
    4. Triggers :meth:`projectmodel.projectModel.projectmodel.export` upon completion.

    .. seealso::
       * :func:`parse_xml_types`
       * :func:`SmartControllers.factories.instances_factories.create_instance`
    """
    global Project
    if _check_dir(UF.settings["controller_instances_location"]):
        for root, dirs, files in os.walk(UF.settings["controller_instances_location"]):
            for name in files:
                if name.endswith("xml"):
                    if root.endswith(os.path.sep):
                        path_to_files = root
                    else:
                        path_to_files = root + os.path.sep
                    log.info("loading {}".format(path_to_files + name))
                    tree = etree.parse(path_to_files + name) 
                    rootnode = tree.getroot()
                    controller_instance_name = name.split(".")[0]
                    if controller_instance_name not in registered_instances.keys(): # instance names must be unique 
                        current_instance = instance_factory.create_instance(rootnode,controller_instance_name,registered_types)
                        if current_instance is not None:
                            registered_instances[controller_instance_name] = current_instance
                            Project.add_node(current_instance) #TODO check this 
                        else:
                            log.error("cannot register instance {}, couldn't create its class".format(controller_instance_name))
                    else:
                        log.error("cannot register instance {}, an instance with the same name already exist".format(controller_instance_name))
                else:
                    log.info("skipping file {} in the instances folder".format(name))
    else:
        log.error("couldn't find specified directory for controller instances")
        return 
    Project.export()

def create_validators():
    """Initialize recipe validators declared in configuration settings.

    Attempts to instantiate each validator listed in ``UF.settings["validators"]`` 
    via :func:`validation.validation_engine.create`. Successfully created objects 
    are registered in the :data:`validators` dictionary.

    :return: A list of error message strings describing any failed validator instantiations.
    :rtype: list of str

    .. note::
       Registered validators can subsequently be executed using:
       
       .. code-block:: python

          errors = validators[validator_name].validate(
              registered_instances[instance], PVs, os.path.splitext(filename)[0]
          )
    """
    errors = []
    if not validators:
        for validator in UF.settings["validators"]:
            current_validator = validation.create(validator,UF.settings["resources_directory"])
            if current_validator is not None:
                validators[validator] = current_validator
            else:
                message = "current validator is None, unable to create"
                errors.append(message)
    
            
    return errors
