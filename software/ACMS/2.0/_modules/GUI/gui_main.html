    def emit(self, record):

class LogWindow(QWidget):
    """Filterable Log Viewer Window."""
    def __init__(self, handler):
        super().__init__()
        self.setWindowTitle("System Log Viewer")
        self.resize(700, 400)
        
        layout = QVBoxLayout(self)

        # Filter Control
        filter_layout = QHBoxLayout()
        filter_layout.addWidget(QLabel("Min Log Level:"))
        self.level_combo = QComboBox()
        self.level_combo.addItems(["DEBUG", "INFO", "WARNING", "ERROR", "CRITICAL"])
        self.level_combo.setCurrentText("INFO")
        self.level_combo.currentTextChanged.connect(self.filter_logs)
        filter_layout.addWidget(self.level_combo)
        filter_layout.addStretch()
        layout.addLayout(filter_layout)

        # Text Area
        self.text_edit = QTextEdit()
        self.text_edit.setReadOnly(True)
        layout.addWidget(self.text_edit)

        # Store logs locally to re-filter on the fly
        self.all_records = []
        handler.emitter.log_signal.connect(self.append_log)

    def append_log(self, message, levelno):
        self.all_records.append((message, levelno))
        if levelno >= self.get_selected_levelno():
            self.text_edit.append(message)

    def get_selected_levelno(self):
        level_name = self.level_combo.currentText()
        return getattr(logging, level_name)

    def filter_logs(self):
        self.text_edit.clear()
        selected_level = self.get_selected_levelno()
        for msg, levelno in self.all_records:
            if levelno >= selected_level:
                self.text_edit.append(msg)

    def setup_logging(self):
        """Attach QtLogHandler to root logger and prepare Log Window."""
        self.log_handler = QtLogHandler()
        formatter = logging.Formatter('%(asctime)s - %(name)s - %(levelname)s - %(message)s')
        self.log_handler.setFormatter(formatter)
        logging.getLogger().addHandler(self.log_handler)
        
        self.log_window = LogWindow(self.log_handler)

    def setup_menu(self):
        """Construct Top Menu Bar."""
        menu_bar = self.menuBar()

        # Help / Info Menu
        help_menu = menu_bar.addMenu("&Help")

        # 1. Online Documentation

        view_menu = menu_bar.addMenu("&View")

        # 3. Log Display Window
        log_action = QAction("Log Viewer", self)
        log_action.triggered.connect(self.show_log_window)
        view_menu.addAction(log_action)

    def load_tabs(self):
        """Dynamically load python modules present in GUI/tabs directory."""
        tabs_dir = os.path.join(os.path.dirname(__file__), "tabs")
        if not os.path.exists(tabs_dir):
            log.warning("Tabs directory not found: {}".format(tabs_dir))
            return

        for fname in sorted(os.listdir(tabs_dir)):
            if fname.endswith(".py") and not fname.startswith("__"):
                module_name = fname[:-3]
                full_module_path = "GUI.tabs.{}".format(module_name)
                
                try:
                    mod = importlib.import_module(full_module_path)
                    if hasattr(mod, "get_tab"):
                        # Factory function expected in tab module: get_tab(recipe_manager) -> QWidget, tab_title
                        widget, title = mod.get_tab(self.recipe_manager)
                        self.tabs.addTab(widget, title)
                        log.info("Loaded tab module: {}".format(module_name))
                    else:
                        log.warning("Module {} missing 'get_tab' function".format(module_name))
                except Exception as e:
                    log.error("Failed to load tab module {}: {}".format(module_name, e))

    def closeEvent(self, event):
        """Ensure clean shutdown when the main window is closed by operator."""
        log.info("Closing Main Window. Stopping Recipe Manager...")
        
        # Close Log Window if open
        if self.log_window:
            self.log_window.close()

        # Stop background workflow thread
        if self.recipe_manager:
            self.recipe_manager.stop()

        event.accept()
