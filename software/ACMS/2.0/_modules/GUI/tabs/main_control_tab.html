from PyQt5.QtWidgets import QWidget, QVBoxLayout, QPushButton, QLabel, QMessageBox
from PyQt5.QtCore import Qt

def get_tab(recipe_manager):
    """Tab Factory Function required by Main Window dynamic loader."""
    widget = QWidget()
    layout = QVBoxLayout(widget)

    label = QLabel("Smart Controller Operational Controls")
    label.setAlignment(Qt.AlignCenter)
    layout.addWidget(label)

    # Trigger Talk Command
    talk_btn = QPushButton("Talk to Controller")
    talk_btn.setHeight = 40

    def on_talk_clicked():
        if recipe_manager:
            recipe_manager.talk()
            QMessageBox.information(widget, "Talk Executed", "Sent 'talk' command to Recipe Manager.")

    talk_btn.clicked.connect(on_talk_clicked)
    layout.addWidget(talk_btn)

    return widget, "Control Panel"
