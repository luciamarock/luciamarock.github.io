# -*- coding: utf-8 -*-
"""Validation Engine Module for ACMS Recipe & Worker Configurations.

This module provides concrete implementations of validation classes used to verify 
recipe parameters, Smart Controller instance attributes, and worker execution rules. 

Validation logic relies on external rule definitions formatted as either INI configuration 
files (parsed via :class:`configparser.ConfigParser`) or XML schema files (parsed via 
:mod:`xml.etree.ElementTree`).

Module Dependencies:
    - :mod:`utilities.tool_internals`: XML-to-dictionary translation utilities.
    - :mod:`utilities.util_functions`: Access to system environment settings.
    - :mod:`configparser`: INI rule configuration parser.
    - :mod:`xml.etree.ElementTree`: XML schema rule parser.

:author: luciamarock
"""

# validation rules will be an .ini file and will contain a string to be evaluated (ex. "value < 10 and value > 0")
# eval will return True or False (or exception if the string is wrongly formatted)

import utilities.tool_internals as TI
import utilities.util_functions as UF
import xml.etree.ElementTree as ET
import os 
import configparser

import logging 
log = logging.getLogger(__name__)

class TPG_validator():
    """Recipe validator class for TPG Smart Controllers.

    The :class:`TPG_validator` object is typically instantiated during workflow startup when 
    loading JSON recipe files. It reads rule expressions from an INI configuration file 
    named after the target validator class.

    :param filename: Absolute or relative path to the INI rules configuration file.
    :type filename: str
    """

    def __init__(self,filename):
        self._rules = configparser.ConfigParser()
        self._rules.read(filename)

    def validate(self,instance, PVs, filename):
        """Validate a TPG Smart Controller instance against loaded INI rules.

        Extracts target attributes from the controller instance and evaluates the rule 
        expressions defined in the ``[attributes]`` section of the INI file. Furthermore, 
        it maps and registers recipe parameters to the target controller instance if a 
        matching recipe key exists in the ``[recipes]`` section.

        :param instance: The Smart Controller instance registered in the runtime catalog.
        :type instance: object
        :param PVs: Dictionary of Process Variables and parameters loaded from a JSON recipe.
        :type PVs: dict
        :param filename: Base filename of the target JSON recipe (without extension).
        :type filename: str

        :return: List of string error messages encountered during attribute evaluation or 
            recipe key mapping. Returns an empty list if validation passes entirely.
        :rtype: list of str
        """
        errors = []
        attributes_of_interest = list(self._rules["attributes"].keys())
        for attribute in attributes_of_interest:
            value = getattr(instance, attribute, None) # None, default value returned if the attribute is not found
            expression = self._rules["attributes"][attribute]
            if not eval(expression):
                message = instance.instance + ": expression " + expression + " for attribute " + attribute + " failed vaildation check"
                errors.append(message)
        if filename in list(self._rules["recipes"].keys()):
            instance.set_PVs(self._rules["recipes"][filename],PVs)
        else:
            message = instance.instance + ": couldn't find a key for recipe " + filename + ".json"
            errors.append(message)

        return errors

class worker_validator:
    """Validator class for worker process parameters and safety interlocks.

    Loads rule specifications from an XML schema file located in the application's resources 
    directory (``<resources_directory>/rules/<instance_name>_rules.xml``) to verify type 
    constraints, range limits (min/max), and structural validity of interlock definitions.

    :param parameters_dict: Operational parameters configured for the worker instance.
    :type parameters_dict: dict
    :param interlocks: Dictionary of defined interlocks, rearm flags, and error codes.
    :type interlocks: dict
    :param instance_name: Unique identifier name of the target worker instance.
    :type instance_name: str
    """

    def __init__(self,parameters_dict, interlocks, instance_name):
        log.info(f'initializing {instance_name} validator')
        self._report = {}
        self._report["status"] = "clean"
        self._report["messages"] = []
        self._parameters = parameters_dict
        self._interlocks = interlocks
        self._settings = UF.settings
        base_dir = self._settings["resources_directory"]
        rules_directory = base_dir + "rules" + os.path.sep
        rules_filename = rules_directory + instance_name + "_rules.xml"
        try:
            tree = ET.parse(rules_filename)
            root = tree.getroot()            
        except Exception as e:
            log.error(e)
            self._report["status"] = "error"
            self._report["messages"].append("file not found") 
        if self._report["status"] != "clean":
            return 
        try:
            parameters = root.find("parameters")
            self._parameters_rules = TI.element_with_attributes_to_dict(parameters)
        except Exception as e:
            log.error(e)
            self._report["status"] = "error"
            self._report["messages"].append("parameters not found") 
        self._instance_name = instance_name

    def check_data(self):
        """Execute validation checks across all parameters and interlocks.

        Evaluates configured parameters against loaded XML rule definitions (verifying data types 
        and range constraints via helper methods) and validates that all interlock entries contain 
        the required ``rearmable`` and ``codes`` fields.

        :return: A dictionary containing the validation summary with key ``status`` 
            (``'clean'`` or ``'error'``) and key ``messages`` listing encountered issues.
        :rtype: dict
        """
        for key in self._parameters_rules.keys():
            log.info(f"checking parameter {key}")
            if key not in self._parameters.keys():
                log.warning(f'ruled parameter {key} not found in worker configuration file of instance {self._instance_name}')
            else:
                current_rule = self._parameters_rules[key]
                if "type" not in current_rule.keys():
                    self._report["status"] = "error"
                    self._report["messages"].append(f'rule {key} of instance {self._instance_name} does not have a type')
                else:
                    if current_rule["type"] == "float":
                        self._check_float(self._parameters[key], current_rule)
                    #TODO add other types
        for key in self._interlocks:
            current_value = self._interlocks[key]
            if "rearmable" not in current_value.keys() or "codes" not in current_value.keys():
                self._report["status"] = "error"
                self._report["messages"].append(f'error in definition of {key} in {self._instance_name}')

        return self._report

    def _check_float(self, value, rule):
        """Internal helper to validate float bounds against rule minimum and maximum limits.

        :param value: The raw parameter value to check.
        :type value: int, float, or str
        :param rule: Rule dictionary specifying ``min`` and ``max`` numerical bounds.
        :type rule: dict
        """
        try:
            value = float(value)
        except Exception as e:
            self._report["status"] = "error"
            self._report["messages"].append(e)
            return 
        try:
            max_value = float(rule["max"])
        except Exception as e:
            self._report["status"] = "error"
            self._report["messages"].append(e)
            return 
        try:
            min_value = float(rule["min"])
        except Exception as e:
            self._report["status"] = "error"
            self._report["messages"].append(e)
            return
        if value > max_value or value < min_value:
            self._report["status"] = "error"
            self._report["messages"].append(f'value {value} is out of range for instance {self._instance_name}')
