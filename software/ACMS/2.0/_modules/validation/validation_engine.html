class TPG_validator():
    """Recipe validator class 
    
    The TPG_validator object is tipically used when loading the recipes from files (.json),
    this happens at workflow startup (for example). \n
    It receives the "filename" as initialization parameter; it is the full path to the "rules" file to load
    which is an .ini file named after the validator itself. \n
    The TPG_validator (as well as the other validator classes) is instantiated by the "create" function 
    of this module. 
    
    """
    def __init__(self,filename):
        self._rules = configparser.ConfigParser()
        self._rules.read(filename)

    def validate(self,instance, PVs, filename):
        """This method is responsible for validating the TPG instances  
        
        it bases upon the reules file (which is an .ini file named after this validator). \n
        
        :param instance: The (Smart Controller) instance (as retrieved from `controller_interface.registered_instances`)
        :param PVs: The dictionary of parameters, as retrieved from the .json file (recipes) at workflow startup.
        :param filename: the .json file name found in the recipes folder (and sub-folders)
        
        The method tries to find the current "instance.attribute" among the "options" of the .ini file containing the rules, 
        then it tries to evaluate the expression for that attribute, if the evaluation is successfully 
        completed it means that the parameter has passed the validation phase. \n
        Next it searches for a key for storing the set of parameters inside the Controller's Instance, in case, 
        it appends the errors.  
        
        :returns: the errors list (empty if no error occurred)
        """
        errors = []
        attributes_of_interest = list(self._rules["attributes"].keys())
        for attribute in attributes_of_interest:
            value = getattr(instance, attribute, None) # None, default value returned if the attribute is not found
            expression = self._rules["attributes"][attribute]
            if not eval(expression):
                message = instance.instance + ": expression " + expression + " for attribute " + attribute + " failed vaildation check"
                errors.append(message)
        if filename in list(self._rules["recipes"].keys()):
            instance.set_PVs(self._rules["recipes"][filename],PVs)
        else:
            message = instance.instance + ": couldn't find a key for recipe " + filename + ".json"
            errors.append(message)

        return errors

def create(class_name,resources_dir):
    """Instantiates the requested validator class
    
    :returns: the instance or `None` in case the rules (.ini) file does not exist or the requested validator class does not exist
    
    """
    filename = resources_dir + os.path.sep + "rules" + os.path.sep + class_name + "_rules.ini"
    if os.path.isfile(filename):
        log.info("creating {} validator".format(class_name))
        try:
            class_instance = eval(class_name + "(filename)")
        except Exception as e:
            log.error(e)
            return None
    else:
        log.error("file {} not found".format(filename))
        return None
    
    return class_instance
