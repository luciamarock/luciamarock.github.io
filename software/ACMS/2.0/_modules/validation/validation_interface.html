# -*- coding: utf-8 -*-
"""Validation Factory Interface for ACMS Recipe & Rules Engine.

This module provides a dynamic factory mechanism for instantiating custom validator classes 
defined within :mod:`validation.validation_engine`. 

Validator instances are configured dynamically at runtime by loading dedicated INI configuration 
rules files (e.g., ``<class_name>_rules.ini``) from the system resources directory.

Module Dependencies:
    - :mod:`validation.validation_engine`: Core validation engine containing validator implementations.

:author: luciamarock
"""

import os
import validation.validation_engine as VE
import logging

log = logging.getLogger(__name__)

def create(class_name,resources_dir):
    """Instantiate a requested validator class using a corresponding INI rules file.

    Locates and parses the configuration file named ``<class_name>_rules.ini`` inside the 
    ``rules`` subdirectory of ``resources_dir``. If the rule file exists and the requested 
    validator class is defined within :mod:`validation.validation_engine`, it dynamically 
    instantiates and returns the validator object.

    :param class_name: The name of the validator class to instantiate (e.g., ``'RecipeValidator'``).
    :type class_name: str
    :param resources_dir: Absolute or relative base path to the application's resources directory.
    :type resources_dir: str

    :return: An instance of the requested validator class initialized with its rule set, 
        or ``None`` if the rules file is missing or instantiation fails.
    :rtype: object or None
    """
    filename = resources_dir + os.path.sep + "rules" + os.path.sep + class_name + "_rules.ini"
    if os.path.isfile(filename):
        log.info("creating {} validator".format(class_name))
        try:
            class_instance = eval("VE." + class_name + "(filename)")
        except Exception as e:
            log.error(e)
            return None
    else:
        log.error("file {} not found".format(filename))
        return None
    
    return class_instance
