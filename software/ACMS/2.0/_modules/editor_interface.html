def load_configuration(config_path):
    """Loads runtime variables and target fields from the config file."""
    try:
        with open(config_path, "r", encoding="utf-8") as config_file:
            config_data = json.load(config_file)
            
        return (
            config_data["CA_TIMEOUT"],
            config_data["CA_PARALLELISM"],
            config_data["PV_FIELDS"],
            config_data["HMI_FILES"]
        )
    except FileNotFoundError:
        log.error(f"CRITICAL ERROR: Configuration file not found at {config_path}")
        sys.exit(1)
    except KeyError as e:
        log.error(f"CRITICAL ERROR: Missing expected configuration key in JSON: {e}")
        sys.exit(1)
    except json.JSONDecodeError as e:
        log.error(f"CRITICAL ERROR: Invalid JSON syntax in config file: {e}")
        sys.exit(1)

def scan_pv_worker(name, results_dict, fields_to_get, timeout):
    """Worker thread target to extract EPICS metadata safely via shared context."""
    log.debug(f'Scanning {name} ...')
    try:
        # Attach child thread to parent CA context loop
        epics.ca.attach_context(MAIN_CONTEXT)
        
        # Connect to the PV using timeout configurations
        pv = epics.pv.get_pv(name, connect=True, timeout=timeout)
        
        if pv.connected:
            fields = dict([(f, pv.__getattribute__(f)) for f in fields_to_get])
            results_dict[name] = fields
            pv.disconnect()
        else:
            log.warning(f"[-] Cannot connect to {name} (Timed out)")
            results_dict[name] = None
            
    except Exception as e:
        log.error(f"[-] Exception scanning {name}: {e}")
        results_dict[name] = None

def run_network_sweep(pv_names, parallelism, fields_to_get, timeout):
    """Orchestrates concurrent thread batches to profile the PV network."""
    global MAIN_CONTEXT
    log.info(f"Connecting all {len(pv_names)} PVs safely using a shared CA Context...")
    
    # Initialize Channel Access and extract runtime context loop
    epics.ca.initialize_libca()
    MAIN_CONTEXT = epics.ca.current_context()

    pv_map = {}

    # Process batch loops matching the parallelism limits
    for i in range(0, len(pv_names), parallelism):
        chunk = pv_names[i:i + parallelism]
        active_batch = []
        
        for name in chunk:
            t = epics.ca.CAThread(
                target=scan_pv_worker, 
                args=(name, pv_map, fields_to_get, timeout)
            )
            active_batch.append(t)
            t.start()
            
        for t in active_batch:
            t.join()

    log.info("Done scanning network safely.")
    return pv_map

def sanitize_data(pv_map):
    """Deeply cleans floating NaN math flags into valid JSON nulls."""
    log.info("Sanitizing telemetry data fields...")
    for pv_name, fields in pv_map.items():
        if fields is not None:
            for key, value in fields.items():
                if isinstance(value, float) and math.isnan(value):
                    fields[key] = None  # Maps to a clean JSON 'null'
    return pv_map

def save_manifest(pv_map, output_path):
    """Dumps the final, sanitized map cleanly to disk."""
    output_path.parent.mkdir(parents=True, exist_ok=True)
    with open(output_path, 'w', encoding='utf-8') as json_file:
        json.dump(pv_map, json_file, sort_keys=True, indent=4, ensure_ascii=False)
    print(f"Manifest saved cleanly without crashes to: {output_path}")

def scan_archive_network(resources_dir):
    """Scans the archive network and returns a list of PVs."""
    log.info("loading configuration file")
    config_file = resources_dir + "config.json"
    ca_timeout, ca_parallelism, pv_fields, hmi_files = load_configuration(config_file)
    log.info("configuration file loaded")
    log.info("fetching PV list from database")
    try:
        pv_names = get_list_of_pvs('localhost', 'lipac_mock', 'archive', 'rockoDD1o')
        pv_names.sort()
    except Exception as e:
        log.error(f"CRITICAL DATABASE ERROR: {e}")
        sys.exit(1)
    log.info("PV list fetched from database")
    log.info("running network sweep")
    raw_pv_map = run_network_sweep(pv_names, ca_parallelism, pv_fields, ca_timeout)
    log.info("network sweep completed")
    log.info("sanitizing data")
    clean_pv_map = sanitize_data(raw_pv_map)
    log.info("data sanitized")
    return clean_pv_map

def san_bob_file(resources_dir, key):
    log.info(f"Received Template Editor Request from {key}")
    log.info("loading configuration file")
    config_file = resources_dir + "config.json"
    ca_timeout, ca_parallelism, pv_fields, hmi_files = load_configuration(config_file)
    log.info("configuration file loaded")
    bob_file = hmi_files[key]
    log.info(f"fetching PV list from {bob_file}")
    try:

        pv_names.sort()
    except Exception as e:
        log.error(f"CRITICAL ERROR: {e}")
        sys.exit(1)
    log.info("PV list fetched from bob file")
    log.info("running network sweep")
    raw_pv_map = run_network_sweep(pv_names, ca_parallelism, pv_fields, ca_timeout)
    log.info("network sweep completed")
    log.info("sanitizing data")
    clean_pv_map = sanitize_data(raw_pv_map)
    log.info("data sanitized")
    return clean_pv_map
