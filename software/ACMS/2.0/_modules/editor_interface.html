#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""Template Editor Network Profiler and PV Discovery Interface.

This module provides the backend discovery, extraction, and Channel Access (CA) 
sweeping capabilities required by the ACMS Template Editor sub-application. 

In the Recipe Management System (RMS) inherited by ACMS, recipes are dynamically 
generated from high-level operational templates. To assist operators in editing and 
authoring these templates, this module profiles EPICS Process Variables (PVs) by:

1. Extracting candidate PV lists from archive PostgreSQL databases or Display Builder
   (Phoebus ``.bob``) HMI files.
2. Executing concurrent multi-threaded EPICS CA network sweeps using a shared CA context loop.
3. Collecting requested PV fields (e.g., limits, units, current values).
4. Sanitizing telemetry payloads by converting IEEE floating-point ``NaN`` values into 
   JSON-compliant ``null`` literals to prevent serialization crashes.
5. Formatting and delivering structured PV metadata dictionaries to the Template Editor GUI.

Module Dependencies:
    - :mod:`epics`: PyEPICS Channel Access interface and thread context management.
    - :mod:`utilities.bob_parser`: Utility for parsing Phoebus ``.bob`` XML screens.
    - :mod:`utilities.db_query`: Query helper for fetching archived PV lists from SQL.

:author: luciamarock
"""

def load_configuration(config_path):
    """Load runtime parameters and configuration fields from a JSON resource file.

    :param config_path: File path to the JSON configuration file containing
        Channel Access settings, field specifications, and HMI display paths.
    :type config_path: str or Path

    :return: A 4-element tuple containing:
        
        - **CA_TIMEOUT** (*float*): Network connection timeout in seconds per PV.
        - **CA_PARALLELISM** (*int*): Maximum number of concurrent CA worker threads per batch.
        - **PV_FIELDS** (*list of str*): List of EPICS PV attributes to retrieve.
        - **HMI_FILES** (*dict*): Dictionary mapping display key identifiers to target ``.bob`` file paths.
    :rtype: tuple

    :raises SystemExit: If the configuration file is missing, contains invalid JSON syntax, 
        or lacks required keys.
    """
    try:
        with open(config_path, "r", encoding="utf-8") as config_file:
            config_data = json.load(config_file)
            
        return (
            config_data["CA_TIMEOUT"],
            config_data["CA_PARALLELISM"],
            config_data["PV_FIELDS"],
            config_data["HMI_FILES"]
        )
    except FileNotFoundError:
        log.error(f"CRITICAL ERROR: Configuration file not found at {config_path}")
        sys.exit(1)
    except KeyError as e:
        log.error(f"CRITICAL ERROR: Missing expected configuration key in JSON: {e}")
        sys.exit(1)
    except json.JSONDecodeError as e:
        log.error(f"CRITICAL ERROR: Invalid JSON syntax in config file: {e}")
        sys.exit(1)

def scan_pv_worker(name, results_dict, fields_to_get, timeout):
    """Worker thread target to extract EPICS metadata safely via a shared CA context.

    Attaches the executing thread to the global C library CA context (:data:`MAIN_CONTEXT`),
    attempts connection to the specified PV, reads requested attributes, and populates
    the shared results dictionary.

    :param name: The EPICS Process Variable name to connect to.
    :type name: str
    :param results_dict: Shared dictionary where scanned PV metadata is recorded.
    :type results_dict: dict
    :param fields_to_get: List of PV attributes/fields to query.
    :type fields_to_get: list of str
    :param timeout: Connection timeout threshold in seconds.
    :type timeout: float
    """
    log.debug(f'Scanning {name} ...')
    try:
        # Attach child thread to parent CA context loop
        epics.ca.attach_context(MAIN_CONTEXT)
        
        # Connect to the PV using timeout configurations
        pv = epics.pv.get_pv(name, connect=True, timeout=timeout)
        
        if pv.connected:
            fields = dict([(f, pv.__getattribute__(f)) for f in fields_to_get])
            results_dict[name] = fields
            pv.disconnect()
        else:
            log.warning(f"[-] Cannot connect to {name} (Timed out)")
            results_dict[name] = None
            
    except Exception as e:
        log.error(f"[-] Exception scanning {name}: {e}")
        results_dict[name] = None

def run_network_sweep(pv_names, parallelism, fields_to_get, timeout):
    """Orchestrate concurrent thread batches to profile online EPICS PV channels.

    Initializes the C-level libca library context and spawns batch groups of 
    :class:`epics.ca.CAThread` threads bounded by the ``parallelism`` limit to safely query
    the control network without overwhelming local socket descriptors.

    :param pv_names: List of PV names to profile over Channel Access.
    :type pv_names: list of str
    :param parallelism: Number of concurrent worker threads allowed per execution batch.
    :type parallelism: int
    :param fields_to_get: PV attributes to fetch from connected channels.
    :type fields_to_get: list of str
    :param timeout: Channel connection timeout in seconds per PV.
    :type timeout: float

    :return: A dictionary mapping PV names to attribute dicts (or ``None`` if unreachable).
    :rtype: dict
    """
    global MAIN_CONTEXT
    log.info(f"Connecting all {len(pv_names)} PVs safely using a shared CA Context...")
    
    # Initialize Channel Access and extract runtime context loop
    epics.ca.initialize_libca()
    MAIN_CONTEXT = epics.ca.current_context()

    pv_map = {}

    # Process batch loops matching the parallelism limits
    for i in range(0, len(pv_names), parallelism):
        chunk = pv_names[i:i + parallelism]
        active_batch = []
        
        for name in chunk:
            t = epics.ca.CAThread(
                target=scan_pv_worker, 
                args=(name, pv_map, fields_to_get, timeout)
            )
            active_batch.append(t)
            t.start()
            
        for t in active_batch:
            t.join()

    log.info("Done scanning network safely.")
    return pv_map

def sanitize_data(pv_map):
    """Sanitize scanned telemetry metadata by converting float NaN values to JSON nulls.

    EPICS floating-point PVs returning ``NaN`` (Not a Number) cause serialization errors when 
    converting dictionaries into standard JSON strings. This function replaces all ``NaN`` float 
    values with ``None`` (which maps to JSON ``null``).

    :param pv_map: The raw PV metadata dictionary produced by :func:`run_network_sweep`.
    :type pv_map: dict

    :return: The sanitized PV dictionary suitable for JSON serialization and GUI consumption.
    :rtype: dict
    """
    log.info("Sanitizing telemetry data fields...")
    for pv_name, fields in pv_map.items():
        if fields is not None:
            for key, value in fields.items():
                if isinstance(value, float) and math.isnan(value):
                    fields[key] = None  # Maps to a clean JSON 'null'
    return pv_map

def save_manifest(pv_map, output_path):
    """Write the sanitized PV metadata dictionary to a JSON file on disk.

    Creates parent directories if missing and formats the JSON output with 4-space 
    indentation and sorted keys.

    :param pv_map: Sanitized dictionary mapping PV names to field values.
    :type pv_map: dict
    :param output_path: Target file destination path.
    :type output_path: Path or str
    """
    output_path.parent.mkdir(parents=True, exist_ok=True)
    with open(output_path, 'w', encoding='utf-8') as json_file:
        json.dump(pv_map, json_file, sort_keys=True, indent=4, ensure_ascii=False)
    print(f"Manifest saved cleanly without crashes to: {output_path}")

def scan_archive_network(resources_dir):
    """Fetch all archived PVs from the SQL database and profile their live network status.

    This function serves as the primary data payload builder when launching the Template Editor 
    from the main ACMS CLI or GUI interface. It:

    1. Loads system settings from ``config.json``.
    2. Retrieves the complete list of target PVs from the archive SQL database.
    3. Executes a live Channel Access network sweep.
    4. Sanitizes field data and returns the completed payload dictionary.

    :param resources_dir: Base path to the application's resources directory containing ``config.json``.
    :type resources_dir: str

    :return: A sanitized map of all archived PV names and their live EPICS field attributes.
    :rtype: dict

    :raises SystemExit: If database connection or querying fails critically.
    """
    log.info("loading configuration file")
    config_file = resources_dir + "config.json"
    ca_timeout, ca_parallelism, pv_fields, hmi_files = load_configuration(config_file)
    log.info("configuration file loaded")
    log.info("fetching PV list from database")
    try:
        pv_names = get_list_of_pvs('localhost', 'lipac_mock', 'archive', 'rockoDD1o')
        pv_names.sort()
    except Exception as e:
        log.error(f"CRITICAL DATABASE ERROR: {e}")
        sys.exit(1)
    log.info("PV list fetched from database")
    log.info("running network sweep")
    raw_pv_map = run_network_sweep(pv_names, ca_parallelism, pv_fields, ca_timeout)
    log.info("network sweep completed")
    log.info("sanitizing data")
    clean_pv_map = sanitize_data(raw_pv_map)
    log.info("data sanitized")
    return clean_pv_map

def san_bob_file(resources_dir, key):
    """Extract PVs referenced inside a specific Phoebus Display (.bob) file and profile them.

    Used when an operator requests template editing for a specific sub-system HMI display. 
    It parses the targeted ``.bob`` XML screen file, extracts all embedded PV identifiers, and 
    profiles their live EPICS network states.

    :param resources_dir: Base path to the application's resources directory containing ``config.json``.
    :type resources_dir: str
    :param key: Lookup key in ``HMI_FILES`` corresponding to the target ``.bob`` display file.
    :type key: str

    :return: A sanitized dictionary of PVs extracted from the display file along with their 
        live field metadata.
    :rtype: dict

    :raises SystemExit: If the target ``.bob`` display file cannot be located or parsed.
    """
    log.info(f"Received Template Editor Request from {key}")
    log.info("loading configuration file")
    config_file = resources_dir + "config.json"
    ca_timeout, ca_parallelism, pv_fields, hmi_files = load_configuration(config_file)
    log.info("configuration file loaded")
    bob_file = hmi_files[key]
    log.info(f"fetching PV list from {bob_file}")
    try:
        pv_names = extract_pvs_from_bob(bob_file)
        pv_names.sort()
    except Exception as e:
        log.error(f"CRITICAL ERROR: {e}")
        sys.exit(1)
    log.info("PV list fetched from bob file")
    log.info("running network sweep")
    raw_pv_map = run_network_sweep(pv_names, ca_parallelism, pv_fields, ca_timeout)
    log.info("network sweep completed")
    log.info("sanitizing data")
    clean_pv_map = sanitize_data(raw_pv_map)
    log.info("data sanitized")
    return clean_pv_map
