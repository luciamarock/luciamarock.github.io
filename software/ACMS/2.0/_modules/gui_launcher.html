def load_json():
    """
    Loads JSON data either directly from the subprocess pipe via stdin
    or falls back to the local file cache with latin-1 encoding.
    """
    # Scenario A: Intercepting direct process pipeline input stream
    if "--stdin" in sys.argv:
        log.info("[GUI] Streaming telemetry dictionary from subprocess pipe...")
        try:
            raw_data = sys.stdin.read()
            return json.loads(raw_data)
        except Exception as e:
            log.error(f"[GUI] CRITICAL: Error parsing streamed stdin JSON data: {e}")
            return {}

    # Scenario B: Fallback manual execution using local file on disk
    else:
        log.info(f"[GUI] Standard launch. Reading local data snapshot from: {DEFAULT_PVINFO_PATH}")
        try:
            with open(DEFAULT_PVINFO_PATH, 'r', encoding='latin-1') as pv_data:
                pv_dict = json.load(pv_data)
            return pv_dict
        except FileNotFoundError:
            log.error(f"[GUI] Error: Local asset file not found at {DEFAULT_PVINFO_PATH}")
            # Maintaining your fallback samples for local engine debugging
            return {
                "default:sample1": "data1",
                "default:sample2": "data2"
            } 
        except Exception as e:
            log.error(f"[GUI] An unexpected error occurred during JSON file loading: {e}")
            return {}




if __name__ == '__main__':
    # 1. Harvest target data payload from the active channel
    data_to_load = load_json()
    if not data_to_load:  # Safety check for empty dictionaries or failed parses
        sys.exit(1)

    # 2. Start PySide2 process application context loop
    app = QApplication(sys.argv)
    
    # 3. Mount and pass the data directly into your layout application window class
    window = gui.JSONFilterApp(data_to_load)
    window.show()
    
    # Note the underscore: exec_() is required for PySide2/Qt5
    sys.exit(app.exec_())
