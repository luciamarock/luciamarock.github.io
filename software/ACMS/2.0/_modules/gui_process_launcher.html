def launch_template_editor(payload):
    """
    Launch the template editor GUI in an independent process and pass payload via stdin.

    Returns:
        subprocess.Popen: handle to the spawned process.

    Raises:
        FileNotFoundError: if gui launcher script is missing.
        RuntimeError: if the process cannot be started or payload transmission fails.
    """
    current_module_dir = Path(__file__).resolve().parent
    editor_script = current_module_dir / "gui_launcher.py"

    if not editor_script.exists():
        raise FileNotFoundError(
            "Template Editor GUI script not found at target: {}".format(editor_script)
        )

    try:
        proc = subprocess.Popen(
            [sys.executable, str(editor_script), "--stdin"],
            stdin=subprocess.PIPE,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            text=True,
            cwd=str(editor_script.parent),
            start_new_session=True,
        )
    except Exception as exc:
        raise RuntimeError(
            "Failed to spawn external GUI process: {}".format(exc)
        ) from exc

    try:
        if proc.stdin is None:
            raise RuntimeError("Spawned GUI process stdin is unavailable.")
        proc.stdin.write(json.dumps(payload, ensure_ascii=False))
        proc.stdin.close()
    except Exception as exc:
        raise RuntimeError(
            "Failed to send payload to GUI process: {}".format(exc)
        ) from exc

    log.info("Template Editor window successfully detached. pid=%s", proc.pid)
    return proc
