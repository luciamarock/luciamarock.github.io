#!/usr/bin/env python3
"""
External executor for the LCM automation loop (v0.5.2) — PUSH model, statement DSL.

LCM pushes jobs to the executor; the executor pushes results back to LCM. This executor
connects to real EPICS Process Variables (PVs) using pyepics.

- LISTENS for:
    - POST {this}/run {job_id, sequence, inputs, ...} -> runs the job
    - POST {this}/abort {job_id} -> cancels a job
- PUSHES to LCM:
    - POST {lcm}/api/automation/jobs/{id}/events -> set_field / push_timestep / log / progress / done

Standard-library + pyepics — runs with python3.

Start it and point LCM at it:
    python3 tools/lcm_executor.py --port 9100 --lcm http://localhost:8000
    LIPAC_EXECUTOR_URL=http://localhost:9100 uvicorn backend.main:app --port 8000

The job's ``sequence`` is the statement DSL (SPEC §4), e.g.:

    { "steps": [
        { "caput": { "pv": "LIPAC:PUMP:A:CMD", "value": 1 } },
        { "wait": { "seconds": 30 } },
        { "caget": { "pv": "LIPAC:GAUGE:A", "into": "gauge_a" } },
        { "push_field": { "code": "PRES_ACTUAL", "value": "${gauge_a}" } },
        { "push_timestep": { "comment": "after settle" } },
        { "done": { "ok": true } }
    ] }

DSL statements mapped to executor → LCM events:

    * ``caput`` → log / EPICS write (cached PV connection with timeout)
    * ``caget`` → log / EPICS read into a DSL variable
    * ``wait`` / ``wait_until`` → progress (abort-aware)
    * ``if`` / ``for_each`` / ``while`` → control flow (executor-internal)
    * ``push_field`` → set_field (LCM stores it in the job working state)
    * ``push_timestep`` → push_timestep (LCM commits a TIME-STEP)
    * ``log`` → log
    * ``done`` → done (terminal)
"""

def _lcm_post(path: str, body: dict) -> dict:
    data = json.dumps(body).encode()
    req = urllib.request.Request(LCM.rstrip("/") + "/api" + path, data=data, method="POST")
    req.add_header("Content-Type", "application/json")
    try:
        with urllib.request.urlopen(req, timeout=30) as r:
            raw = r.read()
            return json.loads(raw) if raw else {}
    except urllib.error.HTTPError as e:
        return {"_http_error": e.code, "_body": e.read().decode(errors="replace")}
    except Exception as exc:  # noqa: BLE001
        return {"_error": str(exc)}

class _Halt(Exception):
    """Unwinds the interpreter when the job is done or aborted."""




_ALLOWED = (ast.Expression, ast.BoolOp, ast.And, ast.Or, ast.UnaryOp, ast.Not,
            ast.USub, ast.UAdd, ast.BinOp, ast.Add, ast.Sub, ast.Mult, ast.Div,
            ast.Compare, ast.Eq, ast.NotEq, ast.Lt, ast.LtE, ast.Gt, ast.GtE,
            ast.Constant, ast.Name, ast.Load, ast.List, ast.Tuple)

def _ev(node, names: dict):
    if isinstance(node, ast.Expression):
        return _ev(node.body, names)
    if isinstance(node, ast.Constant):
        return node.value
    if isinstance(node, ast.Name):
        if node.id in names:
            return names[node.id]
        raise ValueError(f"unresolved variable '{node.id}'")
    if isinstance(node, ast.List):
        return [_ev(e, names) for e in node.elts]
    if isinstance(node, ast.BoolOp):
        vals = [_ev(v, names) for v in node.values]
        return all(vals) if isinstance(node.op, ast.And) else any(vals)
    if isinstance(node, ast.UnaryOp):
        v = _ev(node.operand, names)
        if isinstance(node.op, ast.Not): return not v
        if isinstance(node.op, ast.USub): return -v
        return +v
    if isinstance(node, ast.BinOp):
        a, b = _ev(node.left, names), _ev(node.right, names)
        if isinstance(node.op, ast.Add): return a + b
        if isinstance(node.op, ast.Sub): return a - b
        if isinstance(node.op, ast.Mult): return a * b
        if isinstance(node.op, ast.Div): return a / b
    if isinstance(node, ast.Compare):
        left = _ev(node.left, names)
        for op, comp in zip(node.ops, node.comparators):
            right = _ev(comp, names)
            ok = ({ast.Eq: left == right, ast.NotEq: left != right, ast.Lt: left < right,
                   ast.LtE: left <= right, ast.Gt: left > right, ast.GtE: left >= right}
                  .get(type(op)))
            if not ok:
                return False
            left = right
        return True
    raise ValueError(f"unsupported expression node {type(node).__name__}")

def _eval(expr: str, vars: dict):
    py = re.sub(r"\$\{(\w+)\}", r"\1", expr)          # ${x} → x
    tree = ast.parse(py, mode="eval")
    for n in ast.walk(tree):
        if not isinstance(n, _ALLOWED):
            raise ValueError(f"illegal expression element {type(n).__name__}")
    names = dict(vars); names["true"] = True; names["false"] = False
    return _ev(tree, names)

def _resolve(value, vars: dict):
    """A value may be a literal or a '${…}' expression string."""
    if isinstance(value, str) and "${" in value:
        return _eval(value, vars)
    return value

def _wait(jid: str, seconds: float) -> None:
    waited = 0.0
    while waited < seconds:
        if _aborted(jid):
            return
        time.sleep(min(0.25, seconds - waited)); waited += 0.25

def get_or_create_pv(pv_name: str, timeout: float = 5.0):
    """
    Checks if a PV instance is already cached. 
    If not, creates it, waits for connection, and caches it.
    """
    global connected_pvs
    if pv_name not in connected_pvs:
        log.info(f"Connecting to EPICS PV: {pv_name}")
        # Create the PV instance (auto_monitor=True is helpful for reads, but default works fine)
        pv_inst = epics.PV(pv_name)
        
        # Wait for connection to ensure the IOC is reachable
        if not pv_inst.wait_for_connection(timeout=timeout):
            log.error(f"Connection timeout or PV not found: {pv_name}")
            
        connected_pvs[pv_name] = pv_inst
        
    return connected_pvs[pv_name]

def run_steps(jid: str, steps: list, vars: dict) -> None:
    for st in steps:
        # Send Heartbeat to LCM and check for abort requests
        hb_res = _lcm_post("/automation/heartbeat", {"job_id": jid})
        abort_ids = hb_res.get("abort_job_ids", [])
        if jid in abort_ids:
            with _LOCK:
                _ABORTED.add(jid)
        if _aborted(jid):
            _event(jid, {"type": "done", "ok": False, "error": "aborted"})
            log.warning(f"   job {jid[:8]} ABORTED"); raise _Halt()
        if not isinstance(st, dict) or not st:
            continue
        kind = "if" if "if" in st else next((k for k in st if k in _KINDS), None)
        if kind is None:
            _event(jid, {"type": "log", "msg": f"unknown statement: {list(st)}"}); continue

        if kind == "caput":
            a = st["caput"]; val = _resolve(a.get("value"), vars)
            pv_name = a.get("pv")
            pv_inst = get_or_create_pv(pv_name)
            pv_inst.put(val, wait=True)
            _event(jid, {"type": "log", "msg": f"caput {pv_name} = {val}"})
            log.debug(f"   caput {pv_name} = {val}")
        elif kind == "caget":
            a = st["caget"]; pv_name = a.get("pv"); into = a.get("into")
            pv_inst = get_or_create_pv(pv_name)
            val = pv_inst.get()
            vars[into] = val
            _event(jid, {"type": "log", "msg": f"caget {pv_name} → {into}={val}"})
            log.debug(f"   caget {pv_name} → {into}={val}")
        elif kind == "wait":
            s = float(st["wait"].get("seconds", 0))
            _event(jid, {"type": "progress", "msg": f"waiting {s}s"}); log.debug(f"   wait {s}s"); _wait(jid, s)
        elif kind == "wait_until":
            a = st["wait_until"]; cond = a.get("condition", "false"); to = float(a.get("timeout_s", 0))
            _event(jid, {"type": "progress", "msg": f"wait_until {cond} (≤{to}s)"})
            waited = 0.0
            while waited < to and not _aborted(jid):
                try:
                    if _eval(cond, vars): break
                except Exception: break
                time.sleep(0.5); waited += 0.5
        elif kind == "if":
            cond = st.get("if"); branch = "then" if _eval(cond, vars) else "else"
            log.debug(f"   if {cond} → {branch}")
            run_steps(jid, st.get(branch) or [], vars)
        elif kind == "for_each":
            a = st["for_each"]; var = a.get("var")
            seq = _resolve(a.get("in"), vars) if isinstance(a.get("in"), str) else a.get("in", [])
            for item in seq or []:
                vars[var] = item
                run_steps(jid, a.get("do") or [], vars)
        elif kind == "while":
            a = st["while"]; cond = a.get("condition", "false"); mx = int(a.get("max_iterations", 0))
            it = 0
            while it < mx and not _aborted(jid) and _eval(cond, vars):
                run_steps(jid, a.get("do") or [], vars); it += 1
            log.debug(f"   while {cond} ran {it}/{mx}")
        elif kind in ("push_field", "set_field"):        # set_field = accepted alias
            a = st.get("push_field") or st.get("set_field") or {}
            code = a.get("code"); val = _resolve(a.get("value"), vars)
            vars[code] = val                              # track locally so ${code} sees it later
            _event(jid, {"type": "set_field", "code": code, "value": val})
            log.debug(f"   {kind} {code} = {val}")
        elif kind == "push_timestep":
            r = _event(jid, {"type": "push_timestep", "comment": st["push_timestep"].get("comment")})
            log.debug(f"   push_timestep '{st['push_timestep'].get('comment')}' → {r}")
        elif kind == "log":
            _event(jid, {"type": "log", "msg": st["log"].get("msg", "")})
        elif kind == "done":
            a = st["done"] or {}
            _event(jid, {"type": "done", "ok": bool(a.get("ok", True)), "error": a.get("error")})
            log.info(f"   job {jid[:8]} done"); raise _Halt()




_KINDS = ("caput", "caget", "wait", "wait_until", "if", "for_each", "while",
          "push_field", "set_field", "push_timestep", "log", "done")

def run_job(job: dict) -> None:
    jid = job["job_id"]
    seq = job.get("sequence") or {}
    steps = seq.get("steps", []) if isinstance(seq, dict) else []
    log.info(f"→ RUN job {jid[:8]}  ({len(steps)} steps, inputs={job.get('inputs')})")
    vars: dict = dict(job.get("inputs") or {})
    try:
        run_steps(jid, steps, vars)
        _event(jid, {"type": "done", "ok": True})          # auto-done if the sequence didn't
        log.info(f"   job {jid[:8]} done (auto)")
    except _Halt:
        pass
    except Exception as exc:  # noqa: BLE001 — report DSL/runtime errors as a failed job
        _event(jid, {"type": "done", "ok": False, "error": f"executor error: {exc}"})
        log.info(f"   job {jid[:8]} FAILED: {exc}")
    finally:
        with _LOCK:
            _ABORTED.discard(jid)
        for k in [k for k in _CAGET_N if k[0] == jid]:
            _CAGET_N.pop(k, None)

    def _reply(self, code: int, body: dict) -> None:
        data = json.dumps(body).encode()
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()

    def do_POST(self):  # noqa: N802
        body = self._read()
        if self.path.rstrip("/") == "/run":
            jid = body.get("job_id")
            if not jid:
                return self._reply(400, {"error": "missing job_id"})
            log.info(f"← POST /run  job {jid[:8]}")
            threading.Thread(target=run_job, args=(body,), daemon=True).start()
            return self._reply(202, {"accepted": True, "job_id": jid})
        if self.path.rstrip("/") == "/abort":
            jid = body.get("job_id")
            with _LOCK:
                if jid:
                    _ABORTED.add(jid)
            log.info(f"← POST /abort  job {str(jid)[:8]}")
            return self._reply(200, {"aborting": jid})
        self._reply(404, {"error": "not found"})

def main() -> None:
    global LCM
    ap = argparse.ArgumentParser()
    ap.add_argument("--port", type=int, default=9100, help="port this mock executor listens on")
    ap.add_argument("--lcm", default="http://localhost:8000", help="LCM base URL to push results to")
    args = ap.parse_args()
    LCM = args.lcm
    srv = ThreadingHTTPServer(("0.0.0.0", args.port), Handler)
    print(f"Mock executor listening on http://localhost:{args.port}  (pushes results to {LCM})")
    print(f"→ set LCM to reach it:  LIPAC_EXECUTOR_URL=http://localhost:{args.port}")
    try:
        srv.serve_forever()
    except KeyboardInterrupt:
        print("\nstopped.")




if __name__ == "__main__":
    main()
