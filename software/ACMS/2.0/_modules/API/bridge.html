def get_system_status() -> dict:
    """Collects status across all registered workflows."""
    status_report = {}
    for name, instance in WI.registered_workflows.items():
        # Map or trigger status check method
        status_report[name] = getattr(instance, "status", "running")
    return status_report

def dispatch_command(workflow_name: str, command: str, payload: dict = None) -> dict:
    """Routes an incoming command to a specific registered workflow."""
    if workflow_name not in WI.registered_workflows:
        return {"status": "error", "message": f"Workflow '{workflow_name}' not registered."}
    
    workflow_instance = WI.registered_workflows[workflow_name]
    
    # Route commands matching CLI capabilities
    if command == "status":
        if hasattr(workflow_instance, "talk"):
            workflow_instance.talk()
        return {"status": "ok", "message": f"Status triggered for {workflow_name}"}
    
    elif command == "stop":
        if hasattr(workflow_instance, "stop"):
            workflow_instance.stop()
        return {"status": "ok", "message": f"Stopped {workflow_name}"}
    
    # Custom PV readback / procedure trigger
    elif hasattr(workflow_instance, command):
        method = getattr(workflow_instance, command)
        result = method(payload) if payload else method()
        return {"status": "ok", "result": result}
    
    else:
        return {"status": "error", "message": f"Command '{command}' not supported."}
