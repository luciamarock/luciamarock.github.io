    def _calculate_readback_PV(self,PVname):
        parts = PVname.split("-")
        readback_name = ""
        for i in range(len(parts)-1):
            readback_name += parts[i] + "-"
        readback_name += "RB"
        return readback_name

    def _writePVs(self,step_dict):
        for PVname in step_dict.keys():
            if PVname not in self._known_PVs.keys():
                self._known_PVs[PVname] = {}
                currentPV = epics.PV(PVname)
                if not currentPV.wait_for_connection():
                    logger.error("{} failed to connect".format(currentPV))
                self._known_PVs[PVname]["PVinstance"] = currentPV
            else:
                currentPV = self._known_PVs[PVname]["PVinstance"]
            currentPV.put(step_dict[PVname])
            self._known_PVs[PVname]["status"] = "unverified"

    def _verify(self):
        for key in self._known_PVs.keys():
            if self._known_PVs[key]["status"] == "unverified" and "COMMIT" not in key:
                currentPV =  self._known_PVs[key]["PVinstance"]
                #readback_name = self._calculate_readback_PV(key)
                readback_name = key
                readbackPV = epics.PV(readback_name)
                if not readbackPV.wait_for_connection():
                    logger.error("{} failed to connect".format(readback_name))
                current_value = currentPV.get()
                readback_value = readbackPV.get()
                #logger.info("verifying {} = {}; readback = {}".format(key,current_value,readback_value))
                if current_value != readback_value:
                    logger.info("{}.value != {}.value; {} != {}".format(readbackPV,currentPV,readback_value,current_value))
                    return False 
                else:
                    self._known_PVs[key]["status"] == "verified"
                readbackPV.disconnect()
        return True

    def compare_with_current(self,recipename):
        if not '.' in recipename:
            recipename += '.json'
        filename = self._recipes_base_dir + recipename
        if os.path.isfile(filename):
            with open(filename, 'r') as recipefile:
                data = json.load(recipefile)
        else:
            #print(f"recipe {filename} does not exist")
            return
        if data:
            for key in data:
                target_value = data[key]
                current_pv = epics.PV(key)
                if not current_pv.wait_for_connection():
                    logger.error("unable to connect to PV {}".format(key))
                else:
                    current_value = current_pv.get()
                current_pv.disconnect()
                if target_value != current_value:
                    print(f"value {target_value} of {key} in recipe {recipename} is different from current value {current_value}")

    def loadRecipe(self,recipename):
        result = "success"
        if not '.' in recipename:
            recipename += '.json'
        filename = self._recipes_base_dir + recipename
        start_time = time.time()
        if os.path.isfile(filename):
            with open(filename, 'r') as recipefile:
                data = json.load(recipefile)
            try:
                self._writePVs(data)
            except Exception as e:
                logger.error(e)
                result = "failure"
            if result == "success":
                elapsed_time = time.time() - start_time
                logger.info('Elapsed time in putting settings is %f', elapsed_time)
        else:
            logger.error("recipe {} does not exist".format(filename))
            result = "failure"
        return result
