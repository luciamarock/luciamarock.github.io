def run_bash_script(script_path):
    time.sleep(2)
    try:
        # Use subprocess to run the Bash script
        process = subprocess.run(["bash", script_path],stdout=subprocess.PIPE, stderr=subprocess.PIPE, check=True)
        log.info("Bash script executed successfully.")
        if process.stdout:
            log.info(process.stdout)  # Log stdout to logger
        if process.stderr:
            log.info(process.stderr)  # Log stderr to logger
    except subprocess.CalledProcessError as e:
        # Handle any errors if the script fails
        log.error("Error executing Bash script: {}".format(e))

    def _create_or_update_PVs(self,recipe_data):
        for parameter, value in recipe_data.items():
            pv_name = self._prefix + parameter
            if pv_name not in self._PVs.keys():
                self._PVs[pv_name] = {}
                current_pv = epics.PV(pv_name)
                if not current_pv.wait_for_connection():
                    log.error("{} failed to connect".format(current_pv))
                self._PVs[pv_name]["PV"] = current_pv
                self._PVs[pv_name]["value"] = value
            else:
                self._PVs[pv_name]["value"] = value

    def verifyRecipe(self,recipename):
        result = "success"
        for key in self._PVs.keys():
            current_pv = self._PVs[key]["PV"]
            current_value = current_pv.get()
            recipe_value = self._PVs[key]["value"]
            if isinstance(recipe_value,list):
                current_casted = current_value.tolist()
                if current_casted != recipe_value:
                    log.info("{}.value != requested value; {} != {}".format(current_pv,current_value,recipe_value))
                    result = "failure"
            elif isinstance(recipe_value,str):
                current_casted = current_pv.get(as_string = True)
                if current_casted != recipe_value:
                    log.info("{}.value != requested value; {} != {}".format(current_pv,current_value,recipe_value))
                    result = "failure"
            elif isinstance(recipe_value,int):
                current_casted = int(current_value)
                if current_casted != recipe_value:
                    log.info("{}.value != requested value; {} != {}".format(current_pv,current_value,recipe_value))
                    result = "failure"
            elif current_pv.type == "time_enum":
                current_casted = current_pv.enum_strs[current_value]
                if current_casted != recipe_value:
                    log.info("{}.value != requested value; {} != {}".format(current_pv,current_value,recipe_value))
                    result = "failure"
            else:
                if current_value != recipe_value:
                    log.info("{}.value != requested value; {} != {}".format(current_pv,current_value,recipe_value))
                    result = "failure"

        return result
