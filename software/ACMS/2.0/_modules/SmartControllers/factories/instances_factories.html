def xml_to_dict(element):
    data = {}
    for child in element:
        if len(child) > 0 or child.attrib:
            data[child.tag] = xml_to_dict(child)
        else:
            data[child.tag] = child.text
    return data

class common_instance():
    """Generic class for representing a Physical component.
    
    :param attributes: initially read from the xml configuration file they become here real class attibutes  
    
    """
    def __init__(self,attributes,root,itstype):
        for attribute in attributes.keys():
            temp = str(attributes[attribute]).replace("\n","")
            string = "self." + attribute + " = " + temp.strip()
            exec(string)
        self._type = itstype
        self._init_auto_load = True
        self._init_load = True
        self._init_save = True
        self._init_combo = True
        self._init_compare_with_current = True
        self._PVs = {}
        self._feedback = None
        self._recipes = {"status":"clean","loaded":None,"to_load":None,"mapping":None}
        self._recipe_template = {"status":"clean","json":None,"desired_name":"Empty"}
        self._save_path = None
        self._verification_timeout = 1
        self._verification_start = None
        try:
            plc_element = root.xpath('//PLC')[0]
            self._PLC = xml_to_dict(plc_element)
        except:
            log.error("no PLC node defined for this instance")
            error("instance {} of type {} does not have PLC actions defined".format(self.instance, self.family))
            self._PLC = {}
        try:
            codac_element = root.xpath('//CODAC')[0]
            self._CODAC = xml_to_dict(codac_element)
        except:
            log.error("no CODAC node defined for this instance")
            error("instance {} of type {} does not have CODAC actions defined".format(self.instance, self.family))
            self._CODAC = {}
        try:
            interface_element = root.xpath('//Interface')[0]
            self._Interface = xml_to_dict(interface_element)
        except:
            log.error("{}: no Interface node defined for this instance".format(self.instance))
            self._Interface = {}

    def monitor_actions(self):
        #for key in self._recipes:
            #print("{} --> {} = {}".format(self.instance, key,self._recipes[key]))
        recipe_to_load = self._recipes["to_load"]
        if recipe_to_load and self._recipes["status"] == "compare":
            self._compare_PV.put(0)
            self._interface_class.compare_with_current(recipe_to_load["name"])
            self._recipes["status"] = "clean"
        if recipe_to_load and self._recipes["status"] == "pending":
            if self._verification_start == None:
                self._verification_start = time.time()
            result = self._interface_class.verifyRecipe(recipe_to_load["name"])
            if result != "success" and time.time() - self._verification_start > self._verification_timeout:
                self._recipes["status"] = "error"
                error("failed to load {} on {}".format(recipe_to_load["name"],self.instance))
                log.error("failed to load {} on {}".format(recipe_to_load["name"],self.instance))
            elif result == "success":
                if self._feedback and self._recipes["mapping"] is not None:
                    self._feedback.put(float(self._recipes["mapping"]))
                self._recipes["to_load"] = None
                self._recipes["loaded"] = recipe_to_load
                self._recipes["status"] = "clean"
                info("{} successfully loaded on {}".format(recipe_to_load["name"],self.instance))
                log.info("{} successfully loaded on {}".format(recipe_to_load["name"],self.instance))
        if recipe_to_load and self._recipes["status"] == "dirty":
            self._combo_PV.put(0)
            result = self._interface_class.loadRecipe(recipe_to_load["name"])
            if result == "success":
                self._recipes["status"] = "pending" # pending verification
            else:
                log.error("could not upload requested recipe {} to instance {}".format(recipe_to_load["name"],self.instance))
                error("could not upload requested recipe {} to instance {}".format(recipe_to_load["name"],self.instance))
                self._recipes["status"] = "error"
        if self._recipe_template["status"] == "dirty" and self._recipe_template["json"]:
            recipe_object = {}
            if "prefix" in self._Interface.keys():
                for key in self._recipe_template["json"].keys():
                    pv_to_get = self._Interface["prefix"] + key
                    current_pv = epics.PV(pv_to_get)
                    if not current_pv.wait_for_connection():
                        error("unable to connect to PV {}".format(pv_to_get))
                        log.error("unable to connect to PV {}".format(pv_to_get))
                    else:
                        current_value = current_pv.get()
                        if isinstance(current_value,np.ndarray):
                            current_value = current_value.tolist()
                        recipe_object[key] = current_value
                    current_pv.disconnect()
            else:
                for pv_to_get in self._recipe_template["json"].keys():
                    current_pv = epics.PV(pv_to_get)
                    if not current_pv.wait_for_connection():
                        error("unable to connect to PV {}".format(pv_to_get))
                        log.error("unable to connect to PV {}".format(pv_to_get))
                    else:
                        current_value = current_pv.get()
                        recipe_object[pv_to_get] = current_value
                    current_pv.disconnect()
            if self._save_path and os.path.isdir(self._save_path):
                #TODO do the validation here, use CI.validators
                if not self._save_path.endswith(os.path.sep):
                    self._save_path = self._save_path + os.path.sep
                filename = self._save_path + self._recipe_template["desired_name"] + ".json"
                with open(filename, 'w') as recipe_file:
                    json.dump(recipe_object, recipe_file, indent=4)
                info("successfully saved recipe {}".format(filename))
                log.info("successfully saved recipe {}".format(filename))
            else:
                error("can't save desired recipe, path {} does not exist".format(self._save_path))
                log.error("can't save desired recipe, path {} does not exist".format(self._save_path))
            self._set_combo_list()
            self._recipe_template["status"] = "clean"
            self._save_PV.put("")

    def start(self):
        if self._Interface:
            if "path" not in self._Interface.keys():
                log.error("{} does not have specified path to interface class".format(self.instance))
                error("{} does not have specified path to interface class".format(self.instance))
                self._interface_class = None
            elif not os.path.exists(self._Interface["path"]):
                log.error("path to interface class not found for instance {}".format(self.instance))
                error("path to interface class not found for instance {}".format(self.instance))
                self._interface_class = None
            else:
                sys.path.append(os.path.join(os.path.dirname(__file__), self._Interface["path"]))
                module_name = self.family + "_Interface"
                try:
                    module = importlib.import_module(module_name)
                    self._interface_class = module.create(self._Interface)
                except:
                    log.error("interface class not found for instance {}".format(self.instance))
                    error("interface class not found for instance {}".format(self.instance))
                    self._interface_class = None
        else:
            log.warning("skipping {} because it does not have a class interface specified".format(self.instance))
            warning("skipping {} because it does not have a class interface specified".format(self.instance))
            self._interface_class = None
            return
        if self._CODAC:
            if "combo" in self._CODAC.keys():
                if "trigger_PV" in self._CODAC["combo"].keys():
                    pv_name = self._CODAC["combo"]["trigger_PV"]
                    self._combo_PV = epics.PV(pv_name)
                    is_connected = self._combo_PV.wait_for_connection(timeout=1)
                    if is_connected:
                        self._combo_PV.add_callback(self._load_action)
                    else:
                        log.error("{} does not exist".format(pv_name))
                        error("{} does not exist".format(pv_name))
                if "combo_selection" in self._CODAC["combo"].keys():
                    pv_name = self._CODAC["combo"]["combo_selection"]
                    self._combo_selection = epics.PV(pv_name)
                    is_connected = self._combo_selection.wait_for_connection(timeout=1)
                    if is_connected:
                        self._combo_selection.add_callback(self._combo_action)
                    else:
                        log.error("{} does not exist".format(pv_name))
                        error("{} does not exist".format(pv_name))
                if "compare_PV" in self._CODAC["combo"].keys():
                    pv_name = self._CODAC["combo"]["compare_PV"]
                    self._compare_PV = epics.PV(pv_name)
                    is_connected = self._compare_PV.wait_for_connection(timeout=1)
                    if is_connected:
                        self._compare_PV.add_callback(self._compare_with_current_action)
                    else:
                        log.error("{} does not exist".format(pv_name))
                        error("{} does not exist".format(pv_name))
            if "save" in self._CODAC.keys():
                if "trigger_PV" in self._CODAC["save"].keys():
                    pv_name = self._CODAC["save"]["trigger_PV"]
                    self._save_PV = epics.PV(pv_name) 
                    is_connected = self._save_PV.wait_for_connection(timeout=1)
                    if is_connected:
                        self._save_PV.add_callback(self._save_action)
                    else:
                        log.error("{} does not exist".format(pv_name))
                        error("{} does not exist".format(pv_name))
                if "path_to_validate" in self._CODAC["save"].keys():
                    self._save_path = self._CODAC["save"]["path_to_validate"]
                if "combo_list" in self._CODAC["save"].keys():
                    pv_name = self._CODAC["save"]["combo_list"]
                    self._combo_list = epics.PV(pv_name)
                    is_connected = self._combo_list.wait_for_connection(timeout=1)
                    if not is_connected:
                        log.error("{} does not exist".format(pv_name))
                        error("{} does not exist".format(pv_name))
                        self._combo_list = None
        if self._PLC:
            if "load" in self._PLC.keys():
                if "trigger_PV" in self._PLC["load"].keys():
                    pv_name = self._PLC["load"]["trigger_PV"]
                    self._load_PV = epics.PV(pv_name) 
                    is_connected = self._load_PV.wait_for_connection(timeout=1)
                    if is_connected:
                        self._load_PV.add_callback(self._auto_load_action)
                    else:
                        log.error("{} does not exist".format(pv_name))
                        error("{} does not exist".format(pv_name))
                if "wait" in self._PLC["load"].keys():
                    self._verification_timeout = float(self._PLC["load"]["wait"])
                if "feedback_PV" in self._PLC["load"].keys():
                    fb_pv_name = self._PLC["load"]["feedback_PV"]
                    self._feedback = epics.PV(fb_pv_name) 
                    is_connected = self._feedback.wait_for_connection(timeout=1)
                    if is_connected:
                        log.info("{} connected to PLC feedback PV {}".format(self.instance,fb_pv_name))
                    else:
                        log.error("{} does not exist".format(fb_pv_name))
                        error("{} does not exist".format(fb_pv_name))
        self._set_combo_list()

    def _combo_action(self,pvname=None, value=None, char_value=None, **kw):
        if self._init_combo:
            self._init_combo = False
        else:
            found = False
            log.info("triggered combo action for instance {}, value = {}".format(self.instance,value))
            for key in self._auto_mode_settings.keys():
                if value == self._auto_mode_settings[key]["name"]:
                    recipe_dict = self._auto_mode_settings[key]
                    self._recipes["to_load"] = recipe_dict
                    self._recipes["mapping"] = key
                    found = True 
                    break
            if not found:
                recipe_dict = {}
                recipe_dict["device"] = self.instance
                recipe_dict["priority"] = None
                recipe_dict["name"] = value
                self._recipes["to_load"] = recipe_dict
                self._recipes["mapping"] = None

    def _save_action(self,pvname=None, value=None, char_value=None, **kw):
        if self._init_save:
            self._init_save = False
        else:
            if not self._recipe_template["json"]:
                current_settings = self._type.get_settings()
                resources_dir = current_settings["resources_directory"]
                if not resources_dir.endswith(os.path.sep):
                    resources_dir = resources_dir + os.path.sep 
                template_dir = resources_dir + "templates" + os.path.sep 
                recipe_template_file = template_dir + self.instance + "_recipe_template.json"
                if not os.path.exists(recipe_template_file):
                    error("cannot save because the template file {} does not exist".format(recipe_template_file))
                    log.error("cannot save because the template file {} does not exist".format(recipe_template_file))
                else:
                    with open(recipe_template_file, 'r') as template:
                        self._recipe_template["json"] = json.load(template)
            if value:
                self._recipe_template["status"] = "dirty"
                self._recipe_template["desired_name"] = value

def create_instance(root,controller_instance_name,registered_types):
    """Called by controllersinterface.parse_xml_instances function
    
    This function finds inside the root (xml root node) object all the required information. 
    In particular it first retrieves the "family" of the curren instance, using the family it retrieves the 
    Type software object from the parameter registered_types and from there it collects the instance_attributes
    (these attributes are the one the user defines in the xml file of the Controller's Types); it means that those 
    attributes are considered "mandatory" and must be found in the instance configuration file. \n 
    Thus it then proceeds to find al the required and expected attributes inside the root element. It collects all the information 
    in the attributes dictionary. 
    
    :param root: The xml root node 
    :param controller_instance_name: The name of the Controller's Instance (same as the nema of the xml configuration file)
    :param registered_types: module level dictionary :obj:`controllersInterface.registered_types`
    
    :returns: The instanciated software object representing the physical component
    
    """
    log.info("generating {} controller instance".format(controller_instance_name))
    family = None
    for familyNode in root.xpath(".//family"):
        family = familyNode.text
        break # taking only the first family attribute in case there is more then one
    if family is not None:
        if family in registered_types.keys():
            instance_attributes = registered_types[family]._arguments
            attributes = {}
            attributes["family"] = "\"" + family + "\""
            attributes["instance"] = "\"" + controller_instance_name + "\""
            for key in instance_attributes: 
                try:
                    temp = float(root.xpath(".//" + key)[0].text)
                    attributes[key] = root.xpath(".//" + key)[0].text
                except: 
                    if root.xpath(".//" + key):
                        node_text = root.xpath(".//" + key)[0].text
                        if node_text is not None:
                            temp = "\"" + node_text + "\""
                        else:
                            temp = "\"" + "\""
                        attributes[key] = temp 
                    else:
                        log.warning("attribute {} not present in the instance definition file".format(key))
            # eventually extend the attributes with instance-specific <attribute> nodes --> see function _additional_attributes
            currentInstance = common_instance(attributes,root,registered_types[family])
            if registered_types[family].register_instance(controller_instance_name):
                return currentInstance
            else:
                return None
        else:
            log.error("specified family is not defined")
            return None
    else:
        log.error("family not specified")
        return None
