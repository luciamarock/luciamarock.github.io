class common_type():
    """Generic class for representing the Controller's types 
    
    It creates class attributes on runtime and takes notes of the attributes to be expected during Controller's instances creation.
    
    """
    def __init__(self,attributes,arguments):
        for attribute in attributes.keys():
            string = "self." + attribute + " = " + str(attributes[attribute]) 
            exec(string)
        self._arguments = arguments
        self._registered_instances = []

    def register_instance(self,instance_name):
        """Called during Controller's Instances creation 
        
        It simply register the unique name of each Controller's Instance belonging to this type.
        
        """
        if instance_name not in self._registered_instances:
            self._registered_instances.append(instance_name)
            return True
        else:
            log.error("cannot register {}, an istance with same name already exist".format(instance_name))
            return False

def create_type(root,name):   
    """Function responsible for instatiating the generic class for the controller's types. 
    
    This function is called by :obj:`controllersinterface.parse_xml_types` function. \n
    Starting from xml root node this function assembles the attributes dictionary and the arguments list. \n
    These entities are then passed to the class common_type during instantiation. 
    
    :param root: The root node (as read from .xml file)
    :param name: The name of the xml file, this name becomes the nme of the Object Class 
    
    :return: The Controller's type object class with its name ready to be registered in the controllerinterface.registered_types dictionary
    
    """
    controller_type_name = name.split(".")[0]
    log.info("generating {} controller type".format(controller_type_name))
    attributes = {}
    attributes["family"] = "\"" + controller_type_name + "\""
    for attributeNode in root.xpath(".//attribute"):
        for key in attributeNode.keys():
            try:
                temp = float(attributeNode.get(key))
                attributes[key] = attributeNode.get(key)
            except:
                temp = "\"" + attributeNode.get(key) + "\""
                attributes[key] = temp
            break # each node will carry only one mandatory attribute
    arguments = []
    for argumentNode in root.xpath(".//argument"):
        try:
            arguments.append(argumentNode.get("name"))
        except:
            log.error("xml attributes of the argument node bad formatted")
    currentInstance = common_type(attributes,arguments)
    return controller_type_name, currentInstance
