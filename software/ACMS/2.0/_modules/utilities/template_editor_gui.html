class JSONFilterApp(QMainWindow):
    def __init__(self, json_data):
        super(JSONFilterApp, self).__init__() # Explicit super call for PySide2 compatibility
        self.setWindowTitle("PySide2 Cascading JSON Filter & Export")
        self.setFixedSize(QSize(850, 650)) 
        self.data = json_data
        if not self.data:
            print("Application closing: No data loaded.")
            sys.exit(1)

        self.max_depth = self._get_max_depth()
        self.comboboxes = []
        self.selected_items = {} 
        self._setup_ui()

    def _get_max_depth(self):
        max_d = 0
        for key in self.data.keys():
            max_d = max(max_d, len(key.split(':')))
        return max_d

    def _setup_ui(self):
        central_widget = QWidget()
        self.setCentralWidget(central_widget)
        main_layout = QVBoxLayout(central_widget)
        
        # --- Top Half: Filters and Controls ---
        filter_section = QWidget()
        filter_layout = QVBoxLayout(filter_section)
        
        combo_layout = QHBoxLayout()
        for i in range(self.max_depth):
            label = QLabel(f"Chunk {i+1}:")
            combo_layout.addWidget(label)
            
            combo = QComboBox()
            combo.setEditable(False)
            combo.setEnabled(False) 
            combo.setProperty("index", i) 
            # In PySide2, lambda works the same way
            combo.currentIndexChanged.connect(lambda index, cb=combo: self._update_filters(cb))
            
            combo_layout.addWidget(combo)
            self.comboboxes.append(combo)

        combo_layout.addStretch(1)
        filter_layout.addLayout(combo_layout)

        result_controls_layout = QHBoxLayout()
        self.result_label = QLabel("Full Key: ")
        self.result_label.setWordWrap(True)
        self.result_label.setStyleSheet("font-size: 12pt; font-weight: bold;")
        self.result_label.setSizePolicy(QSizePolicy.Expanding, QSizePolicy.Preferred)
        result_controls_layout.addWidget(self.result_label)
        
        self.add_button = QPushButton("Add to List")
        self.add_button.setEnabled(False)
        self.add_button.clicked.connect(self._add_to_list)
        self.add_button.setFixedWidth(150)
        result_controls_layout.addWidget(self.add_button)

        filter_layout.addLayout(result_controls_layout)
        main_layout.addWidget(filter_section)
        
        # --- Middle Half: Selection List ---
        selection_label = QLabel("Selected Keys for Export:")
        main_layout.addWidget(selection_label)
        
        self.list_widget = QListWidget()
        self.list_widget.setSelectionMode(QListWidget.ExtendedSelection)
        main_layout.addWidget(self.list_widget)

        # --- Bottom Half: Export Button ---
        export_layout = QHBoxLayout()
        self.remove_button = QPushButton("Remove Selected")
        self.remove_button.clicked.connect(self._remove_from_list)
        export_layout.addWidget(self.remove_button)
        export_layout.addStretch(1)

        self.export_button = QPushButton("Export Selected Keys")
        self.export_button.clicked.connect(self._export_to_file)
        export_layout.addWidget(self.export_button)
        
        main_layout.addLayout(export_layout)
        self._initialize_first_filter()

    def _initialize_first_filter(self):
        first_chunks = set()
        for key in self.data.keys():
            first_chunks.add(key.split(':')[0])
            
        self.comboboxes[0].addItems(sorted(list(first_chunks)))
        self.comboboxes[0].insertItem(0, "— Select —") 
        self.comboboxes[0].setCurrentIndex(0)
        self.comboboxes[0].setEnabled(True)

    def _update_filters(self, current_combo):
        current_index = current_combo.property("index")
        current_path = []
        for i in range(current_index + 1):
            selection = self.comboboxes[i].currentText()
            if selection in ["— Select —", ""]:
                break
            current_path.append(selection)

        current_prefix = ":".join(current_path)

        for i in range(current_index + 1, self.max_depth):
            self.comboboxes[i].clear()
            self.comboboxes[i].setEnabled(False)
            
        next_index = current_index + 1
        if next_index < self.max_depth:
            next_chunks = set()
            if current_path:
                for full_key in self.data.keys():
                    if full_key.startswith(current_prefix + ':'):
                        parts = full_key.split(':')
                        if len(parts) > next_index:
                            next_chunks.add(parts[next_index])

                if next_chunks:
                    self.comboboxes[next_index].clear()
                    self.comboboxes[next_index].addItems(sorted(list(next_chunks)))
                    self.comboboxes[next_index].insertItem(0, "— Select —")
                    self.comboboxes[next_index].setCurrentIndex(0)
                    self.comboboxes[next_index].setEnabled(True)

        self._update_result_label(current_prefix)

    def _update_result_label(self, full_key_prefix):
        if full_key_prefix in self.data:
            value = self.data[full_key_prefix]
            self.result_label.setText(f"Full Key: **{full_key_prefix}**\nValue: **{value}**")
            self.add_button.setEnabled(True)
        else:
            self.result_label.setText(f"Current Path: {full_key_prefix}")
            self.add_button.setEnabled(False)

    def _add_to_list(self):
        current_path = [cb.currentText() for cb in self.comboboxes if cb.currentText() not in ["— Select —", ""]]
        full_key = ":".join(current_path)

        if full_key and full_key in self.data:
            value = self.data[full_key]
            if full_key not in self.selected_items:
                self.selected_items[full_key] = value
                list_item_text = f"{full_key} -> {value}"
                self.list_widget.addItem(list_item_text)
            else:
                QMessageBox.information(self, "Item Already Added", f"The key '{full_key}' is already in the list.")

    def _remove_from_list(self):
        items_to_remove = self.list_widget.selectedItems()
        if not items_to_remove:
            return
        for item in items_to_remove:
            item_text = item.text()
            key_to_remove = item_text.split(" -> ")[0]
            if key_to_remove in self.selected_items:
                del self.selected_items[key_to_remove]
            self.list_widget.takeItem(self.list_widget.row(item))

    def _export_to_file(self):
        if not self.selected_items:
            QMessageBox.warning(self, "Export Failed", "The selected list is empty.")
            return

        filename, _ = QFileDialog.getSaveFileName(
            self, 
            "Save Exported Data", 
            "exported_data.json", 
            "JSON Files (*.json);;Text Files (*.txt);;All Files (*)"
        )

        if filename:
            try:
                with open(filename, 'w') as f:
                    json.dump(self.selected_items, f, indent=4)
                QMessageBox.information(self, "Export Successful", f"Exported {len(self.selected_items)} keys.")
            except Exception as e:
                QMessageBox.critical(self, "Export Error", f"Failed to save file:\n{e}")
