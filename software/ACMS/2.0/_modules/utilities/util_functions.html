"""
Utility Functions Module
========================

Provides shared global state containers and helper utilities accessible across all system packages.

This module acts as the centralized storage for global application settings parsed from 
configuration files (`settings.ini`), as well as runtime state objects such as 
:class:`UserIntention` instances that manage asynchronous command states (e.g., ``IDLE``, 
``PAUSE``, ``ABORT``) for background workflow workers.
"""

import time
import logging 
log = logging.getLogger(__name__)

log.debug("imported util_functions")

#: Global settings dictionary containing configuration parameters and workspace directories.
settings = {
    "resources_directory": "resources",
    "workspace_dir": "workspace"
}

#: Global registry mapping workflow names to their active :class:`UserIntention` instances.

class UserIntention:
    """Generic container representing user control intentions, commands, and target parameters.

    Provides state management for interactive operator commands and handles thread locking 
    during active workflow pause operations.

    :param name: Unique name identifying the associated workflow or process.
    :type name: str
    """
    def __init__(self, name: str):
        """Initialize a UserIntention instance for a given workflow."""
        log.info(f"Creating User Intention object for {name}")
        self.name = name
        self._current_command = "IDLE"

    def checkCommand(self):
        """Check current command status and manage execution flow control.

        Blocks execution by sleeping when state is set to ``"PAUSE"`` until changed.

        :return: Returns ``0`` for normal execution or resumed states, and ``1`` if an ``"ABORT"`` command was issued.
        :rtype: int
        """
        if self._current_command == "IDLE":
            return 0
        if self._current_command == "ABORT":
            log.error("Phase Alignment Procedure Aborted by the User")
            return 1
        while self._current_command == "PAUSE":
            time.sleep(1)
        return 0

def getUserIntention(name):
    """Retrieve or instantiate a singleton :class:`UserIntention` instance for a given workflow.

    :param name: Unique identifier/name of the workflow.
    :type name: str
    :return: Active or newly instantiated UserIntention instance.
    :rtype: UserIntention
    """
    if name not in user_intentions.keys():
        user_intentions[name] = UserIntention(name)
    return user_intentions[name]
