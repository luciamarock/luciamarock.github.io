class UserIntention:
    """Generic container representing user control intentions, commands, and target parameters.

    """
    def __init__(self, name: str):
        log.info(f"Creating User Intention object for {name}")
        self.name = name

    def checkCommand(self):
        if self._current_command == "IDLE":
            return 0
        if self._current_command == "ABORT":
            log.error("Phase Alignment Procedure Aborted by the User")
            return 1
        while self._current_command == "PAUSE":
            time.sleep(1)
        return 0

def getUserIntention(name):
    if name not in user_intentions.keys():
        user_intentions[name] = UserIntention(name)
    return user_intentions[name]
