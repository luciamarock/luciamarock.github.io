class ScopeDevice:
    """Oscilloscope communication interface using SCPI over Ethernet sockets with mock fallback."""

    def __init__(self, ip_address, port=5025, timeout=5, mock=False):
        self.ip_address = ip_address
        self.port = port
        self.timeout = timeout
        self.mock = mock
        self.socket = None
        self.measurements = {}

        if not self.mock:
            self.connect()

    def connect(self):
        """Establishes TCP connection with the oscilloscope."""
        try:
            self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            self.socket.settimeout(self.timeout)
            self.socket.connect((self.ip_address, self.port))
            log.info(f"[Scope] Connected to oscilloscope at {self.ip_address}:{self.port}")
        except Exception as e:
            log.warning(f"[Scope] Failed to connect to {self.ip_address}:{self.port} ({e}). Falling back to MOCK mode.")
            self.mock = True

    def _send(self, command):
        """Sends a SCPI command string to the hardware."""
        if self.mock:
            log.debug(f"[Scope Mock SCPI] -> {command}")
            return
        if self.socket:
            self.socket.sendall(f"{command}\n".encode("utf-8"))

    def _query(self, command):
        """Sends a SCPI query and reads the response string."""
        if self.mock:
            # Simulated scalar value with realistic noise
            val = round(uniform(0.018, 0.022), 5)
            log.debug(f"[Scope Mock Query] -> {command} | Returning: {val}")
            return str(val)

        if self.socket:
            self._send(command)
            response = self.socket.recv(1024).decode("utf-8").strip()
            return response
        return "0.0"

    def createMeasure(self, slot: int, channel: int, meas_type: str = "rms"):
        """Configures a measurement slot for a specific channel and type."""
        self.measurements[slot] = {"channel": channel, "type": meas_type.upper()}
        scpi_cmd = f":MEASure:MEAS{slot}:SOURce CHANnel{channel}; :MEASure:MEAS{slot}:{meas_type.upper()}"
        self._send(scpi_cmd)
        log.info(f"[Scope] Created measurement slot {slot} on CH{channel} (type: {meas_type})")

    def getMeasValue(self, slot: int) -> float:
        """Reads current calculated measurement value from slot."""
        query_cmd = f":MEASure:MEAS{slot}:VALue?"
        raw_val = self._query(query_cmd)
        try:
            return float(raw_val)
        except ValueError:
            log.error(f"[Scope] Invalid numeric string received: {raw_val}")
            return 0.0
