#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Phoebus Display (.bob) XML Parser for EPICS PV Discovery.

This utility module provides XML parsing capabilities for Phoebus Display Builder (``.bob``) 
HMI screen files used in CS-Studio.

It enables the ACMS Template Editor to automatically discover all Process Variables (PVs) 
embedded within screen widgets (such as Text Updates, Action Buttons, and Gauges). 
The extracted PV list is deduplicated, filtered to remove unexpanded Phoebus macros, 
and sorted before being passed to :mod:`editor_interface` for live Channel Access profiling.

Module Dependencies:
    - :mod:`xml.etree.ElementTree`: Standard library XML parsing engine.

@author: luciamarock
"""

import xml.etree.ElementTree as ET
from pathlib import Path

import logging
log = logging.getLogger(__name__)

def extract_pvs_from_bob(bob_file_path: str) -> list:
    """Parse a CS-Studio Phoebus ``.bob`` display file and extract all configured PV names.

    Recursively scans the XML tree structure for ``<pv_name>`` tags, strips whitespace, 
    filters out unexpanded local macros (e.g., ``$(PV_NAME)``), deduplicates identical 
    widget channels, and returns a sorted list of concrete PV names.

    Args:
        bob_file_path (str or Path): File path to the target Phoebus ``.bob`` XML screen file.

    Returns:
        list of str: A sorted list of unique, valid EPICS Process Variable names extracted 
        from the file. Returns an empty list ``[]`` if XML parsing fails or if the file 
        cannot be found.

    Error Handling:
        Catches :class:`xml.etree.ElementTree.ParseError` and :class:`FileNotFoundError`, 
        logging an error message and returning an empty list to prevent application crashes.
    """
    pv_names = set() # Using a set automatically handles duplicate widgets using the same PV
    
    try:
        # Parse the XML structure of the Phoebus file
        tree = ET.parse(bob_file_path)
        root = tree.getroot()
        
        # Recursively find all <pv_name> elements anywhere in the file
        for pv_element in root.iter('pv_name'):
            if pv_element.text:
                clean_name = pv_element.text.strip()
                # Ignore empty strings or internal local macros like $(PV_NAME)
                if clean_name and not clean_name.startswith('$('):
                    pv_names.add(clean_name)
                    
        return sorted(list(pv_names))

    except ET.ParseError as e:
        log.error(f"Error: Failed to parse XML structure of {bob_file_path}: {e}")
        return []
    except FileNotFoundError:
        log.error(f"Error: Phoebus file not found at {bob_file_path}")
        return []
