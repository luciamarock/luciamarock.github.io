def extract_pvs_from_bob(bob_file_path: str) -> list:
    """
    Parses a Phoebus .bob file and extracts all configured PV names.
    """
    pv_names = set() # Using a set automatically handles duplicate widgets using the same PV
    
    try:
        # Parse the XML structure of the Phoebus file
        tree = ET.parse(bob_file_path)
        root = tree.getroot()
        
        # Recursively find all <pv_name> elements anywhere in the file
        for pv_element in root.iter('pv_name'):
            if pv_element.text:
                clean_name = pv_element.text.strip()
                # Ignore empty strings or internal local macros like $(PV_NAME)
                if clean_name and not clean_name.startswith('$('):
                    pv_names.add(clean_name)
                    
        return sorted(list(pv_names))

    except ET.ParseError as e:
        log.error(f"Error: Failed to parse XML structure of {bob_file_path}: {e}")
        return []
    except FileNotFoundError:
        log.error(f"Error: Phoebus file not found at {bob_file_path}")
        return []
