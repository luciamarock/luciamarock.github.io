#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""PostgreSQL Archive Database Interface for PV Discovery.

This module provides relational database access functions required by the ACMS Template Editor 
and Auto Conditioning Tool frameworks to retrieve archived EPICS Process Variable (PV) records.

It uses the modern ``psycopg 3`` library to open managed database connections and query 
archived channel inventories from target SQL databases (such as the local ``lipac_mock`` archive 
or production EPICS archiver appliances). The resulting PV lists are passed to 
:mod:`editor_interface` for multi-threaded Channel Access profiling.

Module Dependencies:
    - :mod:`psycopg`: PostgreSQL database adapter for Python.

:author: luciamarock
"""
import psycopg
import logging 
log = logging.getLogger(__name__)

def get_list_of_pvs(
    server: str = 'localhost',
    database: str = 'lipac_mock',
    username: str = 'archive',
    password: str = 'rockoDD1o',
    use_dashboard: bool = True,
    only_alive: bool = True
) -> list:
    """Connect to a PostgreSQL EPICS archive database and retrieve a sorted list of PV strings.

    Establishes a connection context using ``psycopg.connect()`` and executes targeted SQL 
    queries based on the specified source table and state filtering parameters:

    - When ``use_dashboard=True``, queries the ``pv_status`` dashboard table.
    - When ``use_dashboard=False``, queries the static ``channel`` archiver catalog table.

    :param server: Hostname or IP address of the PostgreSQL database server. Defaults to ``'localhost'``.
    :type server: str, optional
    :param database: Target SQL database name. Defaults to ``'lipac_mock'``.
    :type database: str, optional
    :param username: Database role user account. Defaults to ``'archive'``.
    :type username: str, optional
    :param password: Authentication password. Defaults to ``'rockoDD1o'``.
    :type password: str, optional
    :param use_dashboard: If ``True``, queries live status records from ``pv_status``. 
        If ``False``, queries registered channel definitions from ``channel``. Defaults to ``True``.
    :type use_dashboard: bool, optional
    :param only_alive: If ``True`` (and ``use_dashboard=True``), restricts results 
        to active PVs where ``alive = TRUE``. Defaults to ``True``.
    :type only_alive: bool, optional

    :return: A sorted list of unique EPICS PV names extracted from the database query.
    :rtype: list of str

    :raises psycopg.Error: If a database connection error, authentication failure, or SQL 
        execution exception occurs.
    """
    conn_str = f'postgresql://{username}:{password}@{server}/{database}'
    
    with psycopg.connect(conn_str) as conn:
        with conn.cursor() as cur:
            if use_dashboard:
                if only_alive:
                    log.info("Extracting only ALIVE current-state PVs from 'pv_status'...")
                    cur.execute("SELECT name FROM pv_status WHERE alive = TRUE ORDER BY name;")
                else:
                    log.info("Extracting ALL logged entries from 'pv_status'...")
                    cur.execute("SELECT name FROM pv_status ORDER BY name;")
            else:
                log.info("Extracting registered catalog names from 'channel'...")
                cur.execute("SELECT name FROM channel ORDER BY name;")
                
            # Safely unpack the single-value tuple rows into a list of strings
            return [row[0] for row in cur.fetchall()]
