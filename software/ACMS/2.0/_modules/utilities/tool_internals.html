class EPICSLogHandler(logging.Handler):
    """Custom logging handler that broadcasts log messages to EPICS Process Variables.

    Buffers recent log entries using fixed-length deques and publishes timestamps,
    log levels, and message chunks to defined EPICS record names via Channel Access.

    :param maxlen: Maximum number of recent log records to retain in ring buffers, defaults to 25.
    :type maxlen: int, optional
    :param max_str_len: Maximum length per string chunk for EPICS waveform/string compatibility, defaults to 39.
    :type max_str_len: int, optional
    """
    def __init__(self, maxlen=25, max_str_len=39):
        """Initialize EPICS log PV targets and internal ring buffers."""
        super().__init__()
        self.pv_timestamps = "ACT:LOG:TIMESTAMPS"
        self.pv_levels = "ACT:LOG:LEVELS"
        self.pv_messages_1 = "ACT:LOG:MESSAGES_1"
        self.pv_messages_2 = "ACT:LOG:MESSAGES_2"
        self.pv_messages_3 = "ACT:LOG:MESSAGES_3"
        self.buffer_timestamps = deque(maxlen=maxlen)
        self.buffer_levels = deque(maxlen=maxlen)
        self.buffer_messages_1 = deque(maxlen=maxlen)
        self.buffer_messages_2 = deque(maxlen=maxlen)
        self.buffer_messages_3 = deque(maxlen=maxlen)
        self.max_str_len = max_str_len

    def emit(self, record):
        """Process a log record, chunk strings into ASCII, and publish to EPICS PVs.

        :param record: Log record emitted by the Python logging framework.
        :type record: logging.LogRecord
        """
        try:
            times = f"{record.asctime}"
            levels = f"{record.levelname}"
            messages = f"{record.getMessage()}"
            times = times.encode("ascii", "replace").decode("ascii")  # Ensure ASCII only
            levels = levels.encode("ascii", "replace").decode("ascii")  # Ensure ASCII only
            messages = messages.encode("ascii", "replace").decode("ascii")  # Ensure ASCII only
            first_chunk = messages[:self.max_str_len]  # Truncate safely
            second_chunk = messages[self.max_str_len:2*self.max_str_len]
            third_chunk = messages[2*self.max_str_len:3*self.max_str_len]
            self.buffer_timestamps.appendleft(times)
            self.buffer_levels.appendleft(levels)
            self.buffer_messages_1.appendleft(first_chunk)
            self.buffer_messages_2.appendleft(second_chunk)
            self.buffer_messages_3.appendleft(third_chunk)
            epics.caput(self.pv_timestamps, list(self.buffer_timestamps))
            epics.caput(self.pv_levels, list(self.buffer_levels))
            epics.caput(self.pv_messages_1, list(self.buffer_messages_1))
            epics.caput(self.pv_messages_2, list(self.buffer_messages_2))
            epics.caput(self.pv_messages_3, list(self.buffer_messages_3))
        except Exception:
            self.handleError(record)

def element_to_dict(element):
    """Transform an XML Element object into a dictionary of child tag-text pairs.

    :param element: XML Element instance to transform.
    :type element: xml.etree.ElementTree.Element
    :return: Dictionary mapping child tag names to their inner text contents.
    :rtype: dict
    """
    result = {}
    for child in element:
        result[child.tag] = child.text
    return result

def element_to_list(list_element):
    """Transform an XML parent Element into a list of child text values (e.g., PV lists).

    :param list_element: XML Element containing list items as children.
    :type list_element: xml.etree.ElementTree.Element
    :return: List of text strings extracted from child elements.
    :rtype: list
    """
    result = []
    for pv_element in list_element:
        result.append(pv_element.text)
    return result

def parse_csv(file_path):
    """Parse a CSV file into a list of row dictionaries.

    :param file_path: Path to the target CSV file.
    :type file_path: str
    :return: List of dictionaries representing CSV rows keyed by column header.
    :rtype: list
    """
    data = []
    try:
        with open(file_path, 'r') as csvfile:
            reader = csv.DictReader(csvfile)  # Use DictReader for named columns
            for row in reader:
                data.append(row)
    except FileNotFoundError:
        print(f"Error: CSV file '{file_path}' not found.")
    except Exception as e:
        print(f"An unexpected error occurred: {e}")
    return data

def element_with_attributes_to_dict(element):
    """Transform an XML Element and its children into a dictionary including element attributes.

    :param element: XML Element instance to parse.
    :type element: xml.etree.ElementTree.Element
    :return: Dictionary mapping element text values to their attribute dictionaries.
    :rtype: dict
    """
    result = {}
    for child in element:
        if child.tag and child.text:  # Only process elements with text
            if child.attrib:
                result[child.text] = child.attrib
            else:
                result[child.text] = {} #If there are no attributes, set to empty dictionary
    return result

def parse_epics_db(file_path):
    """Parse an EPICS database (``.db``) text file into a structured dictionary.

    Extracts record types, record names, and nested field assignments using regular expressions.

    :param file_path: Path to the EPICS ``.db`` file.
    :type file_path: str
    :return: Dictionary containing parsed records with keys as record names, or None if file not found.
    :rtype: dict or None
    """
    records = {}
    record_pattern = re.compile(r'record\(([^,]+), "([^"]+)"\) {\s*(.*?)\s*}', re.DOTALL)
    field_pattern = re.compile(r'field\(([^,]+), "([^"]+)"\)')
    
    try:
        with open(file_path, 'r') as f:
            db_content = f.read()
    except FileNotFoundError:
        print(f"Error: File not found at {file_path}")
        return None  # Or raise an exception

    for record_match in record_pattern.finditer(db_content):
        record_type, record_name, fields_content = record_match.groups()
        record_data = {"type": record_type, "fields": {}}

        for field_match in field_pattern.finditer(fields_content):
            field_name, field_value = field_match.groups()
            record_data["fields"][field_name] = field_value

        records[record_name] = record_data

    return records

def parse_epics_xml_by_tag(epics_pv_root):
    """Parse an EPICS configuration XML tree by tag name.

    Extracts PV names from element attributes and descriptions from element inner text.

    :param epics_pv_root: Root XML Element containing PV definitions.
    :type epics_pv_root: xml.etree.ElementTree.Element
    :return: Dictionary structured as ``{tag: {'pv_name': ..., 'description': ...}}``.
    :rtype: dict
    """
    result = {}
    for child in epics_pv_root:
        tag = child.tag
        pv_name = child.attrib.get('name', None)
        description = (child.text or '').strip()
        result[tag] = {'pv_name': pv_name, 'description': description}
    return result

def calculate_min_stable_time(delta_v, ramp_rate, cavity_phase):
    """Calculate minimum required stabilization time based on voltage step, ramp rate, and cavity phase angle.

    :param delta_v: Voltage step magnitude.
    :type delta_v: float
    :param ramp_rate: Configured rate of voltage ramping.
    :type ramp_rate: float
    :param cavity_phase: Current cavity phase angle in degrees.
    :type cavity_phase: float
    :return: Minimum stabilization duration in seconds.
    :rtype: float
    """
    phi_radians = math.radians(cavity_phase)
    sin_phi = math.sin(phi_radians)
    cos_phi = math.cos(phi_radians)
    multiplier = max(sin_phi, cos_phi)
    if ramp_rate > 0:
        min_stable_time = delta_v / ramp_rate * multiplier
        if min_stable_time < 1.0:
            min_stable_time = 1.0 #TODO hardcoded 1 second (should be minimum default) different from default ?
    else:
        return 5.0 #TODO hardcoded 5 seconds (should be default)

    return min_stable_time
